 if(!r->maze_static_count){Builder b={.r=r,.offset=24U<<20,.meshes=r->maze_static};
  for(unsigned y=0;y<v->height*2;y+=4)for(unsigned x=0;x<v->width*2;x+=4){
   float x1=fminf(x+4,v->width*2),y1=fminf(y+4,v->height*2);
   if(!quad(&b,(L3Vec3){x,0,y},(L3Vec3){x,0,y1},(L3Vec3){x1,0,y1},(L3Vec3){x1,0,y},240,235,225))return 0;
  }
  for(unsigned y=0;y<v->height;y++)for(unsigned x=0;x<v->width;x++)if(v->cells[y*20+x].kind==L3_MINI_CELL_WALL)
   if(!box(&b,x*2+1,0,y*2+1,2,1.55f,2,256,256,256))return 0;
  if(!seal(&b)||b.offset>25U<<20)return 0;r->maze_static_count=b.n;
 }
 unsigned page=r->frame&1;Builder b={.r=r,.offset=(26U+page)<<20,.meshes=r->maze_dynamic};
 for(unsigned y=0;y<v->height;y++)for(unsigned x=0;x<v->width;x++){
  unsigned kind=v->cells[y*20+x].kind;if(kind!=L3_MINI_CELL_PELLET&&kind!=L3_MINI_CELL_POWER)continue;
  float size=kind==L3_MINI_CELL_POWER?.42f:.18f;
  if(!box(&b,x*2+1,.15f,y*2+1,size,kind==L3_MINI_CELL_POWER?.7f:.25f,size,256,220,90))return 0;
 }
 for(unsigned i=0;i<actors->count;i++){
  const L3MiniActor *a=&actors->actors[i];float x=a->x256/128.f,z=a->y256/128.f;
  if(i==0){
   if(!courier(&b,x,0,z,v->elapsed_ms*.018f))return 0;
   /* A thin, raised brass guide ring keeps the courier distinguishable from
    * the amber lanterns behind a wall. Same dynamic batch, 16 extra faces. */
   for(unsigned n=0;n<8;n++){
    float a=n*.78539816f,d=(n+1)*.78539816f;
    if(!quad(&b,(L3Vec3){x+.48f*cosf(a),1.68f,z+.48f*sinf(a)},
       (L3Vec3){x+.38f*cosf(a),1.68f,z+.38f*sinf(a)},
       (L3Vec3){x+.38f*cosf(d),1.68f,z+.38f*sinf(d)},
       (L3Vec3){x+.48f*cosf(d),1.68f,z+.48f*sinf(d)},256,244,171))return 0;
   }
  }
  else if(!phantom(&b,x,z,v->elapsed_ms*.001f,!!(a->flags&L3_MINI_ACTOR_FRIGHTENED)))return 0;
 }
 if(!seal(&b)||b.offset>((27U+page)<<20))return 0;r->maze_dynamic_count=b.n;return 1;
}
int l3_maze_render(L3Render *r,const L3Game *g){

 if(!r->maze_events.valid && !l3_maze_events_observe(&r->maze_events,&r->maze_motion,floor,NULL,r->presentation_seconds,r->presentation_epoch))return 0;
 if(!l3_maze_events_sample(&r->maze_events,r->presentation_seconds,floor))return 0;
 for(unsigned i=0;i<400;i++)v.cells[i].kind=floor[i];
 if(!l3_maze_motion_sample(&r->maze_motion,r->presentation_seconds,&actors)||!geometry(r,&v,&actors))return 0;
 /* A stable angled view keeps the whole house navigable. All coordinates are
  * world metres and the matrix is recomputed normally for G3D1 each frame. */
 L3Vec3 eye={v.width,42,v.height+9},target={v.width,0,v.height};
 L3Vec3 f=unit(sub(target,eye)),right=unit(cross(f,(L3Vec3){0,1,0})),down=cross(f,right),rows[]={right,down,f};

 if(!gpu_v1_target(&r->gpu,&t)||!gpu_v1_scissor(&r->gpu,0,0,r->width,r->height)||!gpu_v1_clear(&r->gpu,3,0x0864,65535,0,0,r->width,r->height))return 0;
 gpu_geom_camera cam={65536,80*65536,(int32_t)(r->focal*65536),(int32_t)(r->focal*65536),r->width/2,r->height/2};if(!gpu_geom_set_camera(&r->gpu,&cam))return 0;
 gpu_geom_transform m;for(unsigned a=0;a<3;a++){m.values[a*4]=(int32_t)lroundf(rows[a].x*65536);m.values[a*4+1]=(int32_t)lroundf(rows[a].y*65536);m.values[a*4+2]=(int32_t)lroundf(rows[a].z*65536);m.values[a*4+3]=(int32_t)lroundf(-dot(rows[a],eye)*65536);}

 if(!gpu_geom_set_transform(&r->gpu,&m)||!gpu_geom_set_light(&r->gpu,&light)||!gpu_v1_material(&r->gpu,&material)||!gpu_v1_texture(&r->gpu,&texture))return 0;
 for(unsigned i=0;i<r->maze_static_count;i++){if(!gpu_geom_draw(&r->gpu,&r->maze_static[i],1))return 0;r->draws++;r->triangles+=r->maze_static[i].face_count;}
 material.flags=2|4;if(!gpu_v1_material(&r->gpu,&material))return 0;
 for(unsigned i=0;i<r->maze_dynamic_count;i++){if(!gpu_geom_draw(&r->gpu,&r->maze_dynamic[i],1))return 0;r->draws++;r->triangles+=r->maze_dynamic[i].face_count;}
 return !r->failed;
}

static int triangle(Builder *b,L3Vec3 a,L3Vec3 c,L3Vec3 d,unsigned red,unsigned green,unsigned blue){
 if(b->nv+3>256&&!seal(b))return 0;L3Vec3 p[]={a,c,d},n=unit(cross(sub(c,a),sub(d,a)));unsigned v=b->nv;
 for(unsigned i=0;i<3;i++){b->v[b->nv]=(gpu_geom_vertex){0};if(!gpu_geom_fixed(p[i].x,&b->v[b->nv].x)||!gpu_geom_fixed(p[i].y,&b->v[b->nv].y)||!gpu_geom_fixed(p[i].z,&b->v[b->nv].z))return 0;b->nv++;}
 uint32_t *w=b->f[b->nf++].words;memset(w,0,32);w[0]=v|((v+1)<<8)|((v+2)<<16);w[4]=((uint32_t)((int)lroundf(n.x*32767)&65535)<<16)|((int)lroundf(n.y*32767)&65535);w[5]=(int)lroundf(n.z*32767)&65535;w[6]=red|(green<<9)|(blue<<18);return 1;
}
static int crystal(Builder *b,float x,float z,float height,unsigned sides,unsigned r,unsigned g,unsigned blue){
 for(unsigned i=0;i<sides;i++){
  float a=i*6.2831853f/sides,c=(i+1)*6.2831853f/sides;
  L3Vec3 p={x+.51f*cosf(a),.42f,z+.51f*sinf(a)},q={x+.51f*cosf(c),.42f,z+.51f*sinf(c)};
  L3Vec3 u={x+.45f*cosf(a),height*.72f,z+.45f*sinf(a)},v={x+.45f*cosf(c),height*.72f,z+.45f*sinf(c)};
  float facet=.68f+.32f*(i%3)/2.f;
  if(!quad(b,p,u,v,q,(unsigned)(r*facet),(unsigned)(g*facet),(unsigned)(blue*facet))||
     !triangle(b,u,(L3Vec3){x+.08f,height,z-.06f},v,r,g,blue)||
     !triangle(b,p,q,(L3Vec3){x,.23f,z},r/2,g/2,blue/2))return 0;
 }return 1;
}
static int barrel(Builder *b,float x,float y,float z,float angle){
 for(unsigned i=0;i<8;i++){
  float a=angle+i*.78539816f,c=angle+(i+1)*.78539816f;float radius=.34f;
  L3Vec3 p={x+radius*cosf(a),y+radius*sinf(a),z-.32f},q={x+radius*cosf(c),y+radius*sinf(c),z-.32f},u=p,v=q;u.z=v.z=z+.32f;
  unsigned r=i&1?180:235,g=i&1?125:180,blue=i&1?75:110;
  if(!quad(b,p,u,v,q,r,g,blue)||!triangle(b,(L3Vec3){x,y,z-.32f},p,q,130,100,75)||!triangle(b,(L3Vec3){x,y,z+.32f},v,u,130,100,75))return 0;
 }return 1;
}
