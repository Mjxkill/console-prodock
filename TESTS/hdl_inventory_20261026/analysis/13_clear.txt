   317 wire [15:1] usb_rxrdy ;
   318 wire [15:1] usb_txcork;
   319 wire [ 7:0] usb_txdat [15:1];
   320 wire [11:0] usb_txlen [15:1];
   321 wire [15:1] ep_clk    ;
   322 wire [15:1] ep_rx_rdy ;
   323 wire [15:1] ep_rx_dval;
   324 reg  [ 3:0] s_endpt_sel;
   325 reg         s_usb_txcork;
   326 reg         s_usb_rxrdy ;
   327 reg  [11:0] s_usb_txlen ;
   328 // Length/CORK are registered for s_endpt_sel. On an endpoint change,
   329 // never expose the preceding endpoint's offer to the controller. In
   330 // particular EP0's open zero length would become a spurious bulk ZLP.
   331 wire tx_endpoint_ready = i_usb_endpt == s_endpt_sel;
   332 // An empty IN poll may strobe POP while a newly arrived UART byte is
   333 // already in RAM but not yet visible through the registered length.
   334 // Advance packet RAM only for an advertised payload, never for that ZLP.
   335 wire usb_payload_pop = i_usb_txpop && tx_endpoint_ready &&
   336                        !s_usb_txcork && s_usb_txlen != 0;
   337 
   338 // Bulk IN retransmissions must keep exactly the original payload length.
   339 // After a lost host ACK the FIFO rewinds, but UART bytes may have arrived
   340 // meanwhile. Including them in the retry makes the host discard those NEW
   341 // bytes with the duplicate packet. Freeze the advertised length on its
   342 // first read and release it only when the controller acknowledges pktfin.
   343 reg [15:1] tx_pending;
   344 reg [11:0] tx_saved_len [15:1];
   345 // ACK changes pkt_rp immediately, but wrnum -> wrnum_d0 -> advertised
   346 // length takes three clocks to catch up. Do not publish that stale count
   347 // as a NEW packet when the next IN arrives immediately after retirement.
   348 reg [2:0] tx_retire;
   349 always @(posedge i_clk or posedge i_reset)begin
   350     if(i_reset)tx_retire<=0;
   351     else tx_retire<={tx_retire[1:0],i_usb_txpktfin && i_usb_endpt!=0};
   352 end
   353 `ifdef CONSOLE_USB_DIAG
   354 assign diagnostic_data={4'b0,tx_pending[2],s_usb_txcork,i_usb_txact,usb_payload_pop,
   355                        s_usb_txlen,usb_txlen[2]};
   356 `endif
   357 always @(posedge i_clk or posedge i_reset) begin
   358     if (i_reset) begin
   359         tx_pending <= 0;
   360     end
   361     else if (i_usb_endpt != 0) begin
   362         if (i_usb_txpktfin) begin
   363             tx_pending[i_usb_endpt] <= 1'b0;
   364         end
   365 `ifdef EP1_IN_PACKET
   366         // A flushed packet must not be replayed with its old length.
   367         else if (i_ep1_flush && i_usb_endpt == 4'd1) begin
   368             tx_pending[1] <= 1'b0;
   369         end
   370 `endif
   371         // An empty IN poll can strobe POP without transferring a payload.
   372         // Reserving its zero length would permanently block later arrivals.
   373         else if (usb_payload_pop && !tx_pending[i_usb_endpt]) begin
   374             tx_pending[i_usb_endpt] <= 1'b1;
   375             tx_saved_len[i_usb_endpt] <= s_usb_txlen;
   376         end
   377     end
   378 end
   379 `ifdef EP1_IN_PACKET
   380 wire ep1_reset = i_reset | i_ep1_flush;
   381 `else
   382 wire ep1_reset = i_reset;
   383 `endif
   384 `ifdef EP1_IN_EN
   385 wire [ 7:0] ep1_txdat;
   386 wire        ep1_txcork;
   387 wire [`EP1_IN_BUF_ASIZE:0] ep1_txlen;
   388 `ifdef EP1_IN_PACKET
   389 // Offer min(committed - acknowledged, max), and only once those bytes have
   390 // reached the packet RAM. A partially written block is never visible, so
   391 // the host cannot take it for a short packet. tx_retire hides the three
   392 // clocks during which ep1_offer and ep1_txlen still describe the packet
   393 // that was just acknowledged.
   394 reg  [15:0] ep1_sent;
   395 reg  [11:0] ep1_offer;
   396 wire [15:0] ep1_remaining = i_ep1_tx_committed - ep1_sent;
   397 always @(posedge i_clk or posedge i_reset) begin
   398     if (i_reset) begin
   399         ep1_sent  <= 16'd0;
   400         ep1_offer <= 12'd0;
   401     end
   402     else begin
   403         if (i_ep1_flush)
   404             ep1_sent <= 16'd0;
   405         else if (i_usb_txpktfin && i_usb_endpt == 4'd1 && tx_pending[1])
   406             ep1_sent <= ep1_sent + {4'd0, tx_saved_len[1]};
   407         ep1_offer <= (i_ep1_flush || ep1_remaining >= 16'h8000) ? 12'd0 :
   408                      (ep1_remaining >= {4'd0, i_ep1_tx_max}) ? i_ep1_tx_max : ep1_remaining[11:0];
   409     end
   410 end
   411 assign o_ep1_tx_sent = ep1_sent;
   412 assign usb_txlen[1]  = (ep1_txlen >= {1'b0, ep1_offer}) ? ep1_offer : 12'd0;
   413 `else
   414 assign usb_txlen[1]  = (ep1_txlen >= i_ep1_tx_max) ? i_ep1_tx_max : ep1_txlen[11:0];

  1384 module usb_tx_buf #(

  1424     assign c_fifo_wr_data = i_ep_tx_data;
  1425     wire c_fifo_empty;
  1426     generate if (P_SYNC == 0) begin : g_cross
  1427     clk_cross_fifo #(
  1428        .DSIZE (8  )
  1429       ,.ASIZE (6  )
  1430       ,.AEMPT (1  )
  1431       ,.AFULL (32 )
  1432     )clk_cross_fifo
  1433     (
  1434          .WrClock    (i_ep_clk      )
  1435         ,.Reset      (i_reset       )
  1436         ,.WrEn       (c_fifo_wr     )
  1437         ,.Data       (c_fifo_wr_data)
  1438         ,.AlmostFull (              )
  1439         ,.Full       ()
  1440         ,.RdClock    (i_clk         )
  1441         ,.RPReset    (i_reset       )
  1442         ,.RdEn       (c_fifo_rd     )
  1443         ,.Q          (c_fifo_rd_data)
  1444         ,.AlmostEmpty()
  1445         ,.Empty      (c_fifo_empty  )
  1446     );
  1447 
  1448 
  1449 
  1450     end else begin : g_sync
  1451         assign c_fifo_empty   = 1'b1;
  1452         assign c_fifo_rd_data = {P_DSIZE{1'b0}};
  1453     end endgenerate
  1454     always@(posedge i_clk, posedge i_reset) begin
  1455         if (i_reset) begin
  1456             c_fifo_rd <= 1'b0;
  1457         end
  1458         else begin
  1459             if (c_fifo_empty) begin
  1460                 c_fifo_rd <= 1'b0;
  1461             end
  1462             else begin
  1463                 c_fifo_rd <= 1'b1;
  1464             end
  1465         end
  1466     end
  1467     always@(posedge i_clk, posedge i_reset) begin
  1468         if (i_reset) begin
  1469             c_fifo_rd_dval <= 1'b0;
  1470         end
  1471         else begin
  1472             c_fifo_rd_dval <= c_fifo_rd & (!c_fifo_empty);
  1473         end
  1474     end
  1475 
  1476 //==============================================================
  1477 //======usb packet crc check fifo
  1478 assign pkt_fifo_wr        = P_SYNC ? i_ep_tx_dval : c_fifo_rd_dval;
  1479 assign pkt_fifo_wr_data   = P_SYNC ? i_ep_tx_data : c_fifo_rd_data;
  1480 assign pkt_fifo_rd_pktfin = i_usb_txpktfin&(i_usb_endpt==P_ENDPOINT);
  1481 assign pkt_fifo_rd_act    = i_usb_txact&(i_usb_endpt==P_ENDPOINT);
  1482 assign pkt_fifo_rd        = i_usb_txpop&(i_usb_endpt==P_ENDPOINT);
  1483 assign o_usb_txdat        = pkt_fifo_rd_data;
  1484 assign o_usb_txcork       = pkt_fifo_empty;
  1485 assign o_usb_txlen        = pkt_fifo_wr_num_d0;
  1486 
  1487     always@(posedge i_clk, posedge i_reset) begin
  1488         if (i_reset) begin
  1489             pkt_fifo_wr_num_d0 <= 1'b0;
  1490         end
  1491         else begin
  1492             pkt_fifo_wr_num_d0 <= pkt_fifo_wr_num;
  1493         end
  1494     end
  1495 
  1496     sync_tx_pkt_fifo #(
  1497          .DSIZE (P_DSIZE)
  1498         ,.ASIZE (P_ASIZE)
  1499     )sync_tx_pkt_fifo
  1500     (
  1501          .CLK   (i_clk              )
  1502         ,.RSTn  (!i_reset           )
  1503         ,.write (pkt_fifo_wr        )
  1504         ,.iData (pkt_fifo_wr_data   )
  1505         ,.pktfin(pkt_fifo_rd_pktfin )
  1506         ,.txact (pkt_fifo_rd_act    )
  1507         ,.read  (pkt_fifo_rd        )
  1508         ,.oData (pkt_fifo_rd_data   )
  1509         ,.wrnum (pkt_fifo_wr_num    )
  1510         ,.full  (                   )
  1511         ,.empty (pkt_fifo_empty     )
  1512     );
  1513 endmodule

  1691 module clk_cross_fifo (
  1692       Q,
  1693       Full,
  1694       Empty,
  1695       AlmostEmpty,
  1696       AlmostFull,
  1697       Data,
  1698       WrEn,
  1699       WrClock,
  1700       Reset,
  1701       RdEn,
  1702       RdClock,
  1703       RPReset
  1704       );
  1705       parameter             DSIZE = 8;
  1706       parameter             ASIZE = 6;
  1707       parameter             AEMPT = 1;
  1708       parameter             AFULL = 32;
  1709 
  1710       output  [DSIZE-1:0]   Q;
  1711       output                Full;
  1712       output                Empty;
  1713       output                AlmostEmpty;
  1714       output                AlmostFull;
  1715       input   [DSIZE-1:0]   Data;
  1716       input                 WrEn;
  1717       input                 WrClock;
  1718       input                 Reset;
  1719       input                 RdEn;
  1720       input                 RdClock;
  1721       input                 RPReset;
  1722   
  1723       reg     [DSIZE-1:0]   Q;
  1724       reg                   AlmostEmpty;
  1725       reg                   AlmostFull;
  1726       reg                   Full;
  1727       reg                   Empty;
  1728       reg       [ASIZE:0]   wptr;
  1729       reg       [ASIZE:0]   rptr; 
  1730       reg       [ASIZE:0]   wq2_rptr;
  1731       reg       [ASIZE:0]   rq2_wptr; 
  1732       reg       [ASIZE:0]   wq1_rptr;
  1733       reg       [ASIZE:0]   rq1_wptr;
  1734       reg       [ASIZE:0]   rbin;
  1735       reg       [ASIZE:0]   wbin;
  1736 
  1737       reg       [DSIZE-1:0] mem[0:(1<<ASIZE)-1];
  1738 
  1739       wire      [ASIZE-1:0] waddr;
  1740       wire      [ASIZE-1:0] raddr;
  1741       wire      [ASIZE:0]   rgraynext;
  1742       wire      [ASIZE:0]   rbinnext;
  1743       wire      [ASIZE:0]   wgraynext;
  1744       wire      [ASIZE:0]   wbinnext;
  1745       wire                  rempty_val;
  1746       wire                  wfull_val;
  1747       wire      [ASIZE:0]   wcount_r;
  1748       wire      [ASIZE:0]   rcnt_sub;
  1749       wire      [ASIZE:0]   rcount_w;
  1750       wire      [ASIZE:0]   wcnt_sub;    
  1751 
  1752       always@(posedge RdClock)
  1753       begin
  1754         if(RdEn)
  1755           Q <= mem[raddr];
  1756       end
  1757 
  1758       always@(posedge WrClock)
  1759       begin
  1760         if(WrEn && !Full)
  1761           mem[waddr] <= Data;
  1762       end
  1763 
  1764       always @(posedge WrClock or posedge Reset)
  1765       begin
  1766         if (Reset)
  1767           {wq2_rptr,wq1_rptr} <= 0;
  1768         else
  1769           {wq2_rptr,wq1_rptr} <= {wq1_rptr,rptr};
  1770       end
  1771 
  1772       always @(posedge RdClock or posedge RPReset)
  1773       begin
  1774         if (RPReset)
  1775           {rq2_wptr,rq1_wptr} <= 0;
  1776         else
  1777           {rq2_wptr,rq1_wptr} <= {rq1_wptr,wptr};
  1778       end
  1779 
  1780       always @(posedge RdClock or posedge RPReset)
  1781       begin
  1782         if (RPReset)
  1783           {rbin, rptr} <= 0;
  1784         else
  1785           {rbin, rptr} <= {rbinnext, rgraynext};
  1786       end
  1787 
  1788       assign raddr      = rbin[ASIZE-1:0];
  1789       assign rbinnext   = rbin + (RdEn & ~Empty);
  1790       assign rgraynext  = (rbinnext>>1) ^ rbinnext;
  1791       assign rempty_val = (rgraynext == rq2_wptr);
  1792 
  1793       assign wcount_r   = gry2bin(rq2_wptr);
  1794       assign rcnt_sub   = {(wcount_r[ASIZE] ^ rbinnext[ASIZE]), wcount_r[ASIZE-1:0]} - {1'b0, rbinnext[ASIZE-1:0]};
  1795       assign arempty_val= rcnt_sub <= AEMPT;
  1796 
  1797       always @(posedge RdClock or posedge RPReset)
  1798       begin
  1799       if (RPReset)
  1800         Empty <= 1'b1;
  1801       else
  1802         Empty <= rempty_val;
  1803       end
  1804 
  1805       always @(posedge RdClock or posedge RPReset)
  1806       begin
  1807       if (RPReset)
  1808         AlmostEmpty <= 1'b1;
  1809       else
  1810         AlmostEmpty <= arempty_val;
  1811       end
  1812  
  1813       always @(posedge WrClock or posedge Reset)
  1814       begin
  1815         if (Reset)
  1816           {wbin, wptr} <= 0;
  1817         else
  1818           {wbin, wptr} <= {wbinnext, wgraynext};
  1819       end
  1820    
  1821       assign waddr      = wbin[ASIZE-1:0];
  1822       assign wbinnext   = wbin + (WrEn & ~Full);
  1823       assign wgraynext  = (wbinnext>>1) ^ wbinnext;
  1824       assign wfull_val  = (wgraynext == {~wq2_rptr[ASIZE:ASIZE-1],
  1825                                           wq2_rptr[ASIZE-2:0]});
  1826   
  1827       assign rcount_w   = gry2bin(wq2_rptr);
  1828       assign wcnt_sub   = {(rcount_w[ASIZE] ^ wbinnext[ASIZE]), wbinnext[ASIZE-1:0]} - {1'b0, rcount_w[ASIZE-1:0]};
  1829       assign awfull_val = wcnt_sub >= AFULL;
  1830 
  1831       always @(posedge WrClock or posedge Reset)
  1832       begin
  1833         if (Reset)
  1834           Full <= 1'b0;
  1835         else
  1836           Full <= wfull_val;
  1837       end
  1838 
  1839       always @(posedge WrClock or posedge Reset)
  1840       begin
  1841         if (Reset)
  1842           AlmostFull <= 1'b0;
  1843         else
  1844           AlmostFull <= awfull_val;
  1845       end
  1846 
  1847       function [ASIZE:0]gry2bin;
  1848         input [ASIZE:0] gry_code;
  1849         integer         i;
  1850         begin
  1851           gry2bin[ASIZE]=gry_code[ASIZE];    
  1852           for(i=ASIZE-1;i>=0;i=i-1)        
  1853             gry2bin[i]=gry2bin[i+1]^gry_code[i];
  1854         end
  1855       endfunction
