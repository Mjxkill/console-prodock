   155 // Parameter definitions
   156 localparam [4: 0]  IDLE               = 4'd0;
   157 localparam [4: 0]  SELECT_LANE        = 4'd1;
   158 localparam [4: 0]  SELECT_LANE_WAIT_1 = 4'd2;
   159 localparam [4: 0]  SELECT_LANE_WAIT_2 = 4'd3;
   160 localparam [4: 0]  SELECT_LANE_CHK    = 4'd4;

   162 localparam [4: 0]  LPDDR_WRITING_2    = 4'd6;

   164 localparam [4: 0]  LPDDR_WR_END       = 4'd8;

   169 /////////////////////////////////////////////////////
   170 // Register definitions
   171 reg                 [3 : 0] c_status;
   172 reg                 [3 : 0] n_status;
   173 
   174 reg                         wr_req;
   175 reg                         rd_req;
   176 reg                         ahb_req;
   177 reg                         wr_data_end;
   178 reg                         rd_data_end;
   179 reg                         ahb_end;

   182 reg                [31 : 0] cmd_addr;
   183 reg               [127 : 0] wr_data;
   184 wire              [127 : 0] lpddr_rd_data;
   185 wire                        lpddr_rd_valid;
   186 wire                        lpddr_rd_end;

   188

   190 reg                         lpddr_cmd;
   191 wire                        lpddr_cmd_ready;
   192 wire                        lpddr_data_ready;
   193 reg                         lpddr_wdata_en;
   194 reg                         lpddr_wdata_end;
   195 
   196 // DDR work signals
   197 reg                         lpddr_ready;
   198 reg                 [7 : 0] lpddr_ready_cnt;
   199 reg                 [7 : 0] lpddr_data_end_cnt;

   201 // AHB bus signals
   202 wire                        lpddr_ahb_end;
   203 wire                        lpddr_ahb_req;
   204 wire               [31 : 0] lpddr_addr_ahb;
   205 wire                        lpddr_cmd_en_ahb;
   206 wire                        lpddr_cmd_ahb;
   207 wire                        lpddr_wdata_en_ahb;
   208 wire                        lpddr_wdata_end_ahb;
   209 wire              [127 : 0] lpddr_wr_data_ahb;
   210 reg               [127 : 0] rd_data_lpddr_ahb;
   211 reg                         rd_valid_p_ahb;
   212 wire               [15 : 0] lpddr_wdata_mask_ahb;

   215 //////////////////////////////////////////////////////////
   216 //================= Need User Define ===================//
   217 // LANE 0 WRITE Signal
   218 reg                         rst_n_d1;
   219 reg                         rst_n_d2;
   220 reg                         lpddr_rst_n;
   221 wire                        ddr_clk;
   222 
   223 always @ (posedge ddr_clk or negedge rst_n) begin
   224     if(!rst_n) begin
   225         rst_n_d1    <= 1'b0;
   226         rst_n_d2    <= 1'b0;
   227         lpddr_rst_n <= 1'b0;
   228     end 
   229     else  begin
   230         rst_n_d1    <= 1'b1;
   231         rst_n_d2    <= rst_n_d1;
   232         lpddr_rst_n <= rst_n_d2;
   233     end
   234 end
   235 
   236 always @ (posedge ddr_clk or negedge lpddr_rst_n) begin
   237     if(!lpddr_rst_n)
   238         lpddr_ready <= 1'b0;
   239     else if(init_calib_complete)
   240         lpddr_ready <= 1'b1;
   241     else
   242         lpddr_ready <= 1'b0;
   243 end
   244 
   245 always @ (posedge ddr_clk or negedge lpddr_rst_n) begin
   246     if(!lpddr_rst_n)
   247         c_status <= IDLE;
   248     else
   249         c_status <= n_status;
   250 end
   251 
   252 always @ (*) begin
   253     case (c_status)
   254         IDLE: begin
   255             if ( lpddr_ready == 1'b1 )
   256                 n_status = SELECT_LANE;
   257             else n_status = IDLE;
   258         end
   259 
   260         SELECT_LANE: begin
   261             n_status = SELECT_LANE_CHK;
   262         end
   263 
   264         SELECT_LANE_CHK: begin
   265             if( lpddr_cmd_ready && lpddr_data_ready && ahb_req )
   266                 n_status = LPDDR_AHB;
   267             else if(ahb_req == 1'b0)
   268                 n_status = IDLE;
   269             else 
   270                 n_status = SELECT_LANE;
   271         end
   272 
   273         LPDDR_AHB: begin
   274             if (ahb_end)
   275                 n_status = SELECT_LANE;
   276             else 
   277                 n_status = LPDDR_AHB;
   278         end
   279 
   280         default: begin
   281             n_status = IDLE;
   282         end
   283 
   284     endcase
   285 end
   286 
   287 // Calculate Lane Number & ADDR & Burst Count
   288 always @(posedge ddr_clk or negedge lpddr_rst_n) begin
   289     if(!lpddr_rst_n) begin
   290         ahb_end               <= 1'b0;
   291         ahb_req               <= 1'b0;
   292     end
   293     else begin
   294         ahb_end               <= lpddr_ahb_end;
   295         ahb_req               <= lpddr_ahb_req;
   296     end
   297 end
   298 
   299 
   300 // DDR ADDR & Write Data
   301 always @ (posedge ddr_clk or negedge lpddr_rst_n) begin
   302     if(!lpddr_rst_n) begin
   303         cmd_addr               <= {32{1'b0}};
   304         wr_data                <= 'h0000_0000_0000_0000;
   305         lpddr_cmd_en           <= 1'b0;
   306         lpddr_cmd              <= 1'b0;
   307         lpddr_wdata_en         <= 1'b0;
   308         lpddr_wdata_end        <= 1'b0;
   309     end
   310     else begin
   311         cmd_addr               <= lpddr_addr_ahb;
   312         wr_data                <= lpddr_wr_data_ahb;
   313         lpddr_cmd_en           <= lpddr_cmd_en_ahb;
   314         lpddr_cmd              <= lpddr_cmd_ahb;
   315         lpddr_wdata_en         <= lpddr_wdata_en_ahb;
   316         lpddr_wdata_end        <= lpddr_wdata_end_ahb;
   317     end
   318 end
   319 
   320 always @ (posedge ddr_clk or negedge lpddr_rst_n) begin
   321     if(!lpddr_rst_n)
   322         lpddr_wdata_mask        <= 8'h00;
   323     else 
   324         lpddr_wdata_mask        <= lpddr_wdata_mask_ahb;
   325 end
   326 
   327 // Read Valid & Read Data from DDR
   328 always @ (posedge ddr_clk or negedge lpddr_rst_n) begin
   329     if(!lpddr_rst_n) begin
   330         rd_valid_p_ahb         <= 1'b0;
   331     end
   332     else begin
   333         rd_valid_p_ahb          <= lpddr_rd_valid;
   334     end
   335 end
