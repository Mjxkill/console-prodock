## File 1: `opencode_event_accounting.py` (~470 physical lines)

```python
"""Pure parser for OpenCode ``run --format json`` JSONL event streams.

Observed-only multi-step accounting for bounded agent runs.  No I/O, no
subprocesses, no clock, no randomness: every value is decoded from the
JSONL text passed in, and every derived quantity is either an exactly
stated arithmetic rule or is isolated under ``derived`` with its basis.

Evidence vs. hypotheses
-----------------------
- Evidence: event ids, session ids, terminal reasons, tool names/statuses,
  exit codes, token counters, cost numbers, stream errors.
- Derivation: pairing of starts with finishes, gated sums, and
  ``reached_step_limit`` (only when the caller supplies the configured
  maximum).  Derivations never overwrite observations.

Hard limits (explicit)
----------------------
1. The parser never decides task success.  OpenCode exit code 0 is out of
   scope here: the caller must separately verify process exit, the final
   allowlist diff of owned files, and independently launched tests
   (packet ruling).  Nothing in the result implies success.
2. The exact ``tool_use`` ``part`` sub-schema and the session-id field
   name are NOT confirmed by the packet.  This parser resolves a fixed
   list of alternative spellings, records which path matched, and stores
   a verbatim JSON excerpt of every tool part.  No other fields are read.
3. Per the packet, API HTTP attempts must not be inferred from text.
   ``model_requests_started`` / ``model_requests_finished`` count only
   observed ``step_start`` / ``step_finish`` events; they are never
   labelled HTTP requests, and they are reported under keys distinct
   from step counts and tool counts.
4. ``reached_step_limit`` is ``None`` (unknown) unless the caller supplies
   ``expected_max_steps``; it is never inferred from stream shape alone.
5. Token/cost totals sum only when every unique ``step_finish`` reports
   the field; otherwise the total stays ``None`` (no partial sums, no
   zero filling).
6. Duplicate ids are counted once.  A conflicting duplicate is recorded
   as an error and the first observation is retained.  A duplicated
   ``step_finish`` can never hide a ``length`` truncation.
7. Tool output is stored as a verbatim prefix plus an explicit truncation
   marker.  Nothing is synthesized; no quality scores are produced.
"""

from __future__ import annotations

import json
import math

PROVIDER = "opencode-jsonl"

MAX_NUMBER = 10 ** 12
ERROR_LINE_LIMIT = 200
TOOL_OUTPUT_EXCERPT_CHARS = 500
TOOL_RAW_EXCERPT_CHARS = 2000

# Terminal reasons observed for step_finish (evidence: prior review parser).
TERMINAL_REASONS = ("stop", "length")

# Defensive status vocabulary.  The packet does not confirm the exact
# status strings of tool_use parts; every observed status is also stored
# verbatim in ``statuses_seen`` for independent audit.
TOOL_TERMINAL_STATUSES = frozenset(
    {"completed", "error", "failed", "success", "done"})
TOOL_FAILURE_STATUSES = frozenset({"error", "failed"})

# Candidate paths (inside a tool part) where an integer exit code may live.
# First path resolving to an integer wins; the path used is recorded in
# ``exit_code_source``.  Nothing outside these paths is read.
_EXIT_CODE_PATHS = (
    ("state", "metadata", "exit"),
    ("state", "metadata", "exitCode"),
    ("state", "exitCode"),
    ("state", "exit"),
    ("metadata", "exitCode"),
    ("exitCode",),
    ("exit",),
)

_UNPAIRABLE = object()  # sentinel: an id field is present but invalid


def _valid_number(value, label, errors):
    """Validated numeric read; ``None`` is legal (field simply absent)."""
    if value is None:
        return None
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        errors.append(f"invalid numeric field {label}: {value!r}")
        return None
    if value < 0 or value > MAX_NUMBER or not math.isfinite(value):
        errors.append(f"invalid numeric field {label}: {value!r}")
        return None
    if label.startswith(("tokens.", "usage.")) and not isinstance(value, int):
        errors.append(f"noninteger token count {label}")
        return None
    return value


def _excerpt(value, limit):
    """Verbatim JSON prefix plus an explicit marker.  No synthesis."""
    try:
        text = json.dumps(value, sort_keys=True, default=str)
    except (TypeError, ValueError):
        text = repr(value)
    if len(text) > limit:
        return text[:limit] + "...[verbatim prefix; truncated by parser]"
    return text


def _empty_tokens():
    return {
        "input": None,
        "output": None,
        "reasoning": None,
        "total": None,
        "cache_read": None,
        "cache_write": None,
    }


def _resolve_key(mapping, candidates):
    """First *present* key's value (present means ``key in mapping``)."""
    if not isinstance(mapping, dict):
        return None
    for key in candidates:
        if key in mapping:
            return mapping[key]
    return None


def _dig(mapping, path):
    """Traverse a dict path; return ``(value, found)``."""
    node = mapping
    for key in path:
        if not isinstance(node, dict) or key not in node:
            return None, False
        node = node[key]
    return node, True


def _session_value(event, part, errors, order):
    """First resolvable session id; ``_UNPAIRABLE`` means present but invalid."""
    for source, label in ((event, "event"), (part, "part")):
        if not isinstance(source, dict):
            continue
        value = _resolve_key(source, ("sessionID", "session_id", "sessionId"))
        if value is None:
            continue
        if isinstance(value, str) and value:
            return value
        errors.append(f"line {order}: invalid {label} session id {value!r}")
        return _UNPAIRABLE
    return None


def _pair_steps(starts, finishes, errors):
    """Pair step_start/step_finish observations (two-phase, deterministic).

    Rules (documented so behaviour is auditable, not guessed):
    - identical string ids pair regardless of arrival order;
    - id-less starts and id-less finishes pair first-in/first-out;
    - an id-less finish never steals an id-keyed start, and vice versa;
    - anything unpaired yields an error entry and is never silently merged.

    Returns ``(pairs, orphan_finishes)`` where each pair is
    ``(start_record, finish_record)``.
    """
    start_by_id = {r["id"]: r for r in starts if isinstance(r["id"], str)}
    records = []
    paired_finish_orders = set()

    for fin in finishes:
        fid = fin["id"]
        if not isinstance(fid, str):
            continue
        start_record = start_by_id.get(fid)
        if start_record is None:
            errors.append(
                f"line {fin['order']}: step finish without start (id {fid!r})")
        else:
            records.append((start_record, fin))
            paired_finish_orders.add(fin["order"])

    anon_starts = [r for r in starts if r["id"] is None]
    anon_finishes = [f for f in finishes if f["id"] is None]
    for start_record, fin in zip(anon_starts, anon_finishes):
        records.append((start_record, fin))
        paired_finish_orders.add(fin["order"])
    for fin in anon_finishes[len(anon_starts):]:
        errors.append(f"line {fin['order']}: step finish without start (no id)")
    for start_record in anon_starts[len(anon_finishes):]:
        errors.append(
            f"line {start_record['order']}: step start without finish (no id)")

    paired_start_orders = {pair[0]["order"] for pair in records}
    for start_record in starts:
        sid = start_record["id"]
        if isinstance(sid, str) and start_record["order"] not in paired_start_orders:
            errors.append(
                f"line {start_record['order']}: step start without finish "
                f"(id {sid!r})")
        elif sid is _UNPAIRABLE:
            errors.append(
                f"line {start_record['order']}: step start without finish "
                f"(invalid id)")
    for fin in finishes:
        if fin["id"] is _UNPAIRABLE and fin["order"] not in paired_finish_orders:
            errors.append(
                f"line {fin['order']}: step finish without start (invalid id)")

    records.sort(key=lambda pair: pair[0]["order"])
    orphans = [f for f in finishes if f["order"] not in paired_finish_orders]
    return records, orphans


def _parse_opencode(stdout, errors, expected_session_id=None,
                    expected_max_steps=None, task_id=None, run_id=None):
    lines_total = 0
    lines_invalid_json = 0
    lines_non_object = 0
    duplicate_start_events = 0
    duplicate_finish_events = 0
    tool_repeat_updates = 0
    anon_tool_events = 0
    events_without_session_id = 0
    events_with_invalid_session_id = 0
    session_ids_observed = []
    session_mismatch_events = []
    unrecognized_types = {}

    starts = []       # {"order", "id": str|None|_UNPAIRABLE, "part"}
    finishes = []     # {"order", "id", "reason", "tokens", "cost"}
    tool_events = []  # {"order", "id": str|None, "part"}
    texts = []
    seen_texts = {}
    seen_start_parts = {}
    seen_finish_parts = {}
    truncation_seen = False

    for raw in (stdout or "").splitlines():
        line = raw.strip()
        if not line:
            continue
        lines_total += 1
        try:
            event = json.loads(line)
        except (json.JSONDecodeError, ValueError):
            lines_invalid_json += 1
            errors.append(f"invalid JSON: {line[:ERROR_LINE_LIMIT]}")
            continue
        if not isinstance(event, dict):
            lines_non_object += 1
            errors.append(f"non-object JSON line: {line[:ERROR_LINE_LIMIT]}")
            continue

        order = lines_total
        part = event.get("part")
        part = part if isinstance(part, dict) else {}

        session_value = _session_value(event, part, errors, order)
        if session_value is _UNPAIRABLE:
            events_with_invalid_session_id += 1
        elif session_value is None:
            events_without_session_id += 1
        else:
            if session_value not in session_ids_observed:
                session_ids_observed.append(session_value)
            if expected_session_id is not None and session_value != expected_session_id:
                session_mismatch_events.append(
                    {"order": order, "observed": session_value})
                errors.append(
                    f"line {order}: session mismatch: expected "
                    f"{expected_session_id!r}, observed {session_value!r}")

        event_type = event.get("type")
        if not isinstance(event_type, str):
            errors.append(
                f"line {order}: missing or invalid event type {event_type!r}")
            continue

        if event_type == "step_start":
            sid = part.get("id")
            if sid is not None and not isinstance(sid, str):
                errors.append(f"line {order}: invalid start id {sid!r}")
                sid = _UNPAIRABLE
            if isinstance(sid, str):
                if sid in seen_start_parts:
                    duplicate_start_events += 1
                    if seen_start_parts[sid] != part:
                        errors.append(
                            f"line {order}: conflicting repeated start id {sid!r}")
                    continue
                seen_start_parts[sid] = part
            starts.append({"order": order, "id": sid, "part": part})

        elif event_type == "step_finish":
            fid = part.get("id")
            if fid is not None and not isinstance(fid, str):
                errors.append(f"line {order}: invalid finish id {fid!r}")
                fid = _UNPAIRABLE
            reason = part.get("reason")
            if reason is None:
                reason = event.get("reason")
            if reason is not None and not isinstance(reason, str):
                errors.append(f"line {order}: invalid finish reason {reason!r}")
                reason = None
            # Hardening carried over: even a duplicated or conflicting
            # finish cannot hide a length truncation.
            if reason == "length":
                truncation_seen = True
            if isinstance(fid, str):
                if fid in seen_finish_parts:
                    duplicate_finish_events += 1
                    if seen_finish_parts[fid] != part:
                        errors.append(
                            f"line {order}: conflicting repeated finish id {fid!r}")
                    continue
                seen_finish_parts[fid] = part

            token_data = part.get("tokens")
            if not isinstance(token_data, dict):
                token_data = event.get("tokens")
            token_data = token_data if isinstance(token_data, dict) else {}
            cache = token_data.get("cache")
            cache = cache if isinstance(cache, dict) else {}
            cache_read = cache.get("read")
            if cache_read is None:
                cache_read = token_data.get("cache_read")
            if cache_read is None:
                cache_read = token_data.get("cacheRead")
            cache_write = cache.get("write")
            if cache_write is None:
                cache_write = token_data.get("cache_write")
            if cache_write is None:
                cache_write = token_data.get("cacheWrite")

            observed_tokens = {
                name: _valid_number(token_data.get(name), "tokens." + name, errors)
                for name in ("input", "output", "reasoning", "total")
            }
            observed_tokens["cache_read"] = _valid_number(
                cache_read, "tokens.cache.read", errors)
            observed_tokens["cache_write"] = _valid_number(
                cache_write, "tokens.cache.write", errors)
            observed_cost = _valid_number(
                part.get("cost", event.get("cost")), "cost", errors)
            finishes.append({
                "order": order,
                "id": fid,
                "reason": reason,
                "tokens": observed_tokens,
                "cost": observed_cost,
            })

        elif event_type == "tool_use":
            tid = None
            for key in ("id", "callID", "call_id"):
                if key in part and part[key] is not None:
                    candidate = part[key]
                    if isinstance(candidate, str) and candidate:
                        tid = candidate
                    else:
                        errors.append(
                            f"line {order}: invalid tool id in field {key!r}")
                    break
            if tid is None:
                anon_tool_events += 1
            tool_events.append({"order": order, "id": tid, "part": part})

        elif event_type == "text":
            part_text = part.get("text")
            if not isinstance(part_text, str):
                errors.append(f"line {order}: text event without string text")
                continue
            part_id = part.get("id")
            if part_id is not None and not isinstance(part_id, str):
                errors.append(f"line {order}: invalid text part id {part_id!r}")
                part_id = None
            if isinstance(part_id, str):
                if part_id in seen_texts:
                    if seen_texts[part_id] != part_text:
                        errors.append(
                            f"line {order}: conflicting repeated text part id "
                            f"{part_id!r}")
                    continue
                seen_texts[part_id] = part_text
            texts.append({"id": part_id, "text": part_text})

        elif event_type == "error":
            errors.append(
                f"line {order}: stream error event: "
                f"{_excerpt(event.get('error', event), ERROR_LINE_LIMIT)}")

        else:
            unrecognized_types[event_type] = (
                unrecognized_types.get(event_type, 0) + 1)

    # ---- pair steps -------------------------------------------------------
    pairs, orphan_finish_records = _pair_steps(starts, finishes, errors)
    completed_steps = len(pairs)
    steps_incomplete = len(starts) - completed_steps
    finish_without_start = len(finishes) - completed_steps

    step_records = []
    for start_record, fin in pairs:
        step_records.append({
            "index": 0,
            "start_order": start_record["order"],
            "start_id": (start_record["id"]
                         if isinstance(start_record["id"], str) else None),
            "finish_order": fin["order"],
            "finish_id": fin["id"] if isinstance(fin["id"], str) else None,
            "completed": True,
            "reason": fin["reason"],
            "tokens": fin["tokens"],
            "cost": fin["cost"],
            "tool_ids": [],
        })
    paired_start_orders = {record["start_order"] for record in step_records}
    for start_record in starts:
        if start_record["order"] in paired_start_orders:
            continue
        step_records.append({
            "index": 0,
            "start_order": start_record["order"],
            "start_id": (start_record["id"]
                         if isinstance(start_record["id"], str) else None),
            "finish_order": None,
            "finish_id": None,
            "completed": False,
            "reason": None,
            "tokens": None,
            "cost": None,
            "tool_ids": [],
        })
    step_records.sort(key=lambda record: record["start_order"])
    for position, record in enumerate(step_records, start=1):
        record["index"] = position
    step_by_start_order = {record["start_order"]: record for record in step_records}

    orphan_finishes = [{
        "order": fin["order"],
        "id": fin["id"] if isinstance(fin["id"], str) else None,
        "reason": fin["reason"],
        "tokens": fin["tokens"],
        "cost": fin["cost"],
    } for fin in orphan_finish_records]

    # ---- group tools by resolved id ---------------------------------------
    tool_groups = []
    group_pos = {}
    for rec in tool_events:
        key = rec["id"] if rec["id"] is not None else ("__anon__", rec["order"])
        ev_name = _resolve_key(rec["part"], ("tool", "name", "tool_name"))
        if key in group_pos:
            group = tool_groups[group_pos[key]]
            group["events"].append(rec)
            tool_repeat_updates += 1
            if (isinstance(ev_name, str) and isinstance(group["name"], str)
                    and ev_name != group["name"]):
                errors.append(
                    f"line {rec['order']}: conflicting tool name for id "
                    f"{rec['id']!r}")
        else:
            name = ev_name
            if name is not None and not (isinstance(name, str) and name):
                errors.append(f"line {rec['order']}: invalid tool name {name!r}")
                name = None
            if name is None:
                errors.append(
                    f"line {rec['order']}: tool event missing recognizable name")
            group_pos[key] = len(tool_groups)
            tool_groups.append({"name": name, "events": [rec]})

    # ---- finalize tool records --------------------------------------------
    tools_out = []
    tool_completions = 0
    tool_failures = 0
    tool_starts_without_finish = 0
    for group in tool_groups:
        events = group["events"]
        first = events[0]
        last = events[-1]

        statuses = []
        for ev in events:
            state = ev["part"].get("state")
            state = state if isinstance(state, dict) else {}
            status = state.get("status")
            if status is None:
                status = _resolve_key(ev["part"], ("status",))
            if status is not None and not isinstance(status, str):
                errors.append(f"line {ev['order']}: invalid tool status {status!r}")
                status = None
            statuses.append(status)
        terminal_seen = any(s in TOOL_TERMINAL_STATUSES for s in statuses)
        authoritative = statuses[-1] if statuses else None

        exit_code = None
        exit_source = None
        for path in _EXIT_CODE_PATHS:
            value, found = _dig(last["part"], path)
            if not found:
                continue
            if isinstance(value, bool) or not isinstance(value, int):
                if value is not None:
                    errors.append(
                        f"line {last['order']}: invalid tool exit code {value!r} "
                        f"at {'.'.join(path)}")
            else:
                exit_code = value
                exit_source = ".".join(path)
            break

        output_value = None
        state = last["part"].get("state")
        for source in ((state if isinstance(state, dict) else None), last["part"]):
            if isinstance(source, dict) and isinstance(source.get("output"), str):
                output_value = source.get("output")
                break

        # Attribution uses the step_start that *preceded the tool start* in
        # stream order, so a tool finishing after later step boundaries is
        # still attributed to the step where it began.
        attributed_start = None
        for start_record in starts:
            if start_record["order"] <= first["order"]:
                attributed_start = start_record
            else:
                break

        if terminal_seen:
            tool_completions += 1
            matched = True
        else:
            matched = False
            tool_starts_without_finish += 1
            label = first["id"] if first["id"] is not None else (
                f"line {first['order']} (no id)")
            errors.append(f"tool start without finish: {label}")

        failed = None
        failure_basis = None
        if authoritative in TOOL_FAILURE_STATUSES:
            failed, failure_basis = True, f"status={authoritative!r}"
        elif exit_code is not None and exit_code != 0:
            failed, failure_basis = True, f"exit_code={exit_code}"
        elif terminal_seen:
            failed, failure_basis = False, "terminal status with exit code 0 or no exit code"

        step_target = None
        if attributed_start is not None:
            step_target = step_by_start_order.get(attributed_start["order"])
        if step_target is not None:
            attributed_index = step_target["index"]
            if first["id"] is not None:
                step_target["tool_ids"].append(first["id"])
        else:
            attributed_index = None

        tools_out.append({
            "id": first["id"],
            "name": group["name"],
            "status": authoritative,
            "statuses_seen": statuses,
            "exit_code": exit_code,
            "exit_code_source": exit_source,
            "failed": failed,
            "failure_basis": failure_basis,
            "matched_finish": matched,
            "repeat_updates": len(events) - 1,
            "attributed_step_index": attributed_index,
            "attributed_step_start_id": (step_target["start_id"]
                                         if step_target is not None else None),
            "output_excerpt": (_excerpt(output_value, TOOL_OUTPUT_EXCERPT_CHARS)
                               if output_value is not None else None),
            "raw_excerpt": _excerpt(first["part"], TOOL_RAW_EXCERPT_CHARS),
        })
        if failed is True:
            tool_failures += 1

    # ---- gated aggregates over all unique finishes -------------------------
    totals = _empty_tokens()
    for field in totals:
        values = [fin["tokens"][field] for fin in finishes]
        totals[field] = (sum(values)
                         if values and all(v is not None for v in values) else None)
    costs = [fin["cost"] for fin in finishes]
    cost_total = (sum(costs)
                  if costs and all(c is not None for c in costs) else None)

    reasons = [fin["reason"] for fin in finishes]
    final_reason = reasons[-1] if reasons else None
    final_stop = final_reason == "stop"

    if texts and not finishes:
        errors.append("missing completion event")
    if finishes and final_reason not in TERMINAL_REASONS:
        errors.append("missing final stop event")

    # ---- derived (hypotheses, with basis; never mixed into observations) ---
    if expected_max_steps is None:
        derived = {
            "reached_step_limit": None,
            "basis": "no configured max step count supplied; cannot infer "
                     "from stream shape alone",
        }
    elif final_stop:
        derived = {
            "reached_step_limit": False,
            "basis": f"final step_finish reason is 'stop' after "
                     f"{completed_steps} completed step(s)",
        }
    elif completed_steps == expected_max_steps:
        derived = {
            "reached_step_limit": True,
            "basis": f"completed_steps == configured max ({expected_max_steps}) "
                     f"and final reason was {final_reason!r}",
        }
    elif completed_steps > expected_max_steps:
        derived = {
            "reached_step_limit": False,
            "basis": f"completed_steps {completed_steps} exceeds configured max "
                     f"{expected_max_steps}; configuration or accounting mismatch",
        }
    else:
        derived = {
            "reached_step_limit": False,
            "basis": f"completed_steps {completed_steps} below configured max "
                     f"{expected_max_steps}; stream ended early",
        }

    counts = {
        "lines_total": lines_total,
        "lines_invalid_json": lines_invalid_json,
        "lines_non_object": lines_non_object,
        "model_requests_started": len(starts),
        "model_requests_finished": len(finishes),
        "completed_steps": completed_steps,
        "steps_incomplete": steps_incomplete,
        "finish_without_start": finish_without_start,
        "duplicate_start_events": duplicate_start_events,
        "duplicate_finish_events": duplicate_finish_events,
        "tool_invocations": len(tool_groups),
        "tool_completions": tool_completions,
        "tool_failures": tool_failures,
        "tool_starts_without_finish": tool_starts_without_finish,
        "tool_repeat_updates": tool_repeat_updates,
        "anon_tool_events": anon_tool_events,
        "max_tools_in_one_step": max((len(r["tool_ids"]) for r in step_records),
                                     default=0),
        "unrecognized_event_types": dict(sorted(unrecognized_types.items())),
    }
    identity = {
        "provider": PROVIDER,
        "task_id": task_id,
        "run_id": run_id,
        "session_id_expected": expected_session_id,
        "session_ids_observed": list(session_ids_observed),
        "session_mismatch_events": session_mismatch_events,
        "events_without_session_id": events_without_session_id,
        "events_with_invalid_session_id": events_with_invalid_session_id,
    }

    return {
        "identity": identity,
        "counts": counts,
        "steps": step_records,
        "tools": tools_out,
        "orphan_finishes": orphan_finishes,
        "texts": texts,
        "answer": "".join(item["text"] for item in texts),
        "finishes": reasons,
        "truncated": truncation_seen,
        "final_stop": final_stop,
        "tokens": totals,
        "cost_observed": cost_total,
        "cost_source": "opencode.step_finish.cost" if cost_total is not None else None,
        "derived": derived,
        "errors": errors,
    }


def parse_events(stdout, provider=PROVIDER, *, expected_session_id=None,
                 expected_max_steps=None, task_id=None, run_id=None):
    """Parse an OpenCode JSONL event stream into an accounting dictionary.

    Notes for callers (packet rules, restated):
    - A zero process exit code is NOT success by itself.  Independently
```
