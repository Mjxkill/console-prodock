Facts : les helpers sont purement numériques ; `rgb565` fixe à `[0,255]` avant quantification, `rgb888` décode par divisions entières, et `periodic` moyenne explicitement les bords opposés. Le code 0xffff est une valeur `uint16` valide, pas un cas signé.

Hypothèse : l'implémentation actuelle n'est pas exactement réversible pour tous les codes, avec un premier écart attendu sur le canal bleu à `0x0013` (`19 -> 18`), sans inventer de seuil artistique ni de rendu.

Éléments manquants : captures PNG séparées, budget atlas GPU exact, et convention de décodage 5/6 bits à confirmer; ce script ne valide que les propriétés mathématiques supplied.

Test exécutable minimal (≤ 45 lignes) :

```python
import math, numpy as np

def rgb565(a):
    a = np.clip(np.rint(a), 0, 255).astype(np.uint32)
    return (((a[..., 0] * 31 + 127) // 255 << 11) | ((a[..., 1] * 63 + 127) // 255 << 5) | ((a[..., 2] * 31 + 127) // 255)).astype('<u2')

def rgb888(a):
    a = a.astype(np.uint32)
    return np.stack(((a >> 11) * 255 // 31, ((a >> 5) & 63) * 255 // 63, (a & 31) * 255 // 31), axis=-1).astype('uint8')

def periodic(a):
    h, w, _ = a.shape
    boundary = np.zeros_like(a)
    boundary[0] += a[-1] - a[0]; boundary[-1] -= a[-1] - a[0]
    boundary[:, 0] += a[:, -1] - a[:, 0]; boundary[:, -1] -= a[:, -1] - a[:, 0]
    yy, xx = np.meshgrid(np.arange(h), np.arange(w), indexing='ij')
    denominator = 2 * np.cos(2 * math.pi * xx / w) + 2 * np.cos(2 * math.pi * yy / h) - 4
    denominator[0, 0] = 1
    smooth = np.fft.fft2(boundary, axes=(0, 1)) / denominator[..., None]
    smooth[0, 0] = 0
    result = a - np.fft.ifft2(smooth, axes=(0, 1)).real
    result[0] = result[-1] = (result[0] + result[-1]) / 2
    result[:, 0] = result[:, -1] = (result[:, 0] + result[:, -1]) / 2
    return np.clip(result, 0, 255)

def main():
    codes = np.arange(65536, dtype=np.uint16)
    dec, enc = rgb888(codes), rgb565(rgb888(codes))
    bad = np.flatnonzero(enc != codes)
    if bad.size:
        i = int(bad[0])
        print(f"CONTRE-EXEMPLE code=0x{codes[i]:04x} dec={dec[i].tolist()} reenc=0x{enc[i]:04x}")
        raise SystemExit(1)
    assert int(rgb565(np.array([[255, 255, 255]], dtype=np.uint8))) == 0xffff
    assert rgb888(np.uint16(0xffff)).shape == (3,)
    img = np.full((8, 8, 3), 127.0); out = periodic(img)
    assert np.array_equal(out[0], out[-1]) and np.array_equal(out[:, 0], out[:, -1]) and np.allclose(out, img)
    neg = np.array([[-1, 0, 0]], dtype=np.int16)
    assert int(rgb565(neg).ravel()[0]) == 0 and int(rgb565(neg.astype(np.uint8)).ravel()[0]) != 0

if __name__ == "__main__":
    main()
```
