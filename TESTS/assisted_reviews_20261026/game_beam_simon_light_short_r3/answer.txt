Test C additionnel discriminant (relancer le même couleur et geler/reprendre le flare):

```c
#include "l3_simon_light.h"
#include <assert.h>
#include <string.h>

static unsigned C;
#define K(x) do{C++;assert(x);}while(0)

static L3SimonView V(int p,unsigned c,unsigned ms,unsigned seq,unsigned idx,unsigned fb)
{
    L3SimonView v={0};
    v.grid.kind=5; v.grid.status=L3_MINI_RUNNING; v.grid.width=2; v.grid.height=2;
    v.phase=p; v.active_color=c; v.phase_ms=ms; v.feedback_ms=fb;
    v.sequence_length=seq; v.input_index=idx;
    return v;
}

int main(void)
{
    L3SimonLight l; L3SimonLightFrame f,g; unsigned t;
    l3_simon_light_reset(&l);
    L3SimonView a=V(L3_SIMON_SHOW,0,100,1,0,0), b=V(L3_SIMON_SHOW,0,600,1,0,0);
    K(l3_simon_light_observe(&l,&a,&b,0,.1,1));
    t=l.triggers;
    K(l3_simon_light_observe(&l,&a,&b,0,.2,1));
    K(l.triggers>t);
    K(l3_simon_light_sample(&l,.25,&f));
    K(f.level_q8[0]==128);
    K(l3_simon_light_pause(&l,1,.25));
    K(l3_simon_light_sample(&l,5,&g));
    K(!memcmp(&f,&g,sizeof f));
    K(l3_simon_light_pause(&l,0,5));
    K(l3_simon_light_sample(&l,5.01,&g));
    K(g.level_q8[0]>128);
    printf("PASS same-color/pause checks=%u\n",C);
    return 0;
}
```
