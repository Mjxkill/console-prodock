   63     wire [28:0] scan_addr=scan_addresses[scan_rp];
   64     assign g_ready = rst_n && scan_count!=2;
   65     wire scan_enqueue=g_valid && g_ready;
   66     wire scan_issue=scan_valid && issue && m_ready;
   67     always @(posedge clk)begin
   68         if(!rst_n)begin scan_wp<=0;scan_rp<=0;scan_count<=0;end
   69         else begin
   70             if(scan_enqueue)begin scan_addresses[scan_wp]<=g_addr;scan_wp<=!scan_wp;end
   71             if(scan_issue)scan_rp<=!scan_rp;
   72             case({scan_enqueue,scan_issue})
   73                 2'b10:scan_count<=scan_count+1'b1;
   74                 2'b01:scan_count<=scan_count-1'b1;
   75                 default:;
   76             endcase
   77         end
   78     end
   79     // b_grant est gardé tel quel par la synthèse (syn_keep) : sans cela, la LUT de b_ready était partagée
   80     // avec m_wr_en, qui dépend de b_valid, et l'analyse de timing comptait une boucle combinatoire
   81     // b_valid -> arbitre -> b_ready -> blitter/V1 qui n'existe pas logiquement (r600, -0,47 ns).
   82     wire b_grant /* synthesis syn_keep=1 */ = issue && !scan_valid && take_blitter && m_ready && (!b_write || m_wr_ready);
   83     assign b_ready = b_grant;
   84     assign c_ready = rst_n && !cpu_pending;
   85     assign c_wr_ready = c_ready;
   86     wire cpu_fire = issue && !scan_valid && !take_blitter && cpu_pending && m_ready && (cpu_cmd != 0 || m_wr_ready);
   87     assign m_en = scan_issue || (b_valid && b_ready) || cpu_fire;
   88     assign m_cmd = scan_valid ? 3'b001 : take_blitter ? {2'b0,!b_write} : cpu_cmd;
   89     assign m_addr = scan_valid ? scan_addr : take_blitter ? b_addr : cpu_addr;
   90     assign m_wr_en = (b_valid && b_ready && b_write) || (cpu_fire && cpu_cmd == 0);
   91     assign m_wr_end = m_wr_en;
   92     assign m_wr_data = take_blitter ? b_wr_data : cpu_data;
   93     assign m_wr_mask = take_blitter ? b_wr_mask : cpu_mask;
   94     wire push = scan_issue || (b_valid && b_ready && !b_write) || (cpu_fire && cpu_cmd == 1);
   95     wire pop = m_rd_valid && !empty;
   96     wire [4:0] next_wp = wp + {4'b0, push};
   97     wire [4:0] next_rp = rp + {4'b0, pop};
   98     wire [4:0] incremented_wp = wp + 5'd1;
   99     wire one_slot_left = incremented_wp == {~rp[4],rp[3:0]};
  100     always @(posedge clk) begin
  101         if (!rst_n) begin
  102             wp <= 0; rp <= 0; full <= 0; prefer_blitter <= 1;
  103             take_blitter<=0;
  104             cpu_pending <= 0; cpu_cmd <= 0; cpu_addr <= 0; cpu_data <= 0; cpu_mask <= 32'hffffffff;
  105         end
  106         else begin
  107             // Cut the valid -> arbitration -> ready round trip. A stale grant
  108             // can only insert a bubble: native commands still require valid.
  109             // Acknowledgement remains the actual DDR command/data acceptance.
  110             if(b_valid && b_ready) take_blitter<=0;
  111             else take_blitter<=b_valid && (prefer_blitter || !cpu_pending);
  112             // While empty, sample the payload independently of the late
  113             // upstream request. The acceptance cycle captures the same words;
  114             // once pending, every word remains held until DDR accepts it.
  115             // This leaves c_en driving only the valid bit, not 320 payload CEs.
  116             if (!cpu_pending) begin
  117                 cpu_cmd <= c_cmd; cpu_addr <= c_addr;
  118                 cpu_data <= c_wr_data; cpu_mask <= c_wr_mask;
  119             end
  120             if (c_en && c_ready) cpu_pending <= 1;
  121             else if (cpu_fire) cpu_pending <= 0;
  122             if (!scan_valid) begin
  123                 if (b_valid && b_ready) prefer_blitter <= 0;
  124                 else if (cpu_fire) prefer_blitter <= 1;
  125             end
  126             if (push) tags[wp[3:0]] <= scan_valid ? 2'd1 : take_blitter ? 2'd2 : 2'd0;
  127             wp <= next_wp; rp <= next_rp;
  128             // Compare stable pointers before the late command acceptance.
  129             // Simultaneous push/pop leaves occupancy unchanged.
  130             if(push && !pop)full<=one_slot_left;
  131             else if(pop && !push)full<=0;
  132         end
  133     end
  134     assign c_rd_data = m_rd_data;
  135     assign g_data = m_rd_data;
  136     assign c_rd_valid = pop && tags[rp[3:0]] == 0;
  137     assign c_rd_end = c_rd_valid && m_rd_end;
  138     assign g_data_valid = pop && tags[rp[3:0]] == 1;
  139     assign b_rd_data = m_rd_data;
  140     assign b_rd_valid = pop && tags[rp[3:0]] == 2;
  141 endmodule
