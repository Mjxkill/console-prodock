 * Two independent CPU shadows describe the last bytes uploaded to each page. */
#define MINI_BASE (22U<<20)
#define MINI_PAGE (1U<<20)
enum { INK=64,SHADOW,PANEL,EDGE,EDGE_LIGHT,WELL,SLOT,SLOT_LIGHT,MUTED,FAINT,
       TEXT,WHITE,BRASS_DARK,BRASS,GOLD,GOLD_LIGHT,GREEN_DARK,GREEN,GREEN_LIGHT,
       RED_DARK,RED,RED_LIGHT,BLUE_DARK,BLUE };
typedef struct {L3Render *r;uint8_t *pixels;int w,h,scale,wide;} Canvas;
static int unit(const Canvas *c,int n){return n*c->scale/2;}
static int min(int a,int b){return a<b?a:b;}
static int max(int a,int b){return a>b?a:b;}
static void box(Canvas *c,int x,int y,int w,int h,unsigned color){
 int x0=max(x,0),x1=min(x+w,c->w),y0=max(y,0),y1=min(y+h,c->h);
 if(x1<=x0||y1<=y0)return;
 for(int yy=y0;yy<y1;yy++)memset(c->pixels+(size_t)yy*c->w+x0,(int)color,(size_t)(x1-x0));
}
static void pixel(Canvas *c,int x,int y,unsigned color){if(x>=0&&x<c->w&&y>=0&&y<c->h)c->pixels[(size_t)y*c->w+x]=(uint8_t)color;}
static void round_box(Canvas *c,int x,int y,int w,int h,int radius,unsigned color){
 radius=min(radius,min(w,h)/2);if(radius<1){box(c,x,y,w,h,color);return;}
 for(int yy=0;yy<h;yy++){
  int dy=yy<radius?radius-yy-1:yy>=h-radius?yy-(h-radius):0,cut=0;
  if(dy)while(cut<radius && (radius-cut-1)*(radius-cut-1)+dy*dy>radius*radius)cut++;
  box(c,x+cut,y+yy,w-2*cut,1,color);
 }
}
static void circle(Canvas *c,int x,int y,int radius,unsigned color){
 for(int yy=-radius;yy<=radius;yy++){int xx=radius;while(xx>0&&xx*xx+yy*yy>radius*radius)xx--;box(c,x-xx,y+yy,2*xx+1,1,color);}
}
static void line(Canvas *c,int x0,int y0,int x1,int y1,int thick,unsigned color){
 int dx=abs(x1-x0),sx=x0<x1?1:-1,dy=-abs(y1-y0),sy=y0<y1?1:-1,e=dx+dy;
 for(;;){box(c,x0-thick/2,y0-thick/2,thick,thick,color);if(x0==x1&&y0==y1)break;int q=2*e;if(q>=dy){e+=dy;x0+=sx;}if(q<=dx){e+=dx;y0+=sy;}}
}
static void panel(Canvas *c,int x,int y,int w,int h){
 int r=unit(c,10);round_box(c,x,y+unit(c,4),w,h,r,SHADOW);round_box(c,x,y,w,h,r,EDGE);
 round_box(c,x+1,y+1,w-2,h-2,r,PANEL);box(c,x+r,y+1,w-2*r,1,EDGE_LIGHT);
}
static void glass_panel(Canvas *c,int x,int y,int w,int h){
 int radius=unit(c,10);
 for(int yy=0;yy<h;yy++){
  int dy=yy<radius?radius-yy-1:yy>=h-radius?yy-(h-radius):0,cut=0;
  if(dy)while(cut<radius&&(radius-cut-1)*(radius-cut-1)+dy*dy>radius*radius)cut++;
  for(int xx=cut;xx<w-cut;xx++)if(x+xx>=0&&x+xx<c->w&&y+yy>=0&&y+yy<c->h){
   uint8_t *p=c->pixels+(size_t)(y+yy)*c->w+x+xx;
   *p=(xx==cut||xx==w-cut-1||!yy||yy==h-1)?BRASS_DARK:(*p>=32&&*p<128?puzzle_glass_tint[*p-32]:PANEL);
  }
 }
}
static unsigned next_char(const unsigned char **p){
 unsigned c=*(*p)++;if(c>=0xc2&&c<0xe0&&**p>=0x80){c=((c&31)<<6)|(**p&63);(*p)++;}
 else if(c>=0xe0&&c<0xf0&&(*p)[0]&&(*p)[1]){c=((c&15)<<12)|(((*p)[0]&63)<<6)|((*p)[1]&63);*p+=2;}
 if(c==0x2019)return '\'';if(c==0x2014||c==0x2013)return '-';return c<256?c:'?';
}
static int font_index(const Canvas *c,int f){return f+(c->wide?5:0);}
static int text_width(int f,const char *s){
 const unsigned char *p=(const unsigned char *)s;int w=0;while(*p)w+=puzzle_fonts[f].glyph[next_char(&p)].advance;return w;
}
static void label(Canvas *c,int x,int y,int f,const char *s,unsigned color){
 const PuzzleFont *font=puzzle_fonts+f;const unsigned char *p=(const unsigned char *)s;
 while(*p){unsigned ch=next_char(&p);const PuzzleGlyph *g=font->glyph+ch;
  for(unsigned yy=0;yy<font->height;yy++)for(unsigned xx=0;xx<g->width;xx++){
   unsigned pos=yy*g->width+xx,a=(font->pixels[g->offset+pos/4]>>(2*(pos%4)))&3;
   if(a)pixel(c,x+(int)xx,y+(int)yy,a==3?color:a==2?(color==GOLD||color==GOLD_LIGHT?BRASS:MUTED):FAINT);
  }x+=g->advance;
 }
}
static void centered(Canvas *c,int x,int y,int w,int f,const char *s,unsigned color){label(c,x+(w-text_width(f,s))/2,y,f,s,color);}
static int wrapped(Canvas *c,int x,int y,int width,int f,const char *s,unsigned color){
 char row[256]={0};size_t n=0;int start=y;const char *p=s;
 while(*p){while(*p==' ')p++;if(!*p)break;const char *end=p;while(*end&&*end!=' '&&*end!='\n')end++;
  size_t z=(size_t)(end-p);if(z>180)z=180;char trial[512];snprintf(trial,sizeof(trial),"%s%s%.*s",row,n?" ":"",(int)z,p);
  if(n&&text_width(f,trial)>width){label(c,x,y,f,row,color);y+=(int)puzzle_fonts[f].height+unit(c,2);n=0;row[0]=0;}
  if(n&&n+1<sizeof(row))row[n++]=' ';size_t room=sizeof(row)-n-1;if(z>room)z=room;memcpy(row+n,p,z);n+=z;row[n]=0;
  p=end;if(*p=='\n'){label(c,x,y,f,row,color);y+=(int)puzzle_fonts[f].height+unit(c,2);n=0;row[0]=0;p++;}
 }
 if(n){label(c,x,y,f,row,color);y+=(int)puzzle_fonts[f].height;}return y-start;
}
int l3_ui_text(uint8_t *dst,unsigned width,unsigned height,int x,int y,unsigned style,const char *text,unsigned color,unsigned wrap_width){
 if(!dst||!text||!width||!height||width>16384||height>16384||style>2||color>255)return 0;
 Canvas c={NULL,dst,(int)width,(int)height,width>=1000?3:2,width>=1000};int f=font_index(&c,(int)style);
 if(wrap_width)return wrapped(&c,x,y,(int)(wrap_width>width?width:wrap_width),f,text,color);
 label(&c,x,y,f,text,color);return *text?(int)puzzle_fonts[f].height:0;
}

}

static void wire(Canvas *c,int x,int y,int s,const L3MiniCell *cell){

  int thick=pass==0?t*2+2:pass==1?t*2:max(1,t/2),off=pass==0?2:0;unsigned col=pass==0?SHADOW:pass==1?base:light;
  if(cell->value&1)line(c,x+mid,y+mid+off,x+mid,y+off,thick,col);

static int drop_distance(const L3MiniView *v){

  int fits=1;for(unsigned y=0;y<v->height;y++)for(unsigned x=0;x<v->width;x++)if(v->cells[y*20+x].flags&L3_MINI_FLAG_ACTIVE){

  }if(!fits)break;distance=d;
 }return found?distance:0;
}
