import json
import math


def _valid_number(value, label, errors):
    if value is None:
        return None
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        errors.append(f"invalid numeric field {label}: {value!r}")
        return None
    if not math.isfinite(value) or value < 0:
        errors.append(f"invalid numeric field {label}: {value!r}")
        return None
    return value


def _add(left, right):
    if right is None:
        return left
    return right if left is None else left + right


def _result(answer, errors, finishes, truncated, tokens, cost):
    return {
        "answer": answer,
        "errors": errors,
        "finishes": finishes,
        "truncated": truncated,
        "tokens": tokens,
        "cost": cost,
    }


def _empty_tokens():
    return {
        "input": None,
        "output": None,
        "reasoning": None,
        "total": None,
        "cache_read": None,
        "cache_write": None,
    }


def _parse_opencode(stdout, errors):
    texts = []
    tokens = _empty_tokens()
    cost = None
    finishes = []
    seen_ids = []
    truncated = False

    for raw in (stdout or "").splitlines():
        line = raw.strip()
        if not line:
            continue
        try:
            event = json.loads(line)
        except (json.JSONDecodeError, ValueError):
            errors.append(f"invalid JSON: {raw.strip()}")
            continue
        if not isinstance(event, dict):
            errors.append(f"invalid JSON event: {raw.strip()}")
            continue

        event_type = event.get("type")
        if event_type == "text":
            part = event.get("part")
            if isinstance(part, dict) and isinstance(part.get("text"), str):
                texts.append(part["text"])

        elif event_type == "error":
            errors.append(event.get("error", event))

        elif event_type == "step_finish":
            part = event.get("part")
            part = part if isinstance(part, dict) else {}
            finish_id = part.get("id")
            if finish_id is not None and finish_id in seen_ids:
                continue
            if finish_id is not None:
                seen_ids.append(finish_id)

            reason = part.get("reason")
            if reason is None:
                reason = event.get("reason")
            finishes.append(reason)
            if reason == "length":
                truncated = True

            token_data = part.get("tokens")
            if not isinstance(token_data, dict):
                token_data = event.get("tokens")
            token_data = token_data if isinstance(token_data, dict) else {}

            tokens["total"] = _add(
                tokens["total"],
                _valid_number(token_data.get("total"), "tokens.total", errors),
            )
            tokens["input"] = _add(
                tokens["input"],
                _valid_number(token_data.get("input"), "tokens.input", errors),
            )
            tokens["output"] = _add(
                tokens["output"],
                _valid_number(token_data.get("output"), "tokens.output", errors),
            )
            tokens["reasoning"] = _add(
                tokens["reasoning"],
                _valid_number(token_data.get("reasoning"), "tokens.reasoning", errors),
            )

            cache = token_data.get("cache")
            cache = cache if isinstance(cache, dict) else {}
            cache_read = cache.get("read")
            if cache_read is None:
                cache_read = token_data.get("cache_read")
            if cache_read is None:
                cache_read = token_data.get("cacheRead")
            cache_write = cache.get("write")
            if cache_write is None:
                cache_write = token_data.get("cache_write")
            if cache_write is None:
                cache_write = token_data.get("cacheWrite")

            tokens["cache_read"] = _add(
                tokens["cache_read"],
                _valid_number(cache_read, "tokens.cache.read", errors),
            )
            tokens["cache_write"] = _add(
                tokens["cache_write"],
                _valid_number(cache_write, "tokens.cache.write", errors),
            )

            cost_value = part.get("cost")
            if cost_value is None:
                cost_value = event.get("cost")
            cost = _add(
                cost,
                _valid_number(cost_value, "cost", errors),
            )

    return _result("".join(texts), errors, finishes, truncated, tokens, cost)


def _parse_claude(stdout, errors):
    tokens = _empty_tokens()
    text = (stdout or "").strip()
    if not text:
        return _result("", errors, [], False, tokens, None)

    data = None
    try:
        data = json.loads(text)
    except (json.JSONDecodeError, ValueError):
        for raw in text.splitlines():
            line = raw.strip()
            if not line:
                continue
            try:
                candidate = json.loads(line)
            except (json.JSONDecodeError, ValueError):
                continue
            if isinstance(candidate, dict) and candidate.get("type") == "result":
                data = candidate
                break
        if data is None:
            errors.append(f"invalid JSON: {text[:120]}")
            return _result("", errors, [], False, tokens, None)

    if not isinstance(data, dict):
        errors.append("invalid Claude result: expected object")
        return _result("", errors, [], False, tokens, None)
    if data.get("type") != "result":
        return _result("", errors, [], False, tokens, None)

    answer = data.get("result")
    answer = answer if isinstance(answer, str) else ""

    if data.get("is_error") is True or "error" in data:
        error_value = data.get("error")
        errors.append(
            error_value if error_value is not None else "Claude result reported an error"
        )

    usage = data.get("usage")
    if usage is None:
        pass
    elif not isinstance(usage, dict):
        errors.append(f"invalid usage object: {usage!r}")
    else:
        tokens["input"] = _valid_number(
            usage.get("input_tokens"), "usage.input_tokens", errors
        )
        tokens["output"] = _valid_number(
            usage.get("output_tokens"), "usage.output_tokens", errors
        )
        tokens["cache_read"] = _valid_number(
            usage.get("cache_read_input_tokens"),
            "usage.cache_read_input_tokens",
            errors,
        )
        tokens["cache_write"] = _valid_number(
            usage.get("cache_creation_input_tokens"),
            "usage.cache_creation_input_tokens",
            errors,
        )

    cost = _valid_number(data.get("total_cost_usd"), "total_cost_usd", errors)
    return _result(answer, errors, [], False, tokens, cost)


def parse_events(stdout, provider):
    errors = []
    if "claude" in (provider or "").lower():
        return _parse_claude(stdout, errors)
    return _parse_opencode(stdout, errors)


def test_duplicate_finish_id_not_double_counted():
    finish_line = (
        '{"type":"step_finish","part":{"id":"s1","reason":"stop",'
        '"tokens":{"total":7,"input":3,"output":4,"reasoning":0,'
        '"cache":{"read":1,"write":2}},"cost":0.25}}'
    )
    stdout = finish_line + "\n" + finish_line + "\n"
    result = parse_events(stdout, "opencode")
    assert result["errors"] == []
    assert result["finishes"] == ["stop"]
    assert result["truncated"] is False
    assert result["tokens"]["total"] == 7
    assert result["tokens"]["input"] == 3
    assert result["cost"] == 0.25


def test_length_finish_with_answer_is_truncated():
    stdout = (
        '{"type":"text","part":{"text":"hello"}}\n'
        '{"type":"step_finish","part":{"id":"a","reason":"length"}}\n'
    )
    result = parse_events(stdout, "opencode")
    assert result["answer"] == "hello"
    assert result["truncated"] is True
    assert result["finishes"] == ["length"]
    assert all(value is None for value in result["tokens"].values())


def test_claude_missing_usage_stays_none():
    stdout = '{"type":"result","result":"done","is_error":false}'
    result = parse_events(stdout, "claude")
    assert result["answer"] == "done"
    assert result["errors"] == []
    assert result["finishes"] == []
    assert result["truncated"] is False
    assert all(value is None for value in result["tokens"].values())
    assert result["cost"] is None
