Faits : le code proposé compare uniquement `src` (canvas CPU) à `shadow` CPU, émet un intervalle par ligne via callback, met à jour le shadow uniquement dans les zones émises, valide les bornes, supporte `force`.

```c
#include <stddef.h>
#include <stdint.h>
#include <string.h>

typedef struct { size_t scanned, published, spans; } l3_ui_diff_stats_t;

int l3_ui_publish_diff(const uint32_t *src, uint32_t *shadow,
                       void (*emit)(size_t, const void *, size_t),
                       size_t width_words, size_t height_rows, int force,
                       l3_ui_diff_stats_t *st) {
    if (!src || !shadow || !emit || width_words == 0 || height_rows == 0) return 0;
    if (height_rows > (size_t)-1 / width_words) return 0;
    size_t w = width_words, rows = height_rows, total_words = w * rows;
    if (total_words * sizeof *src > (1u << 20)) return 0;

    size_t scanned = 0, published = 0, spans = 0;
    if (force) {
        size_t bytes = total_words * sizeof *src;
        emit(0, src, bytes);
        memcpy(shadow, src, bytes);
        scanned = published = bytes; spans = 1;
    } else {
        for (size_t r = 0; r < rows; r++) {
            size_t base = r * w, first = w, last = 0; int any = 0;
            for (size_t c = 0; c < w; c++) {
                scanned += sizeof *src;
                if (src[base + c] != shadow[base + c]) {
                    if (!any) { first = c; any = 1; }
                    last = c;
                }
            }
            if (!any) continue;
            size_t off = (base + first) * sizeof *src;
            size_t len = (last - first + 1) * sizeof *src;
            emit(off, src + base + first, len);
            for (size_t c = first; c <= last; c++) shadow[base + c] = src[base + c];
            published += len; spans++;
        }
    }
    if (st) { st->scanned = scanned; st->published = published; st->spans = spans; }
    return 1;
}
```

Contre-exemple minimal : page 0 force-full avec canvas tout nul émet bien offset 0 sur `w*h` octets malgré zéros, puis une révision ne changeant qu'une ligne n'émet qu'un seul span sur cette ligne ; l’invalidation repasse `force=1` et réémet la frame complète. Le chronomètre global reste autour de l’appel complet, pas dans le callback. Hypothèse à vérifier : `emit` inclut déjà l’offset de page (`20MiB` ou `21MiB`). Preuve manquante : types réels de `f->ui`, alignement exact et constance de largeur multiple de 4 au resize.
