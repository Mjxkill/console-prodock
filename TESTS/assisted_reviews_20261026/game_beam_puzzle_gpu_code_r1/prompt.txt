55: static int triangle_planes(const gpu_v1_context *c,const gpu_vertex v[3],gpu_raster_plane p[7],unsigned *mask)
56: {
57:     for(unsigned i=0;i<3;++i)if(v[i].x<INT16_MIN || v[i].x>INT16_MAX || v[i].y<INT16_MIN || v[i].y>INT16_MAX ||

60:     int64_t area=(int64_t)(v[1].x-v[0].x)*(v[2].y-v[0].y)-(int64_t)(v[1].y-v[0].y)*(v[2].x-v[0].x);
61:     *mask=(c->flags&1 ? 3U:0U)|(c->flags&16 ? 4U:0U)|(c->flags&134 ? 8U:0U)|(c->flags&64 ? 112U:0U);
62:     if(!area){*mask=0;return 1;}
63:     for(unsigned k=0;k<7;++k)if(*mask&(1U<<k)) {
64:         int64_t f[3];

66:             int value=k==0?v[i].u:k==1?v[i].v:k==2?v[i].q:k==3?v[i].z:k==4?v[i].r:k==5?v[i].g:v[i].b;
67:             f[i]=k==2?value:(int64_t)value*(k<2 && (c->flags&16)?v[i].q:65536);
68:         }

70:         int64_t dy=((int64_t)(v[1].x-v[0].x)*(f[2]-f[0])-(int64_t)(v[2].x-v[0].x)*(f[1]-f[0]))/area;
