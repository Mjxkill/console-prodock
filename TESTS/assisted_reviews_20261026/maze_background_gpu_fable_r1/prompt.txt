116:                 if(reverse)begin
117:                     count<=cursor-l < (cursor[3:0]==0 ? 16:cursor[3:0]) ? cursor-l:(cursor[3:0]==0 ? 16:cursor[3:0]);
118:                     px<=cursor-l < (cursor[3:0]==0 ? 16:cursor[3:0]) ? l[15:0]:(cursor-1'b1)&16'hfff0;
119:                 end else begin
120:                     count<=r-cursor < 16-cursor[3:0] ? r-cursor:16-cursor[3:0];px<=cursor;
121:                 end
122:                 state<=ADDRESS;
123:             end
124:             ADDRESS:begin
125:                 source_address<=source_row+(($signed({16'd0,px})-source_l)<<1);
126:                 destination_address<=destination_row+{15'd0,px,1'b0};second<=0;state<=READ0;
127:             end
128:             READ0:begin addr<={source_address[30:5],3'b0};valid<=1;write<=0;state<=WAIT0;end
129:             WAIT0:begin
130:                 if(valid && ready)valid<=0;
131:                 if(rd_valid)begin first<=rd_data;state<=source_lane+count>16 ? READ1:ALIGN;end
132:             end
133:             READ1:begin addr<=addr+29'd8;valid<=1;state<=WAIT1;end
134:             WAIT1:begin
135:                 if(valid && ready)valid<=0;
136:                 if(rd_valid)begin second<=rd_data;state<=ALIGN;end
137:             end
138:             ALIGN:begin aligned<=shifted[255:0];state<=WRITE;end
139:             WRITE:begin
140:                 addr<={destination_address[30:5],3'b0};
141:                 for(integer p=0;p<16;p=p+1)wr_mask[p*2+:2]<=(p>=dst_lane && p<dst_lane+count) ? 2'b00:2'b11;
142:                 valid<=1;write<=1;state<=SENT;
143:             end
144:             SENT:if(ready)begin valid<=0;state<=NEXT;end
145:             NEXT:begin
146:                 if(reverse ? px>l:px+count<r)begin cursor<=reverse ? px:px+count;state<=CHUNK;end
147:                 else if(row_has_next)begin row_y<=reverse ? row_y-1'b1:row_y+1'b1;state<=ROW;end
148:                 else state<=FINISH;
149:             end
150:             FINISH:begin done<=1;state<=IDLE;end
151:             default:begin valid<=0;bad<=1;state<=FINISH;end
152:             endcase
153:         end
154:     end
155: endmodule
