FILE games/luma3d/candidates/adviser_compact_r1/compact_proposal.h
/* Isolated adviser-only proposal. Includes no public engine change. */
typedef struct L3CompactMergeState {
    uint8_t  board[16];
    uint32_t rng;
    uint32_t score;
    uint32_t moves;
    uint8_t  status;
} L3CompactMergeState;

static void compact_from_l3(const L3MiniState *s, L3CompactMergeState *c)
{
    unsigned x, y;

    for (y = 0; y < 4u; ++y)
        for (x = 0; x < 4u; ++x)
            c->board[y * 4u + x] = s->board[at(x, y)];

    c->rng    = s->meta[RNG];
    c->score  = s->meta[SCORE];
    c->moves  = s->meta[MOVES];
    c->status = (uint8_t)s->meta[ST];
}

static uint32_t compact_random32(L3CompactMergeState *s)
{
    uint32_t x = s->rng;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    s->rng = x;
    return x;
}

static uint32_t compact_add_sat(uint32_t a, uint32_t b)
{
    return UINT32_MAX - a < b ? UINT32_MAX : a + b;
}

static unsigned compact_index(unsigned dir, unsigned line, unsigned pos)
{
    /* Copie exacte de la sémantique de merge_index, mais en coordonnées board[4]. */
    if (dir == 0) return pos * 4u + line;          /* at(line,pos) */
    if (dir == 1) return line * 4u + (3u - pos);   /* at(3-pos,line) */
    if (dir == 2) return (3u - pos) * 4u + line;   /* at(line,3-pos) */
    return line * 4u + pos;                        /* at(pos,line) */
}

static int compact_slide(L3CompactMergeState *s, unsigned dir)
{
    unsigned line, pos;
    int changed = 0;

    for (line = 0; line < 4; ++line) {
        uint8_t packed[4] = {0, 0, 0, 0};
        uint8_t out[4] = {0, 0, 0, 0};
        unsigned n = 0, k = 0, i = 0;

        for (pos = 0; pos < 4; ++pos) {
            uint8_t v = s->board[compact_index(dir, line, pos)];
            if (v) packed[n++] = v;
        }
        while (i < n) {
            unsigned v = packed[i++];
            if (i < n && packed[i] == v) {
                ++i;
                ++v;
                s->score = compact_add_sat(s->score, 1u << v);
            }
            out[k++] = (uint8_t)v;
        }
        for (pos = 0; pos < 4; ++pos) {
            unsigned q = compact_index(dir, line, pos);
            if (s->board[q] != out[pos]) changed = 1;
            s->board[q] = out[pos];
        }
    }
    return changed;
}

static int compact_has_move(const L3CompactMergeState *s)
{
    unsigned x, y;

    for (y = 0; y < 4; ++y)
        for (x = 0; x < 4; ++x) {
            unsigned q = y * 4u + x;
            uint8_t v = s->board[q];
            if (!v) return 1;
            if (x < 3 && s->board[q + 1] == v) return 1;
            if (y < 3 && s->board[q + 4] == v) return 1;
        }
    return 0;
}

static void compact_spawn(L3CompactMergeState *s)
{
    unsigned freecells[16];
    unsigned n = 0, x, y, q;
    uint8_t v;

    for (y = 0; y < 4; ++y)
        for (x = 0; x < 4; ++x)
            if (!s->board[y * 4u + x])
                freecells[n++] = y * 4u + x;

    if (!n) return;

    q = freecells[compact_random32(s) % n];
    v = (uint8_t)((compact_random32(s) % 10u) == 0u ? 2u : 1u);
    s->board[q] = v;
}

static void compact_move(L3CompactMergeState *s, unsigned dir)
{
    unsigned i;

    if (!compact_slide(s, dir)) return;

    ++s->moves;
    compact_spawn(s);

    for (i = 0; i < 16; ++i)
        if (s->board[i] == 10) {
            s->status = L3_MINI_WON;
            return;
        }

    if (!compact_has_move(s)) s->status = L3_MINI_LOST;
}

static int compact_quality(const L3CompactMergeState *s)
{
    unsigned x, y;
    int empty = 0, smooth = 0, mono = 0, max = 0, corner = 0;

    if (s->status == L3_MINI_WON) return 10000000;
    if (s->status == L3_MINI_LOST) return -10000000;

    for (y = 0; y < 4; ++y)
        for (x = 0; x < 4; ++x) {
            uint8_t v = s->board[y * 4u + x];
            int a, b;

            if (!v) ++empty;
            if (v > max) max = v;
            if ((x == 0 || x == 3) && (y == 0 || y == 3) && v > corner)
                corner = v;

            if (x < 3) {
                a = s->board[y * 4u + x + 1];
                if (v && a) {
                    b = v - a;
                    smooth -= b < 0 ? -b : b;
                }
            }
            if (y < 3) {
                a = s->board[(y + 1) * 4u + x];
                if (v && a) {
                    b = v - a;
                    smooth -= b < 0 ? -b : b;
                }
            }
        }

    for (y = 0; y < 4; ++y) {
        int plus = 0, minus = 0;
        for (x = 0; x < 3; ++x) {
            int d = (int)s->board[y * 4u + x] - (int)s->board[y * 4u + x + 1];
            if (d > 0) plus += d;
            else minus -= d;
        }
        mono -= plus < minus ? plus : minus;
    }

    for (x = 0; x < 4; ++x) {
        int plus = 0, minus = 0;
        for (y = 0; y < 3; ++y) {
            int d = (int)s->board[y * 4u + x] - (int)s->board[(y + 1) * 4u + x];
            if (d > 0) plus += d;
            else minus -= d;
        }
        mono -= plus < minus ? plus : minus;
    }

    return 1200 * empty + 220 * mono + 35 * smooth + 700 * max + 600 * corner;
}

static int compact_search(const L3CompactMergeState *s, unsigned depth)
{
    unsigned d;
    int best = INT_MIN;

    if (!depth || s->status != L3_MINI_RUNNING)
        return compact_quality(s);

    for (d = 0; d < 4; ++d) {
        L3CompactMergeState t = *s;
        int value;

        compact_move(&t, d);
        if (t.moves == s->moves) continue;
        value = compact_search(&t, depth - 1u);
        if (value > best) best = value;
    }
    return best == INT_MIN ? compact_quality(s) : best;
}
