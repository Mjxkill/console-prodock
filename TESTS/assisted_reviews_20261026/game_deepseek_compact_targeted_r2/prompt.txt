/* A random monotone route is built first; its pipes are then rotated.
 * aux's bit4 marks the constructed witness path; low4 are its solved pipes. */
static void circuit_generate(L3MiniState *s)
{
    unsigned x = 0, y = 0, i;
    for (y = 0; y < 6; ++y) for (x = 0; x < 6; ++x) {
        static const uint8_t masks[3] = {3, 5, 7};
        s->aux[at(x,y)] = masks[random32(s) % 3u];
    }
    x = 0; y = 0; s->aux[0] = WEST | 16u;
    while (x != 5 || y != 5) {
        int right = y == 5 || (x != 5 && (random32(s) & 1u));
        unsigned old = at(x,y);
        if (right) { s->aux[old] |= EAST; ++x; s->aux[at(x,y)] = WEST | 16u; }
        else { s->aux[old] |= SOUTH; ++y; s->aux[at(x,y)] = NORTH | 16u; }
    }
    s->aux[at(5,5)] |= EAST;
    for (y = 0; y < 6; ++y) for (x = 0; x < 6; ++x) {
        uint8_t m = s->aux[at(x,y)] & 15u;
        unsigned turns = random32(s) % 4u;
        for (i = 0; i < turns; ++i) m = turn(m);
        s->board[at(x,y)] = m;
    }
    while (s->board[0] & WEST) s->board[0] = turn(s->board[0]);
}

            int xx = (int)x + dx[d], yy = (int)y + dy[d];
            unsigned q;

static void offsets(unsigned piece, unsigned rotation, int *xs, int *ys)
{
    unsigned i, j;
    int minx = 9, miny = 9;
    for (i = 0; i < 4; ++i) {
        int x = shapes[piece][2u*i], y = shapes[piece][2u*i+1u];
        for (j = 0; j < rotation; ++j) { int t = x; x = -y; y = t; }
        xs[i] = x; ys[i] = y;
        if (x < minx) minx = x;
        if (y < miny) miny = y;
    }
    for (i = 0; i < 4; ++i) { xs[i] -= minx; ys[i] -= miny; }
}

    } else if(kind==L3_MINI_BLOCKS) {
        if(d==1 || d==3)try_move(s,d);
        if((pressed&L3_MINI_ROTATE) || d==0)try_rotate(s);
        if(pressed&L3_MINI_ACT) {
            while(current_fits(s,(int)s->data[CX],(int)s->data[CY]+1,s->data[ROT]))++s->data[CY];
            blocks_lock(s);
        } else if(d==2)blocks_down(s);
    } else if(kind==L3_MINI_SHAPES) {
        if(d>=0)try_move(s,d);
        if(pressed&L3_MINI_ROTATE)try_rotate(s);
        if(pressed&L3_MINI_ACT)assembly_lock(s);
    } else if(kind==L3_MINI_MERGE1024) {
        if(d>=0)merge_move(s,(unsigned)d);
    } else {
        if(d<0)d=direction(held);
        if(d>=0)s->data[NEXTDIR]=(unsigned)d;
    }
}

uint32_t l3_mini_step(L3MiniState *s,uint32_t pressed,uint32_t held,uint32_t dt_ms)
{
    uint32_t oldstatus,phase,n,i;
    if(!l3_mini_validate(s) || ((pressed|held)&~L3_MINI_INPUT_MASK))return L3_MINI_EVENT_INVALID;
    if(pressed&L3_MINI_BACK)return L3_MINI_EVENT_EXIT;
    if(s->meta[ST]!=L3_MINI_RUNNING)return 0;
    oldstatus=s->meta[ST];
    input(s,pressed,held);
    if(dt_ms>L3_MINI_MAX_DT_MS)dt_ms=L3_MINI_MAX_DT_MS;
    phase=s->meta[PHASE]+60u*dt_ms;n=phase/1000u;
    if(s->meta[ST]==L3_MINI_RUNNING)s->meta[PHASE]=phase%1000u;
    for(i=0;i<n && s->meta[ST]==L3_MINI_RUNNING;++i) {
        s->meta[TICKS]=add_sat(s->meta[TICKS],1);
        if(s->meta[K]==L3_MINI_BLOCKS) {
            unsigned period=(held&L3_MINI_DOWN)?3u:30u;
            if(++s->data[TIMER]>=period){s->data[TIMER]=0;blocks_down(s);}
        } else if(s->meta[K]==L3_MINI_MAZE)maze_tick(s);
    }
    if(s->meta[ST]!=oldstatus)return L3_MINI_EVENT_CHANGED|(s->meta[ST]==L3_MINI_WON?L3_MINI_EVENT_WON:L3_MINI_EVENT_LOST);
    return (pressed || held || dt_ms)?L3_MINI_EVENT_CHANGED:0;
}

/* Validation checks canonical padding, dimensions, ranges and game invariants.
 * Saved data is not trusted merely because its checksum matches. */
int l3_mini_validate(const L3MiniState *s)
{
    unsigned i,x,y,kind,status;L3MiniState initial;
    if(!s || s->meta[V]!=L3_MINI_VERSION || s->meta[K]>=5 || s->meta[ST]>L3_MINI_LOST || !s->meta[RNG] || s->meta[PHASE]>=1000 || s->meta[PHASE]%20u)return 0;
    kind=s->meta[K];status=s->meta[ST];
    for(i=12;i<16;++i)if(s->meta[i])return 0;
    if(status!=L3_MINI_RUNNING && s->meta[PHASE])return 0;
    l3_mini_init(&initial,(L3MiniKind)kind,s->meta[SEED]);
    if(s->meta[W]!=initial.meta[W] || s->meta[H]!=initial.meta[H] || s->meta[GOAL]!=initial.meta[GOAL])return 0;
    for(y=0;y<20;++y)for(x=0;x<20;++x) {
        i=at(x,y);
        if((x>=s->meta[W] || y>=s->meta[H]) && (s->board[i] || s->aux[i]))return 0;
    }
    if(kind!=L3_MINI_CIRCUIT && kind!=L3_MINI_SHAPES)for(i=0;i<400;++i)if(s->aux[i])return 0;
    if(kind==L3_MINI_CIRCUIT) {
        int won;
        if(s->data[CX]>=6 || s->data[CY]>=6 || s->data[LIVES]>120 || s->meta[MOVES]+s->data[LIVES]!=120)return 0;
        if(memcmp(s->aux,initial.aux,400)!=0 || s->meta[RNG]!=initial.meta[RNG])return 0;
        for(i=0;i<32;++i)if(i!=CX && i!=CY && i!=LIVES && s->data[i])return 0;
        for(y=0;y<6;++y)for(x=0;x<6;++x) {
            uint8_t v=s->aux[at(x,y)]&15u;unsigned r;int same=0;
            for(r=0;r<4;++r){if(v==s->board[at(x,y)])same=1;v=turn(v);}if(!same)return 0;
        }
        won=circuit_won(s);
        if((status==L3_MINI_WON)!=won || (status==L3_MINI_LOST)!=(s->data[LIVES]==0 && !won))return 0;
        if(s->meta[SCORE]!=(won?s->data[LIVES]+1u:0u))return 0;
    } else if(kind==L3_MINI_BLOCKS) {
        unsigned occupied=0;
        if(s->data[PIECE]>=7 || s->data[ROT]>=4 || s->data[CX]>=8 || s->data[CY]>=14 || s->data[TIMER]>=30 || s->data[COUNT]>7 || s->data[NEXTDIR]>=7 || !s->data[AUXCOUNT])return 0;
        for(i=0;i<400;++i){if(s->board[i]>8)return 0;occupied+=s->board[i]!=0;}
        for(i=0;i<32;++i)if(i!=CX && i!=CY && i!=ROT && i!=PIECE && i!=TIMER && i!=COUNT && i!=NEXTDIR && i!=AUXCOUNT && s->data[i])return 0;
        if(s->meta[SCORE]%100u || (status==L3_MINI_WON)!=(s->data[COUNT]>=4))return 0;
        if(s->data[NEXTDIR]!=initial.data[NEXTDIR] || s->meta[MOVES]>28 ||
           occupied+8u*s->data[COUNT]!=24u+4u*s->meta[MOVES] ||
           s->data[AUXCOUNT]!=s->meta[MOVES]+(status==L3_MINI_WON?0u:1u) ||
           s->meta[SCORE]<100u*s->data[COUNT] || s->meta[SCORE]>400u*s->data[COUNT])return 0;
        if(status==L3_MINI_RUNNING && !current_fits(s,(int)s->data[CX],(int)s->data[CY],s->data[ROT]))return 0;
        if(status==L3_MINI_LOST && (s->data[CX]!=3 || s->data[CY]!=0 || s->data[ROT]!=0 || current_fits(s,3,0,0)))return 0;
        for(y=0;y<14;++y){unsigned full=0;for(x=0;x<8;++x)full+=s->board[at(x,y)]!=0;if(full==8)return 0;}
    } else if(kind==L3_MINI_SHAPES) {
        unsigned active;
        if(s->data[COUNT]>4 || s->data[LIVES]>8 || s->data[ROT]>=4 || s->data[CX]>=8 || s->data[CY]>=8)return 0;
        active=s->data[COUNT]<4?s->data[COUNT]:3;
        if(s->data[PIECE]!=assembly_types[active] || !current_fits(s,(int)s->data[CX],(int)s->data[CY],s->data[ROT]))return 0;
        if(memcmp(s->aux,initial.aux,400)!=0 || s->meta[SCORE]!=s->data[COUNT]*100u || s->meta[MOVES]!=s->data[COUNT]+8u-s->data[LIVES])return 0;
        for(i=0;i<400;++i)if(s->board[i]!=(s->aux[i] && s->aux[i]<=s->data[COUNT]?s->aux[i]:0))return 0;
        for(i=0;i<32;++i)if(i!=CX && i!=CY && i!=ROT && i!=PIECE && i!=COUNT && i!=LIVES && s->data[i])return 0;
        if((status==L3_MINI_WON)!=(s->data[COUNT]==4) || (status==L3_MINI_LOST)!=(s->data[LIVES]==0))return 0;
    } else if(kind==L3_MINI_MERGE1024) {
        uint32_t sum=0;int won=0;
        for(i=0;i<400;++i){unsigned v=s->board[i];if(v>10)return 0;if(v)sum+=1u<<v;if(v==10)won=1;}
        if(s->data[0]!=s->meta[MOVES]+2u || s->data[1]>s->data[0] || (uint64_t)sum!=2ull*s->data[0]+2ull*s->data[1] || s->meta[SCORE]%4u)return 0;
        for(i=2;i<32;++i)if(s->data[i])return 0;
        if((status==L3_MINI_WON)!=won || (status==L3_MINI_LOST)!=(!won && !merge_has_move(s)))return 0;
    } else {
        unsigned remaining=0,power_eaten=0,pellet_eaten=0;
        if(s->data[CX]>=13 || s->data[CY]>=11 || s->data[DIR]>4 || s->data[NEXTDIR]>4 || s->data[TIMER]>=6 || s->data[ETIMER]>=14 || s->data[LIVES]>3 || s->data[POWER]>300 || s->data[AUXCOUNT]>120)return 0;
        if(!maze_open(s,(int)s->data[CX],(int)s->data[CY]))return 0;
        for(i=EX0;i<=EX1;i+=2)if(s->data[i]>=13 || s->data[i+1]>=11 || !maze_open(s,(int)s->data[i],(int)s->data[i+1]))return 0;
        for(y=0;y<11;++y)for(x=0;x<13;++x) {
            unsigned q=at(x,y),v=s->board[q];char c=maze[y][x];
            if(c=='#'){if(v!=1)return 0;}
            else {if(v!=0 && v!=(c=='o'?3u:2u))return 0;if(v)++remaining;else if(c=='o')++power_eaten;else ++pellet_eaten;}
        }
        if(remaining+s->data[COUNT]!=s->meta[GOAL] || s->meta[SCORE]!=50u*power_eaten+10u*pellet_eaten)return 0;
        for(i=16;i<32;++i)if(s->data[i])return 0;
        if(s->data[ROT] || s->data[PIECE] || s->meta[RNG]!=initial.meta[RNG])return 0;
        if((status==L3_MINI_WON)!=(remaining==0) || (status==L3_MINI_LOST)!=(s->data[LIVES]==0))return 0;
    }
    return 1;
}

int l3_mini_view(const L3MiniState *s,L3MiniView *v)
{
    unsigned x,y,kind;uint8_t seen[400];uint64_t elapsed;
    if(!v || !l3_mini_validate(s))return 0;
    memset(v,0,sizeof(*v));kind=s->meta[K];
    v->kind=kind;v->status=s->meta[ST];v->width=(uint16_t)s->meta[W];v->height=(uint16_t)s->meta[H];
    v->score=s->meta[SCORE];v->target=s->meta[GOAL];v->moves=s->meta[MOVES];
    elapsed=((uint64_t)s->meta[TICKS]*1000u+s->meta[PHASE])/60u;
    v->elapsed_ms=elapsed>UINT32_MAX?UINT32_MAX:(uint32_t)elapsed;
    v->title=titles[kind];v->help=helps[kind];v->objective=objectives[kind];
    v->cursor_x=-1;v->cursor_y=-1;
    if(kind==L3_MINI_CIRCUIT){circuit_reachable(s,seen);v->progress=s->meta[ST]==L3_MINI_WON;v->lives=(uint16_t)s->data[LIVES];}
    else if(kind==L3_MINI_BLOCKS || kind==L3_MINI_SHAPES || kind==L3_MINI_MAZE){v->progress=(uint16_t)s->data[COUNT];v->lives=(uint16_t)s->data[LIVES];}
    for(y=0;y<s->meta[H];++y)for(x=0;x<s->meta[W];++x) {
        unsigned q=at(x,y),b=s->board[q];L3MiniCell *c=&v->cells[q];c->color=1;
        if(kind==L3_MINI_CIRCUIT) {
            c->kind=L3_MINI_CELL_WIRE;c->value=(uint16_t)b;c->color=seen[q]?10:6;c->glyph='+';
            if(seen[q])c->flags|=L3_MINI_FLAG_CONNECTED;
            if(q==0)c->kind=L3_MINI_CELL_START;
            if(x==5 && y==5)c->kind=L3_MINI_CELL_FINISH;
        } else if(kind==L3_MINI_BLOCKS) {
            if(b){c->kind=L3_MINI_CELL_BLOCK;c->color=(uint8_t)(b==8?8:b+2u);c->glyph='#';}
        } else if(kind==L3_MINI_SHAPES) {
            if(s->aux[q]){c->kind=b?L3_MINI_CELL_BLOCK:L3_MINI_CELL_TARGET;c->color=(uint8_t)(s->aux[q]+9u);c->value=s->aux[q];c->glyph=b?'#':'.';}
        } else if(kind==L3_MINI_MERGE1024) {
            if(b){c->kind=L3_MINI_CELL_NUMBER;c->value=(uint16_t)(1u<<b);c->color=(uint8_t)(b+2u);c->glyph=(uint8_t)(b<10?'0'+b:'A');if(c->value>v->progress)v->progress=c->value;}
        } else {
            if(b==1){c->kind=L3_MINI_CELL_WALL;c->color=4;c->glyph='#';}
            if(b==2){c->kind=L3_MINI_CELL_PELLET;c->color=14;c->glyph='.';}
            if(b==3){c->kind=L3_MINI_CELL_POWER;c->color=11;c->glyph='O';}
        }
    }
    if(kind==L3_MINI_CIRCUIT) {
        v->cursor_x=(int16_t)s->data[CX];v->cursor_y=(int16_t)s->data[CY];
        v->cells[at(s->data[CX],s->data[CY])].flags|=L3_MINI_FLAG_CURSOR;
    } else if((kind==L3_MINI_BLOCKS || kind==L3_MINI_SHAPES) && s->meta[ST]==L3_MINI_RUNNING) {
        int xs[4],ys[4];unsigned i;
        offsets(s->data[PIECE],s->data[ROT],xs,ys);
        for(i=0;i<4;++i){L3MiniCell *c=&v->cells[at(s->data[CX]+(unsigned)xs[i],s->data[CY]+(unsigned)ys[i])];
            c->kind=L3_MINI_CELL_BLOCK;c->color=(uint8_t)(kind==L3_MINI_SHAPES?s->data[COUNT]+10u:s->data[PIECE]+3u);c->flags|=L3_MINI_FLAG_ACTIVE;c->glyph='@';}
        v->cursor_x=(int16_t)s->data[CX];v->cursor_y=(int16_t)s->data[CY];
    } else if(kind==L3_MINI_MAZE) {
        unsigned i;
        for(i=0;i<2;++i){L3MiniCell *c=&v->cells[at(s->data[EX0+2u*i],s->data[EY0+2u*i])];c->kind=L3_MINI_CELL_ENEMY;c->color=s->data[POWER]?6:(uint8_t)(12u+i);c->glyph='X';}
        {L3MiniCell *c=&v->cells[at(s->data[CX],s->data[CY])];c->kind=L3_MINI_CELL_PLAYER;c->color=14;c->glyph='@';if(s->data[POWER])c->flags|=L3_MINI_FLAG_ACTIVE;}
    }
    return 1;
}

static void put32(uint8_t *p,uint32_t v)
{ p[0]=(uint8_t)v;p[1]=(uint8_t)(v>>8);p[2]=(uint8_t)(v>>16);p[3]=(uint8_t)(v>>24); }
static uint32_t get32(const uint8_t *p)
{ return (uint32_t)p[0]|((uint32_t)p[1]<<8)|((uint32_t)p[2]<<16)|((uint32_t)p[3]<<24); }
static uint32_t checksum(const uint8_t *p)
{
    uint32_t h=2166136261u;unsigned i;
    for(i=0;i<L3_MINI_SAVE_BYTES;++i)if(i<12 || i>=16){h^=p[i];h*=16777619u;}
    return h;
}
size_t l3_mini_encode(const L3MiniState *s,void *output,size_t capacity)
{
    uint8_t bytes[L3_MINI_SAVE_BYTES];unsigned i;
    if(!l3_mini_validate(s))return 0;
    if(!output)return L3_MINI_SAVE_BYTES;
    if(capacity<L3_MINI_SAVE_BYTES)return 0;
    memcpy(bytes,"L3MG",4);put32(bytes+4,L3_MINI_VERSION);put32(bytes+8,992u);put32(bytes+12,0);
    for(i=0;i<16;++i)put32(bytes+16u+4u*i,s->meta[i]);
    for(i=0;i<32;++i)put32(bytes+80u+4u*i,s->data[i]);
    memcpy(bytes+208,s->board,400);memcpy(bytes+608,s->aux,400);
    put32(bytes+12,checksum(bytes));memcpy(output,bytes,sizeof(bytes));return sizeof(bytes);
}
int l3_mini_decode(L3MiniState *s,const void *input_bytes,size_t size)
{
    const uint8_t *p=(const uint8_t *)input_bytes;L3MiniState tmp;unsigned i;
    if(!s || !p || size!=L3_MINI_SAVE_BYTES || memcmp(p,"L3MG",4)!=0 || get32(p+4)!=L3_MINI_VERSION || get32(p+8)!=992u || get32(p+12)!=checksum(p))return 0;
    memset(&tmp,0,sizeof(tmp));
    for(i=0;i<16;++i)tmp.meta[i]=get32(p+16u+4u*i);
    for(i=0;i<32;++i)tmp.data[i]=get32(p+80u+4u*i);
    memcpy(tmp.board,p+208,400);memcpy(tmp.aux,p+608,400);
    if(!l3_mini_validate(&tmp))return 0;
    *s=tmp;return 1;
}

/* Adviser only: inspect a legal position and choose a normal next command. */
static int merge_quality(const L3MiniState *s)
{
    unsigned x,y;int empty=0,smooth=0,mono=0,max=0,corner=0;
    for(y=0;y<4;++y)for(x=0;x<4;++x) {
        int v=s->board[at(x,y)],a,b;
        if(!v)++empty;
        if(v>max)max=v;
        if(x==0 || x==3)if((y==0 || y==3) && v>corner)corner=v;
        if(x<3){a=s->board[at(x+1,y)];if(v && a){b=v-a;smooth-=b<0?-b:b;}}
        if(y<3){a=s->board[at(x,y+1)];if(v && a){b=v-a;smooth-=b<0?-b:b;}}
    }
    for(y=0;y<4;++y) {
        int plus=0,minus=0;
        for(x=0;x<3;++x){int d=(int)s->board[at(x,y)]-(int)s->board[at(x+1,y)];if(d>0)plus+=d;else minus-=d;}
        mono-=plus<minus?plus:minus;
    }
    for(x=0;x<4;++x) {
        int plus=0,minus=0;
        for(y=0;y<3;++y){int d=(int)s->board[at(x,y)]-(int)s->board[at(x,y+1)];if(d>0)plus+=d;else minus-=d;}
        mono-=plus<minus?plus:minus;
    }
    if(s->meta[ST]==L3_MINI_WON)return 10000000;
    if(s->meta[ST]==L3_MINI_LOST)return -10000000;
    return 1200*empty+220*mono+35*smooth+700*max+600*corner;
}
static int merge_search(const L3MiniState *s,unsigned depth)
{
    unsigned d;int best=INT_MIN;
    if(!depth || s->meta[ST]!=L3_MINI_RUNNING)return merge_quality(s);
    for(d=0;d<4;++d) {
        L3MiniState t=*s;int value;
        merge_move(&t,d);if(t.meta[MOVES]==s->meta[MOVES])continue;
        value=merge_search(&t,depth-1u);if(value>best)best=value;
    }
    return best==INT_MIN?merge_quality(s):best;
}
int l3_mini_demo_input(const L3MiniState *s,uint32_t *pressed,uint32_t *held,uint32_t *dt_ms)
{
    unsigned kind;uint32_t key=0,dt=0;
    if(!pressed || !held || !dt_ms || !l3_mini_validate(s) || s->meta[ST]!=L3_MINI_RUNNING)return 0;
    kind=s->meta[K];
    if(kind==L3_MINI_CIRCUIT) {
        unsigned x,y,tx=0,ty=0;int found=0;
        for(y=0;y<6 && !found;++y)for(x=0;x<6;++x) {
            unsigned q=at(x,y);
            if((s->aux[q]&16u) && s->board[q]!=(s->aux[q]&15u)){tx=x;ty=y;found=1;break;}
        }
        if(!found)return 0;
        if(s->data[CX]<tx)key=L3_MINI_RIGHT;else if(s->data[CX]>tx)key=L3_MINI_LEFT;
        else if(s->data[CY]<ty)key=L3_MINI_DOWN;else if(s->data[CY]>ty)key=L3_MINI_UP;
        else key=L3_MINI_ROTATE;
    } else if(kind==L3_MINI_BLOCKS) {
        int best=INT_MIN,bx=-1;unsigned br=0,r;int x;
        for(r=0;r<4;++r)for(x=0;x<8;++x) {
            L3MiniState t=*s;int y=(int)s->data[CY],height=0,holes=0,bump=0,prev=0,value;unsigned xx,n;
            if(!piece_fits(s,s->data[PIECE],r,x,y,1))continue;
            while(piece_fits(s,s->data[PIECE],r,x,y+1,1))++y;
            t.data[CX]=(unsigned)x;t.data[CY]=(unsigned)y;t.data[ROT]=r;
            n=t.data[COUNT];blocks_lock(&t);n=t.data[COUNT]-n;
            for(xx=0;xx<8;++xx){int yy,h=0,seen=0;for(yy=0;yy<14;++yy){if(t.board[at(xx,(unsigned)yy)]){if(!seen)h=14-yy;seen=1;}else if(seen)++holes;}height+=h;if(xx){int diff=h-prev;bump+=diff<0?-diff:diff;}prev=h;}
            value=(int)n*1000-holes*400-height*10-bump*20;
            if(t.meta[ST]==L3_MINI_WON)value+=100000;
            if(t.meta[ST]==L3_MINI_LOST)value-=100000;
            if(value>best){best=value;bx=x;br=r;}
        }
        if(bx<0)key=L3_MINI_ACT;
        else if(s->data[ROT]!=br)key=L3_MINI_ROTATE;
        else if(s->data[CX]<(unsigned)bx)key=L3_MINI_RIGHT;
        else if(s->data[CX]>(unsigned)bx)key=L3_MINI_LEFT;
        else key=L3_MINI_ACT;
    } else if(kind==L3_MINI_SHAPES) {
        unsigned x,y,r,tx=0,ty=0,tr=0;int found=0;
        for(r=0;r<4 && !found;++r)for(y=0;y<8 && !found;++y)for(x=0;x<8;++x)
            if(assembly_match(s,(int)x,(int)y,r)){tx=x;ty=y;tr=r;found=1;break;}
        if(!found)return 0;
        /* Move to the interior before rotating, so edge bounds cannot jam it. */
        if(s->data[ROT]!=tr){if(s->data[CX]>4)key=L3_MINI_LEFT;else if(s->data[CY]>4)key=L3_MINI_UP;else key=L3_MINI_ROTATE;}
        else if(s->data[CX]<tx)key=L3_MINI_RIGHT;else if(s->data[CX]>tx)key=L3_MINI_LEFT;
        else if(s->data[CY]<ty)key=L3_MINI_DOWN;else if(s->data[CY]>ty)key=L3_MINI_UP;
        else key=L3_MINI_ACT;
    } else if(kind==L3_MINI_MERGE1024) {
        unsigned d;int best=INT_MIN;
        for(d=0;d<4;++d){L3MiniState t=*s;int value;merge_move(&t,d);if(t.meta[MOVES]==s->meta[MOVES])continue;
            value=merge_search(&t,3);if(value>best){best=value;key=keys[d];}}
        if(!key)return 0;
    } else {
        unsigned q=at(s->data[CX],s->data[CY]),d,target=0,i;uint16_t dist[400],back[400];int best=INT_MAX;
        maze_distances(s,q,dist);

static uint32_t compact_random32(L3CompactMergeState *s)

static uint32_t compact_add_sat(uint32_t a, uint32_t b)
