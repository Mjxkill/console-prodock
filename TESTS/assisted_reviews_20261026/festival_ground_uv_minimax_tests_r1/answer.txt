#include <assert.h>
#include <math.h>
#include <stdint.h>
#include <stdio.h>
#include "festival_surface_uv.h"

static uint32_t uvt_pack(int16_t u, int16_t v)
{
    return ((uint32_t)(uint16_t)u << 16) | (uint32_t)(uint16_t)v;
}
static int32_t uvt_unpack(uint32_t p, int high)
{
    uint32_t q = high ? (p >> 16) : (p & UINT32_C(0xffff));
    return q <= UINT32_C(0x7fff) ? (int32_t)q
                                  : ((int32_t)q - INT32_C(65536));
}

static int uvt_ok(int texture, int horizontal, float u, float v, uint32_t expected)
{
    uint32_t got = UINT32_C(0xdeadbeef);
    int rc = festival_surface_uv(texture, horizontal, u, v, &got);
    assert(rc == 1);
    assert(got == expected);
    return 1;
}
static int uvt_bad(int texture, int horizontal, float u, float v)
{
    uint32_t got = UINT32_C(0xa5a5a5a5);
    int rc = festival_surface_uv(texture, horizontal, u, v, &got);
    assert(rc == 0);
    assert(got == UINT32_C(0xa5a5a5a5));
    return 1;
}

int main(void)
{
    int count = 0;
    uint32_t a, b;
    int rc;

    count += uvt_ok(0, 1, 1.0f, 0.0f, uvt_pack(29, -38));
    count += uvt_ok(0, 1, 0.0f, 1.0f, uvt_pack(38, 29));
    count += uvt_ok(0, 1, -1.0f, 0.0f, uvt_pack(-29, 38));
    count += uvt_ok(0, 1, 0.0f, -1.0f, uvt_pack(-38, -29));
    count += uvt_ok(1, 1, 1.0f, -2.0f, uvt_pack(64, -128));
    count += uvt_ok(2, 1, 1.0f, -2.0f, uvt_pack(32, -64));
    count += uvt_ok(0, 0, 1.0f, -2.0f, uvt_pack(32, -64));
    count += uvt_ok(1, 0, 1.0f, -1.0f, uvt_pack(32, -32));

    count += uvt_ok(0, 1, 96.0f, 96.0f, uvt_pack(6451, -922));
    count += uvt_ok(0, 1, 96.0f, -96.0f, uvt_pack(-922, -6451));
    count += uvt_ok(0, 1, -96.0f, 96.0f, uvt_pack(922, 6451));
    count += uvt_ok(0, 1, -96.0f, -96.0f, uvt_pack(-6451, 922));
    count += uvt_ok(0, 1, 96.0f, 0.0f, uvt_pack(2765, -3686));
    count += uvt_ok(0, 1, -96.0f, 0.0f, uvt_pack(-2765, 3686));
    count += uvt_ok(0, 1, 0.0f, 96.0f, uvt_pack(3686, 2765));
    count += uvt_ok(0, 1, 0.0f, -96.0f, uvt_pack(-3686, -2765));
    count += uvt_ok(1, 1, 96.0f, -96.0f, uvt_pack(6144, -6144));
    count += uvt_ok(2, 1, -96.0f, 96.0f, uvt_pack(-3072, 3072));

    count += uvt_ok(2, 1, 0.015625f, 0.0f, uvt_pack(1, 0));
    count += uvt_ok(2, 1, -0.015625f, 0.0f, uvt_pack(-1, 0));
    count += uvt_ok(2, 1, 0.046875f, 0.0f, uvt_pack(2, 0));
    count += uvt_ok(2, 1, -0.046875f, 0.0f, uvt_pack(-2, 0));
    count += uvt_ok(2, 1, 0.0f, 0.015625f, uvt_pack(0, 1));
    count += uvt_ok(2, 1, 0.0f, -0.015625f, uvt_pack(0, -1));

    count += uvt_ok(2, 1, 1023.96875f, -1024.0f, uvt_pack(32767, -32768));
    count += uvt_ok(2, 1, -1024.0f, 1023.96875f, uvt_pack(-32768, 32767));
    count += uvt_ok(2, 1, -1024.01f, 0.0f, uvt_pack(-32768, 0));
    count += uvt_ok(1, 1, -512.0f, 0.0f, uvt_pack(-32768, 0));

    count += uvt_ok(2, 1, 129.0f, -130.0f, uvt_pack(4128, -4160));
    count += uvt_ok(1, 1, 129.0f, -130.0f, uvt_pack(8256, -8320));

    a = b = 0;
    rc = festival_surface_uv(0, 1, 48.0f, -24.0f, &a);
    assert(rc == 1);
    rc = festival_surface_uv(0, 1, 48.0f, -24.0f, &b);
    assert(rc == 1);
    assert(a == b);
    assert(a == uvt_pack(461, -2534));
    count += 2;

    {
        const float ru[4] = { 36.0f, -48.0f, -36.0f, 48.0f };
        const float rv[4] = { 48.0f, 36.0f, -48.0f, -36.0f };
        const int16_t eu[4] = { 2880, 0, -2880, 0 };
        const int16_t ev[4] = { 0, 2880, 0, -2880 };
        uint32_t p[4];
        int i;
        for (i = 0; i < 4; ++i) {
            rc = festival_surface_uv(0, 1, ru[i], rv[i], &p[i]);
            assert(rc == 1);
            assert(p[i] == uvt_pack(eu[i], ev[i]));
            ++count;
        }
        for (i = 0; i < 4; ++i) {
            int32_t du = uvt_unpack(p[i], 1);
            int32_t dv = uvt_unpack(p[i], 0);
            int64_t d2 = (int64_t)du * du + (int64_t)dv * dv;
            assert(d2 == INT64_C(8294400));
        }
    }

    count += uvt_bad(0, 1, NAN, 0.0f);
    count += uvt_bad(0, 1, 0.0f, NAN);
    count += uvt_bad(1, 1, INFINITY, 0.0f);
    count += uvt_bad(1, 1, 0.0f, -INFINITY);
    count += uvt_bad(2, 1, INFINITY, INFINITY);
    count += uvt_bad(2, 1, 1023.984375f, 0.0f);
    count += uvt_bad(2, 1, -1024.015625f, 0.0f);
    count += uvt_bad(2, 1, 1024.0f, 0.0f);
    count += uvt_bad(2, 1, -1024.1f, 0.0f);
    count += uvt_bad(1, 1, 512.0f, 0.0f);
    count += uvt_bad(0, 1, 1024.0f, 0.0f);

    rc = festival_surface_uv(0, 1, 0.0f, 0.0f, NULL);
    assert(rc == 0);
    ++count;

    printf("OK: %d UV test cases passed\n", count);
    return 0;
}
