def check(cond,msg,errors):
    if not cond: errors.append(msg)
def sha_file(path):
    h=hashlib.sha256()
    with open(path,"rb") as f:
        for block in iter(lambda: f.read(65536), b""): h.update(block)
    return h.hexdigest()
def queue_sha(q): return hashlib.sha256(json.dumps(q,sort_keys=True,separators=(",",":")).encode()).hexdigest()
def safe_packet(project,rel):
    p=Path(rel); root=project.resolve()
    if p.is_absolute() or ".." in p.parts: raise ValueError("path_traversal")
    if p.suffix!=".txt": raise ValueError("packet_not_txt")
    cur=project
    for part in p.parts:
        cur=cur/part
        if cur.is_symlink(): raise ValueError("packet_symlink")
    full=(project/p).resolve()
    if root!=full and root not in full.parents: raise ValueError("path_escape")
    if not full.is_file(): raise ValueError("packet_missing")
    if full.stat().st_size>MAX_PACKET: raise ValueError("packet_too_large")
    return full
def validate_job(j,project,e):
    if not isinstance(j,dict): e.append("job_not_object"); return
    for k in ("task_id","run_id"): check(isinstance(j.get(k),str) and NAME.match(j.get(k)) is not None,"bad_"+k,e)
    for k in ("domain","task_type","acceptance_criteria"): check(isinstance(j.get(k),str) and bool(j.get(k)),"bad_"+k,e)
    check(j.get("model") in MODELS,"bad_model",e); check(j.get("difficulty","unspecified") in DIFFS,"bad_difficulty",e)
    check(isinstance(j.get("priority"),int) and not isinstance(j.get("priority"),bool),"bad_priority",e)
    rel=j.get("packet"); check(isinstance(rel,str) and bool(rel),"bad_packet",e); full=None
    if isinstance(rel,str) and rel:
        try: full=safe_packet(project,rel)
        except ValueError as ex: e.append("packet:"+str(ex))
        declared=j.get("packet_sha256")
        check(isinstance(declared,str) and re.fullmatch(r"[0-9a-f]{64}",declared or "") is not None,"bad_packet_sha256",e)

    after=j.get("after"); check(isinstance(after,list),"bad_after",e)
    if isinstance(after,list):
        for a in after:
            check(isinstance(a,dict),"bad_after_item",e)
            if isinstance(a,dict):

def has_cycle(graph):
    color={}
    def visit(n):
        c=color.get(n,0)
        if c==1: return True
        if c==2: return False
        color[n]=1
        if any(visit(x) for x in graph.get(n,())): return True
        color[n]=2; return False
    return any(visit(n) for n in list(graph))
def validate_queue(q,project):
    e=[]
    if not isinstance(q,dict): return ["queue_not_object"]
    check(q.get("schema")==SCHEMA,"bad_schema",e)
    ma=q.get("max_active"); check(isinstance(ma,int) and not isinstance(ma,bool) and 1<=ma<=8,"bad_max_active",e)
    jobs=q.get("jobs"); check(isinstance(jobs,list),"bad_jobs",e)
    if not isinstance(jobs,list): return e
    tids=[]; rids=[]
    for j in jobs:
        validate_job(j,project,e)
        if isinstance(j,dict):
            if isinstance(j.get("task_id"),str): tids.append(j["task_id"])
            if isinstance(j.get("run_id"),str): rids.append(j["run_id"])
    check(len(tids)==len(set(tids)),"duplicate_task_id",e); check(len(rids)==len(set(rids)),"duplicate_run_id",e)
    known=set(tids); graph={}
    for j in jobs:
        if not isinstance(j,dict) or not isinstance(j.get("task_id"),str): continue
        deps=[]
        for a in j.get("after",[]) if isinstance(j.get("after"),list) else []:
            if isinstance(a,dict) and isinstance(a.get("task_id"),str):
                deps.append(a["task_id"]); check(a["task_id"] in known,"missing_dependency",e)
        graph[j["task_id"]]=deps
    check(has_cycle(graph),"dependency_cycle",e); return e
def load_state(session):
    path=Path(session)/"state.json"
    if path.is_symlink(): raise ValueError("state_symlink")
    if not path.exists(): return None
    try: return json.loads(path.read_text(encoding="utf-8"))
    except (OSError,ValueError) as exc: raise ValueError("state_invalid_json") from exc
def save_state(session,state):
    d=Path(session); d.mkdir(parents=True,exist_ok=True); path=d/"state.json"
    if path.is_symlink(): raise ValueError("state_symlink")
    tmp=d/(".state.%d.%d.tmp"%(os.getpid(),time.time_ns())); fd=os.open(tmp,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    try:
        with os.fdopen(fd,"w",encoding="utf-8") as f:
            json.dump(state,f,sort_keys=True); f.write("\n"); f.flush(); os.fsync(f.fileno())
        os.replace(tmp,path); dfd=os.open(d,os.O_RDONLY|os.O_DIRECTORY)
        try: os.fsync(dfd)
        finally: os.close(dfd)
    finally:
        if tmp.exists(): tmp.unlink()
def new_state(q): return {"schema":SCHEMA,"queue_sha256":queue_sha(q),"jobs":{j["task_id"]:{"status":"PENDING"} for j in q["jobs"]}}
def reconcile(state,q,project,session):
    if not isinstance(state,dict) or not isinstance(state.get("jobs"),dict): raise ValueError("state_jobs_invalid")
    old=state["jobs"]; jobs={}
    for j in q["jobs"]:
        tid=j["task_id"]; rec=old.get(tid) if isinstance(old.get(tid),dict) else {}
        rec.update({k:j[k] for k in ("task_id","run_id","packet","packet_sha256","model","domain","task_type","difficulty","acceptance_criteria","priority","after","parent_task_id","focus") if k in j})
        rec.setdefault("difficulty","unspecified"); rec.setdefault("after",[]); s=rec.get("status")
        if s not in ALL: rec["status"]="NEEDS_REVIEW"; rec["error"]="invalid_status"
        elif s=="RUNNING":
            c=rec.get("claim") if isinstance(rec.get("claim"),dict) else {}
            if not isinstance(c.get("pid"),int) or c.get("pid",0)<=0 or c.get("start_ticks") is None: rec["status"]="NEEDS_REVIEW"; rec["error"]="orphan_claim"
            elif not pid_alive(c["pid"],c["start_ticks"]): rec["status"]="NEEDS_REVIEW"; rec["error"]="orphan_claim_dead_pid"
            else: rec["status"]="NEEDS_REVIEW"; rec["error"]="orphan_claim_live_pid"
        elif rec.get("claim") and s not in TERMINAL: rec["status"]="NEEDS_REVIEW"; rec["error"]="claim_without_terminal"
        jobs[tid]=rec
    state.update({"schema":SCHEMA,"queue_sha256":queue_sha(q),"jobs":jobs,"unlisted":{t:r for t,r in old.items() if t not in jobs}}); return state
def read_start_ticks(pid):
    try: return int(Path("/proc/%d/stat"%pid).read_text(encoding="utf-8").rsplit(")",1)[1].split()[19])
    except (OSError,ValueError,IndexError): return None
def pid_alive(pid,ticks): return isinstance(pid,int) and pid>0 and ticks is not None and read_start_ticks(pid)==ticks
def claude_ok(q,allow): return all(j["model"] not in CLAUDE_MODELS for j in q["jobs"]) or (q.get("allow_claude") is True and allow)
class SessionLock:
    def __init__(self,session): self.session=Path(session)
    def __enter__(self):
        if self.session.is_symlink(): raise ValueError("session_symlink")
        self.session.mkdir(parents=True,exist_ok=True)
        self.fd=os.open(self.session/"queue.lock",os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW,0o600)
        if not stat.S_ISREG(os.fstat(self.fd).st_mode): os.close(self.fd); raise ValueError("lock_not_regular")
        try: fcntl.flock(self.fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except OSError as exc: os.close(self.fd); raise ValueError("session_locked") from exc
        return self
    def __exit__(self,*exc): fcntl.flock(self.fd,fcntl.LOCK_UN); os.close(self.fd)
def _open_log(path): return open(path,"xb")
def start_job(state,job,project,session,runner):
    run_dir=Path(session)/"runs"/job["run_id"]
    try: run_dir.parent.mkdir(parents=True,exist_ok=True); run_dir.mkdir(exist_ok=False)
    except OSError: job["status"]="FAILED"; job["error"]="run_dir_exists_or_unwritable"; job["updated_at"]=now_iso(); return None
    try: log=_open_log(run_dir/"runner.log")
    except OSError: job["status"]="FAILED"; job["error"]="log_open_failed"; job["updated_at"]=now_iso(); return None
    job["status"]="RUNNING"; job["claim"]={"claimed_at":now_iso(),"pid":None,"start_ticks":None}; job.pop("error",None); job["updated_at"]=now_iso()
    try: save_state(session,state)
    except Exception: log.close(); job["status"]="NEEDS_REVIEW"; job["error"]="claim_write_failed"; return None
    cmd=[str(runner),"--project",str(project),"--out",str(run_dir),"--packet",job["packet"],"--model",job["model"],"--task-id",job["task_id"],"--run-id",job["run_id"],"--domain",job["domain"],"--task-type",job["task_type"],"--difficulty",job.get("difficulty","unspecified"),"--acceptance-criteria",job["acceptance_criteria"],"--queued-at",job["claim"]["claimed_at"]]
    if job.get("parent_task_id"): cmd+=["--parent-task-id",job["parent_task_id"]]
    if job.get("focus"): cmd+=["--focus",job["focus"]]
    try: proc=subprocess.Popen(cmd,cwd=str(project),stdout=log,stderr=subprocess.STDOUT,shell=False,close_fds=True)
    except OSError as exc:
        log.close(); job["status"]="FAILED"; job["error"]="popen_failed:"+str(exc); job["updated_at"]=now_iso(); save_state(session,state); return None
    log.close(); job["claim"].update({"pid":proc.pid,"start_ticks":read_start_ticks(proc.pid)})
    try: save_state(session,state)
    except Exception: job["error"]="claim_identity_persist_failed"
    return proc
def finish(state,job,rc,session):
    job["exit_code"]=rc; job["updated_at"]=now_iso(); p=Path(session)/"runs"/job["run_id"]/STATUS_FILE; doc=None
    try:
        if p.is_symlink(): raise OSError("status_symlink")
        doc=json.loads(p.read_text(encoding="utf-8"))
    except (OSError,ValueError): doc=None
    s=doc.get("status") if isinstance(doc,dict) else None
    if rc!=0: job["status"]="FAILED"; job["error"]="exit_nonzero:%s"%rc
    elif s==SUCCESS: job["status"]=SUCCESS; job.pop("error",None)
    elif s in FAIL or s in EXTRA: job["status"]=s; job["error"]="runner_status:%s"%s
    else: job["status"]="NEEDS_REVIEW"; job["error"]="invalid_status"
def is_ready(r,jobs):
    for a in r.get("after",[]):
        dep=jobs.get(a.get("task_id")); ds=dep.get("status") if isinstance(dep,dict) else None
        if ds not in TERMINAL: return False
        if a.get("condition")=="on_success" and ds!=SUCCESS: return False
    return True
def settle(state,q):
    jobs=state["jobs"]; changed=True
    while changed:
        changed=False
        for r in jobs.values():
            if r.get("status")!="PENDING": continue
            block=None
            for a in r.get("after",[]):
                dep=jobs.get(a["task_id"]); ds=dep.get("status") if isinstance(dep,dict) else None
                if ds in TERMINAL and a["condition"]=="on_success" and ds!=SUCCESS: block="%s:%s"%(a["task_id"],ds); break
            if block: r["status"]="BLOCKED_DEPENDENCY"; r["error"]="dependency_failed:"+block; r["updated_at"]=now_iso(); changed=True
def summarize(state):
    counts={}
    for r in state["jobs"].values(): counts[r.get("status","NEEDS_REVIEW")]=counts.get(r.get("status","NEEDS_REVIEW"),0)+1
    bad=any(s in FAIL or s in EXTRA for s in counts); done=bool(counts) and all(s==SUCCESS for s in counts)
    return {"counts":counts,"result":"NEEDS_ATTENTION" if bad else ("DONE" if done else ("PENDING" if counts else "EMPTY")),"exit_code":1 if bad else 0}
def run_session(q,project,session,runner,allow_claude=False):
    if not claude_ok(q,allow_claude): raise ValueError("claude_not_allowed")
    state=load_state(session) or new_state(q); state=reconcile(state,q,project,session); save_state(session,state); active={}
    try:
        while True:
            settle(state,q); save_state(session,state); slots=q["max_active"]-len(active)
            for r in sorted((x for x in state["jobs"].values() if x["status"]=="PENDING"),key=lambda x:(-x.get("priority",0),x["task_id"])):
                if slots<=0: break
                if not is_ready(r,state["jobs"]): continue
                p=start_job(state,r,project,session,runner)
                if p is None: save_state(session,state); continue
                active[r["task_id"]]=p; slots-=1
            for tid,p in list(active.items()):
                rc=p.poll()
                if rc is not None: finish(state,state["jobs"][tid],rc,session); del active[tid]
            save_state(session,state)
            if not active:
                settle(state,q); save_state(session,state); pend=[r for r in state["jobs"].values() if r["status"]=="PENDING"]
                if not pend: break
                if any(is_ready(r,state["jobs"]) for r in pend): continue
                for r in pend: r["status"]="NEEDS_REVIEW"; r["error"]="dependency_stall"; r["updated_at"]=now_iso()
                save_state(session,state); break
            time.sleep(0.05)
    except KeyboardInterrupt:
        for tid in active:
            j=state["jobs"].get(tid)
            if j and j.get("status")=="RUNNING": j["status"]="NEEDS_REVIEW"; j["error"]="interrupted_parent"
        try: save_state(session,state)
        except Exception: pass
        raise
    return summarize(state)
def main(argv=None):
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--project",required=True); ap.add_argument("--session",required=True); ap.add_argument("--queue",required=True)
    ap.add_argument("--runner"); ap.add_argument("--runner-sha256"); ap.add_argument("--run",action="store_true"); ap.add_argument("--allow-claude",action="store_true")
    a=ap.parse_args(argv); project=Path(a.project).resolve()
    if not project.is_dir(): print(json.dumps({"error":"project_not_dir"})); return 2
    try: q=json.loads(Path(a.queue).read_text(encoding="utf-8"))
    except (OSError,ValueError) as ex: print(json.dumps({"error":"queue_read:"+str(ex)})); return 2
    errors=validate_queue(q,project)
    if errors: print(json.dumps({"errors":errors},sort_keys=True)); return 2
    try:
        if a.run:
            if not a.runner or not a.runner_sha256: print(json.dumps({"error":"runner_and_sha_required"})); return 2
            rp=Path(a.runner)
            if rp.is_symlink() or not rp.is_file(): print(json.dumps({"error":"bad_runner"})); return 2
            if sha_file(rp)!=a.runner_sha256.lower(): print(json.dumps({"error":"runner_sha_mismatch"})); return 2
            with SessionLock(a.session): summary=run_session(q,project,a.session,rp.resolve(),a.allow_claude)
        else:
            state=load_state(a.session) or new_state(q); summary=summarize(reconcile(state,q,project,a.session))
    except (OSError,ValueError) as ex: print(json.dumps({"error":str(ex)})); return 2
    print(json.dumps(summary,sort_keys=True)); return summary["exit_code"]
if __name__=="__main__": sys.exit(main())
