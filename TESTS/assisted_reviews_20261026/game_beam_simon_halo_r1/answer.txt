**Facts packet**

- L’enveloppe est dérivée des champs phase/horloge existants, pas des frames.  
- Couleur RGB565 reste la valeur exacte du cristal ; seule l’alpha halo (`alpha4`) varie.

## Conception minimale

Helper d’état compact `L3SimonHalo` :  
- `color_rgb565` jamais modifié par l’enveloppe (couleur pilote exacte).  
- `level_q8` 0..255, `alpha4 = level_q8 >> 4` pour ARGB4444 distincte.  
- États soft : attaque quadratique douce, pic lisible, fade progressif.  
- `restarts` prouve qu’un déclenchement identique même couleur relance l’enveloppe.  
- `paused` gèle `elapsed_ms`; aucun rattrapage silencieux.  
- `dt` borné à 250 ms.  
- Addition saturante 32 bits (pas d’overflow wrap silencieux).  

### `l3_simon_halo.h`

```c
#ifndef L3_SIMON_HALO_H
#define L3_SIMON_HALO_H
#include "l3_simon.h"
#include <stdint.h>

#define L3_SIMON_HALO_DT_MAX 250u

typedef struct {
    uint16_t color_rgb565;
    uint8_t active;
    uint8_t paused;
    uint8_t level_q8;
    uint8_t alpha4;   /* ARGB4444 distincte de RGB565 */
    uint8_t radius_q8;
    uint8_t last_phase;
    uint8_t last_active_color;
    uint32_t last_feedback_ms;
    uint32_t elapsed_ms;
    uint32_t attack_ms;
    uint32_t hold_ms;
    uint32_t fade_ms;
    uint32_t restarts;
} L3SimonHalo;

void l3_simon_halo_reset(L3SimonHalo *);
void l3_simon_halo_set_paused(L3SimonHalo *, int);
void l3_simon_halo_trigger(L3SimonHalo *, uint16_t color_rgb565,
                           uint32_t attack_ms, uint32_t hold_ms, uint32_t fade_ms);
uint32_t l3_simon_halo_update(L3SimonHalo *, uint32_t dt_ms);
int l3_simon_halo_sync(L3SimonHalo *, const L3SimonView *,
                       const uint16_t colors_rgb565[4], uint32_t dt_ms);
#endif
```

### `l3_simon_halo.c`

```c
#include "l3_simon_halo.h"
#include <limits.h>
#include <string.h>

static uint32_t add_ms(uint32_t a, uint32_t b)
{ return (UINT32_MAX - a < b) ? UINT32_MAX : a + b; }

static void refresh(L3SimonHalo *h)
{
    if (!h || !h->active) { h->level_q8 = 0; h->radius_q8 = 0; h->alpha4 = 0; return; }
    if (!h->attack_ms || !h->fade_ms) { h->active = 0; refresh(h); return; }
    if (h->elapsed_ms >= h->attack_ms + h->hold_ms + h->fade_ms) {
        h->active = 0; refresh(h); return;
    }
    if (h->elapsed_ms < h->attack_ms) {
        uint32_t x = h->elapsed_ms * 255u / h->attack_ms;
        h->level_q8 = (uint8_t)((x * x) / 255u);
        h->radius_q8 = (uint8_t)(96u + (x * 159u) / 255u);
    } else if (h->elapsed_ms < h->attack_ms + h->hold_ms) {
        h->level_q8 = 255; h->radius_q8 = 255;
    } else {
        uint32_t t = h->elapsed_ms - h->attack_ms - h->hold_ms;
        h->level_q8 = (uint8_t)(255u - (255u * t) / h->fade_ms);
        h->radius_q8 = 255;
    }
    h->alpha4 = h->level_q8 >> 4;
}

void l3_simon_halo_reset(L3SimonHalo *h)
{
    if (h) { memset(h, 0, sizeof(*h)); h->last_phase = 255; h->last_active_color = 255; }
}

void l3_simon_halo_set_paused(L3SimonHalo *h, int paused)
{ if (h) h->paused = !!paused; }

void l3_simon_halo_trigger(L3SimonHalo *h, uint16_t color_rgb565,
                           uint32_t attack_ms, uint32_t hold_ms, uint32_t fade_ms)
{
    if (!h) return;
    h->color_rgb565 = color_rgb565; /* RGB565 exact */
    h->attack_ms = attack_ms; h->hold_ms = hold_ms; h->fade_ms = fade_ms;
    h->elapsed_ms = 0; h->active = 1; h->restarts++;
    refresh(h);
}

uint32_t l3_simon_halo_update(L3SimonHalo *h, uint32_t dt_ms)
{
    if (!h || !h->active || h->paused) return 0;
    uint32_t dt = (dt_ms > L3_SIMON_HALO_DT_MAX) ? L3_SIMON_HALO_DT_MAX : dt_ms;
    h->elapsed_ms = add_ms(h->elapsed_ms, dt);
    refresh(h);
    return dt;
}

int l3_simon_halo_sync(L3SimonHalo *h, const L3SimonView *v,
                       const uint16_t colors_rgb565[4], uint32_t dt_ms)
{
    int changed = 0;
    if (!h || !v) return 0;

    if (v->phase == L3_SIMON_SHOW) {
        if (h->last_phase != L3_SIMON_SHOW || h->last_active_color != v->active_color) {
            uint16_t c = (v->active_color < 4 && colors_rgb565) ?
                           colors_rgb565[v->active_color] : 0u;
            l3_simon_halo_trigger(h, c, 120u, 360u, 120u);
            changed = 1; /* relance même si couleur identique */
        }
    } else if (v->phase == L3_SIMON_INPUT && v->feedback_ms > 0 &&
               (h->last_phase != L3_SIMON_INPUT || h->last_feedback_ms == 0 ||
                v->feedback_ms > h->last_feedback_ms)) {
        uint16_t c = (v->active_color < 4 && colors_rgb565) ?
                       colors_rgb565[v->active_color] : 0u;
        l3_simon_halo_trigger(h, c, 40u, 80u, 120u);
        changed = 1;
    }

    h->last_phase = v->phase;
    h->last_active_color = v->active_color;
    h->last_feedback_ms = v->feedback_ms;
    if (dt_ms && !h->paused) l3_simon_halo_update(h, dt_ms);
    return changed;
}
```

## Tests numériques (à formaliser, non exécutés ici)

1. **Entrée**  
   `reset` → `active=0`, `level_q8=0`, `radius_q8=0`, `restarts=0`, `paused=0`.

2. **Sortie**  
   Après flash en cours, `reset` → retour identique à entered. Aucune accumulation.

3. **Attaque / pic / fade (SHOW 600 ms)**  
   `trigger(0x07E0, 120,360,120)`  
   - `update(0)` : `level_q8=0`, `radius_q8=96`  
   - `update(60)` cumul 60 → `level_q8≈63`, `alpha4=3`  
   - cumul 120 → `level_q8=255`, `radius_q8=255`  
   - cumul 480 → encore `level=255`  
   - cumul 600 → `active=0`, `level=0`.

4. **Pause / reprise**  
   Cumul 60, `set_paused(1)`, `update(1000)` → appliqué `0`, `elapsed=60`, `level≈63`.  
   `set_paused(0)`, `update(0)` → pas d’avance fantôme, puis normalement.

5. **Long dt**  
   `update(1000)` avec actif → retour `250`, pas 1000 ; `elapsed` avance de 250 max.

6. **Wrap / overflow 32 bits**  
   `add_ms(0xfffffff0,31)` sature à `0xffffffff` au lieu de wrapper.  
   Équivalent modulo pour capture d’horloge externe :  
   `(uint32_t)(0x0000000f - 0xfffffff0)` = 31.

7. **Deux mêmes couleurs**  
   Deux `trigger(même RGB565,...)` consécutifs sans phase entre eux :  
   `restarts=2`, deuxième `elapsed=0`, `level=0` même couleur.  
   En synchro Simon : `SHOW couleur 1 (600→0)` → `GAP` → `SHOW couleur 1` doit déclencher `changed=1` malgré couleur identique.

8. **Alpha distinct**  
   `color_rgb565` reste inchangé ; seule `alpha4 = level >> 4`. Pas de scanout fractionnel.

## Intégration Simon (agent `luma_minigames`)

- Au `enter/exit` ou reset de partie : `l3_simon_halo_reset`.  
- Après `l3_simon_step`: appeler `l3_simon_halo_sync(halo, view, colors, dt)` ; ne pas redessiner le canvas.  
- Dessiner les quads halo seulement pour la cellule active, avec RGB565 exact et alpha ARGB4444 dynamique.  
- Pendant GAP, la synchro laisse le halo mourir puis passer à 0 → séparation noire nette avant la deuxième même couleur.  
- Aucune modification des règles, séquence ou save Simon.
