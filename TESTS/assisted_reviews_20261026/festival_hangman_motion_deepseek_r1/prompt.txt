/* Static 3D garden. Each triangle is transformed/projected by G3D1. */
typedef struct {gpu_geom_vertex v[240];gpu_geom_face f[240];unsigned nv,nf,at,texture;L3Render *r;L3Festival *owner;} Mesh;
static int mesh_flush(Mesh *m){
    if(!m->nf)return 1;if(m->owner->mesh_count>=24||m->at+(m->nv+m->nf)*32>(18u<<20))return 0;
    unsigned va=m->at;publish(m->r,m->at,m->v,m->nv*32);m->at+=m->nv*32;unsigned fa=m->at;publish(m->r,m->at,m->f,m->nf*32);m->at+=m->nf*32;
    m->owner->mesh_texture[m->owner->mesh_count]=m->texture;
    m->owner->meshes[m->owner->mesh_count++]=(gpu_geom_mesh){PRODOCK_GPU_BASE+va,PRODOCK_GPU_BASE+fa,m->nv,m->nf};m->nv=m->nf=0;return 1;
}
static int mesh_texture(Mesh *m,unsigned texture){if(!mesh_flush(m))return 0;m->texture=texture;return 1;}
static int tri(Mesh *m,L3Vec3 a,L3Vec3 b,L3Vec3 c,unsigned red,unsigned green,unsigned blue){
    if(m->nv+3>240||m->nf==240)if(!mesh_flush(m))return 0;
    unsigned start=m->nv;L3Vec3 p[3]={a,b,c};
    for(unsigned i=0;i<3;i++){gpu_geom_vertex v={0};if(!gpu_geom_fixed(p[i].x,&v.x)||!gpu_geom_fixed(p[i].y,&v.y)||!gpu_geom_fixed(p[i].z,&v.z))return 0;m->v[m->nv++]=v;}
    gpu_geom_face f={{0}};f.words[0]=(3u<<24)|start|((start+1)<<8)|((start+2)<<16);f.words[4]=32767;f.words[6]=red|(green<<9)|(blue<<18);
    float nx=fabsf((b.y-a.y)*(c.z-a.z)-(b.z-a.z)*(c.y-a.y));
    float ny=fabsf((b.z-a.z)*(c.x-a.x)-(b.x-a.x)*(c.z-a.z));
    float nz=fabsf((b.x-a.x)*(c.y-a.y)-(b.y-a.y)*(c.x-a.x));
    for(unsigned i=0;i<3;i++){
        float u=nx>ny&&nx>nz?p[i].z:p[i].x,v=ny>=nx&&ny>=nz?p[i].z:-p[i].y;
        int uu=(int)lroundf(u*32),vv=(int)lroundf(v*32);
        f.words[i+1]=((uint32_t)(uint16_t)uu<<16)|(uint16_t)vv;
    }
    m->f[m->nf++]=f;return 1;
}
static int quad3(Mesh *m,L3Vec3 a,L3Vec3 b,L3Vec3 c,L3Vec3 d,unsigned r,unsigned g,unsigned bcol){return tri(m,a,b,c,r,g,bcol)&&tri(m,a,c,d,r,g,bcol);}
static int box(Mesh *m,float x,float y,float z,float w,float h,float d,unsigned r,unsigned g,unsigned b){
    L3Vec3 a={x-w/2,y,z-d/2},bb={x+w/2,y,z-d/2},c={x+w/2,y,z+d/2},dd={x-w/2,y,z+d/2};
    L3Vec3 e=a,f=bb,gg=c,hh=dd;e.y=f.y=gg.y=hh.y=y+h;
    return quad3(m,e,f,gg,hh,r,g,b)&&quad3(m,a,bb,f,e,r*3/4,g*3/4,b*3/4)&&quad3(m,bb,c,gg,f,r*7/8,g*7/8,b*7/8)&&quad3(m,c,dd,hh,gg,r*2/3,g*2/3,b*2/3)&&quad3(m,dd,a,e,hh,r*4/5,g*4/5,b*4/5);
}
static void garden_tree(unsigned i,float *x,float *z,float *height,float *width){
    /* Three irregular groves leave a view northward towards the distant sky.
       The trunk stays outside the accessible square, including its thickness. */
    static const float centers[3]={.25f,2.25f,4.35f};
    float angle=centers[i/12]+((int)(i%12)-5.5f)*.095f+.06f*sinf(i*1.83f);
    float radius=24+4*sinf(i*2.3f)+2*cosf(i*.63f);
    *x=sinf(angle)*radius;*z=cosf(angle)*radius;
    float extent=fmaxf(fabsf(*x),fabsf(*z));
    if(extent<16.5f){*x*=16.5f/extent;*z*=16.5f/extent;}
    *height=4.2f+1.1f*(1+sinf(i*1.71f));
    *width=1.8f+.45f*(1+cosf(i*2.11f));
}
static int garden_geometry(L3Render *r,L3Festival *f){
    f->mesh_count=0;Mesh m={.at=GEOMETRY,.r=r,.owner=f};
    if(!quad3(&m,(L3Vec3){-40,0,-40},(L3Vec3){40,0,-40},(L3Vec3){40,0,40},(L3Vec3){-40,0,40},152,204,102)||!mesh_texture(&m,1))return 0;
    /* Grid tiles form four courts, a central ring and multiple connecting paths. */
    for(int z=-14;z<14;z+=2)for(int x=-14;x<14;x+=2){

static L3Vec3 sub3(L3Vec3 a,L3Vec3 b){return (L3Vec3){a.x-b.x,a.y-b.y,a.z-b.z};}
static float dot3(L3Vec3 a,L3Vec3 b){return a.x*b.x+a.y*b.y+a.z*b.z;}
