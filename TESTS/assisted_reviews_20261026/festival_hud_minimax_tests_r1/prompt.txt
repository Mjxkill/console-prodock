#ifndef FESTIVAL_HUD_PANEL_H
#define FESTIVAL_HUD_PANEL_H

#include <stddef.h>
#include <stdint.h>

/* Palette existante : 65 ombre, 66 champ, 67 bord, 68 bord clair,
 * 76 laiton sombre, 77 laiton, 78 or. Aucun pixel transparent n'est écrit. */
static inline void festival_hud_panel_shape(
    uint8_t *pixels, unsigned width, unsigned height,
    int64_t left, int64_t top, int64_t right, int64_t bottom,
    int64_t cut, uint8_t base, uint8_t top_color, uint8_t bottom_color)
{
    if (pixels == NULL || right <= left || bottom <= top)
        return;

    const int64_t cw = (int64_t)width;
    const int64_t ch = (int64_t)height;
    int64_t x0 = (left < 0) ? 0 : left;
    int64_t y0 = (top < 0) ? 0 : top;
    int64_t x1 = (right > cw) ? cw : right;
    int64_t y1 = (bottom > ch) ? ch : bottom;

    if (x0 > cw) x0 = cw;
    if (y0 > ch) y0 = ch;
    if (x1 < 0) x1 = 0;
    if (y1 < 0) y1 = 0;
    if (x0 >= x1 || y0 >= y1)
        return;

    for (int64_t py = y0; py < y1; ++py) {
        const int64_t dt = py - top;
        const int64_t db = bottom - 1 - py;

        for (int64_t px = x0; px < x1; ++px) {
            const int64_t dl = px - left;
            const int64_t dr = right - 1 - px;

            if (dl < cut && dt < cut && dl + dt < cut)
                continue;
            if (dr < cut && dt < cut && dr + dt < cut)
                continue;
            if (dl < cut && db < cut && dl + db < cut)
                continue;
            if (dr < cut && db < cut && dr + db < cut)
                continue;

            uint8_t color = base;
            if (py == top || px == left)
                color = top_color;
            else if (py == bottom - 1 || px == right - 1)
                color = bottom_color;

            pixels[(size_t)py * (size_t)width + (size_t)px] = color;
        }
    }
}

static inline void festival_hud_panel(
    uint8_t *pixels, unsigned width, unsigned height,
    int x, int y, int w, int h)
{
    if (pixels == NULL || w <= 0 || h <= 0 ||
        w > 16384 || h > 16384)
        return;
    if (width == 0u || height == 0u ||
        width > 16384u || height > 16384u)
        return;
    if (!((width == 800u && height == 480u) ||
          (width == 1280u && height == 720u)))
        return;

    /* Bornes exclusives à droite/bas ; int64_t évite tout débordement. */
    const int64_t left = (int64_t)x;
    const int64_t top = (int64_t)y;
    const int64_t right = left + (int64_t)w;
    const int64_t bottom = top + (int64_t)h;

    /* Ombre opaque décalée de deux pixels, donc au plus panel + 2. */
    festival_hud_panel_shape(pixels, width, height,
        left + 2, top + 2, right + 2, bottom + 2,
        3, 65, 65, 65);

    /* Laiton : un pixel, avec lumière en haut/à gauche et ombre en bas/à droite. */
    festival_hud_panel_shape(pixels, width, height,
        left, top, right, bottom,
        3, 77, 78, 76);

    /* Bord intérieur puis champ bleu-noir, tous deux opaques. */
    festival_hud_panel_shape(pixels, width, height,
        left + 1, top + 1, right - 1, bottom - 1,
        2, 67, 68, 67);
    festival_hud_panel_shape(pixels, width, height,
        left + 2, top + 2, right - 2, bottom - 2,
        1, 66, 66, 66);
}
