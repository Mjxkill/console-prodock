#ifndef L3_MEMORY_SAMPLE_STATS_H
#define L3_MEMORY_SAMPLE_STATS_H

#include <stdio.h>
#include <string.h>

#include "memory_scene.h"

typedef struct {
    unsigned frames;
    unsigned cold_frames;
    unsigned deal_frames;
    unsigned reveal_frames;
    unsigned hide_frames;
    unsigned pair_frames;
    unsigned wait_frames;
    unsigned idle_frames;
    unsigned settled_frames;
    unsigned stable_frames;
    unsigned active_frames;
    unsigned stable_tail;
    unsigned complete;

    unsigned seen_actions;
    unsigned completed_actions;
    unsigned skipped_actions;

    unsigned commands;
    unsigned active_commands;
    size_t published_bytes;
    double published_seconds;

    unsigned pose_changes;
    unsigned active_pose_changes;
    unsigned page_pose_changes;

    unsigned old_n;
    L3MemoryPose old[32];
    unsigned page_old_n[2];
    L3MemoryPose page_old[2][32];

    double seconds;
    double active_seconds;
    double max_active_seconds;

    unsigned final_frames;
    unsigned have_last_frame;
    unsigned last_frame_id;
} L3MemorySample;

typedef L3MemorySample L3MemorySampleStats;

static void l3_memory_sample_observe(L3MemorySample *s, int before_cold,
    const L3MemoryScene *scene, double elapsed_reel, unsigned commands,
    size_t pubbytes, double pubseconds, unsigned frameid)
{
    unsigned trace_kind = 0;
    unsigned changes;
    unsigned page_changes_total = 0;
    unsigned active;
    unsigned settled_now;
    unsigned page;

    if (!s || !scene) return;

    if (s->have_last_frame && frameid == s->last_frame_id) return;
    s->last_frame_id = frameid;
    s->have_last_frame = 1;

    if (scene->trace.count != 0u)
        trace_kind = scene->trace.events[scene->trace.count - 1u].kind;

    if (scene->actions > s->seen_actions) {
        if (scene->actions > s->seen_actions + 1u)
            s->skipped_actions += scene->actions - s->seen_actions - 1u;
        s->seen_actions = scene->actions;
    }

    active = !before_cold && scene->duration > 0.0f && scene->elapsed < scene->duration;

    changes = 0;
    for (unsigned i = 0u; i < scene->poses_n && i < 32u; ++i) {
        for (unsigned j = 0u; j < s->old_n; ++j) {
            if (scene->poses[i].id == s->old[j].id) {
                const L3MemoryPose *a = &scene->poses[i];
                const L3MemoryPose *b = &s->old[j];
                if (a->id != b->id || a->tile != b->tile || a->atlas != b->atlas ||
                    a->x != b->x || a->y != b->y || a->w != b->w ||
                    a->h != b->h || a->alpha != b->alpha) {
                    ++changes;
                }
                break;
            }
        }
    }

    for (page = 0u; page < 2u; ++page) {
        unsigned pchanges = 0u;
        for (unsigned i = 0u; i < scene->page_n[page] && i < 32u; ++i) {
            for (unsigned j = 0u; j < s->page_old_n[page]; ++j) {
                if (scene->pages[page][i].id == s->page_old[page][j].id) {
                    const L3MemoryPose *a = &scene->pages[page][i];
                    const L3MemoryPose *b = &s->page_old[page][j];
                    if (a->id != b->id || a->tile != b->tile || a->atlas != b->atlas ||
                        a->x != b->x || a->y != b->y || a->w != b->w ||
                        a->h != b->h || a->alpha != b->alpha) {
                        ++pchanges;
                    }
                    break;
                }
            }
        }
        page_changes_total += pchanges;
    }

    s->pose_changes += changes;
    s->page_pose_changes += page_changes_total;
    if (active) {
        s->active_pose_changes += changes;
    }

    s->frames++;
    s->seconds += elapsed_reel;
    s->commands += commands;
    s->published_bytes += pubbytes;
    s->published_seconds += pubseconds;

    if (before_cold) {
        ++s->cold_frames;
    } else if (active) {
        ++s->active_frames;
        s->active_seconds += elapsed_reel;
        s->active_commands += commands;
        if (elapsed_reel > s->max_active_seconds) s->max_active_seconds = elapsed_reel;

        if (trace_kind == L3_MEMORY_DEAL) ++s->deal_frames;
        else if (trace_kind == L3_MEMORY_REVEAL) ++s->reveal_frames;
        else if (trace_kind == L3_MEMORY_HIDE) ++s->hide_frames;
        else if (trace_kind == L3_MEMORY_PAIR) ++s->pair_frames;
    } else if (scene->wait > 0.0f) {
        ++s->wait_frames;
    } else if (scene->actions > 0u) {
        ++s->settled_frames;
    } else {
        ++s->idle_frames;
    }

    if (!before_cold && !active && scene->wait <= 0.0f && scene->actions > 0u) {
        ++s->stable_frames;
        ++s->stable_tail;
    } else {
        s->stable_tail = 0u;
    }
    s->complete = s->stable_tail >= 2u;
    s->final_frames = s->stable_tail > 2u ? 2u : s->stable_tail;

    if (!active && !before_cold && scene->actions > 0u)
        s->completed_actions = scene->actions;

    memcpy(s->old, scene->poses, scene->poses_n * sizeof(*scene->poses));
    s->old_n = scene->poses_n;
    for (page = 0u; page < 2u; ++page) {
        memcpy(s->page_old[page], scene->pages[page],
               scene->page_n[page] * sizeof(*scene->pages[page]));
        s->page_old_n[page] = scene->page_n[page];
    }
}

static int l3_memory_sample_valid(const L3MemorySample *s, const L3MemoryScene *scene)
{
    unsigned pages_valid;

    if (!s || !scene || !scene->automatic) return 0;
    if (!s->complete || !s->seen_actions) return 0;
    if (s->completed_actions != s->seen_actions) return 0;
    if (s->skipped_actions != 0u) return 0;
    if (!s->active_frames || !s->active_commands) return 0;

    pages_valid = scene->page_valid[0] && scene->page_valid[1];
    if (!pages_valid) return 0;

    return 1;
}

static void l3_memory_sample_report(const L3MemorySample *s, const L3MemoryScene *scene, int hardware)
{
    unsigned final_pages;

    if (!s || !scene) return;
    final_pages = (scene->page_valid[0] ? 1u : 0u) + (scene->page_valid[1] ? 1u : 0u);

    printf("LUMA3D_MEMORY_SAMPLE frames=%u cold_frames=%u deal_frames=%u reveal_frames=%u hide_frames=%u pair_frames=%u wait_frames=%u idle_frames=%u settled_frames=%u stable_frames=%u active_frames=%u stable_tail=%u complete=%u final_frames=%u final_pages=%u seen_actions=%u completed_actions=%u skipped_actions=%u seconds=%.6f active_seconds=%.6f",
        s->frames, s->cold_frames, s->deal_frames, s->reveal_frames, s->hide_frames,
        s->pair_frames, s->wait_frames, s->idle_frames, s->settled_frames,
        s->stable_frames, s->active_frames, s->stable_tail, s->complete,
        s->final_frames, final_pages, s->seen_actions, s->completed_actions,
        s->skipped_actions, s->seconds, s->active_seconds);

    if (hardware) {
        printf(" active_fps=%.3f max_active_ms=%.3f",
            s->active_seconds > 0.0 ? s->active_frames / s->active_seconds : 0.0,
            s->max_active_seconds * 1000.0);
    }

    printf(" commands=%u active_commands=%u published_bytes=%zu published_seconds=%.6f pose_changes=%u active_pose_changes=%u page_pose_changes=%u old_n=%u page0_n=%u page1_n=%u automatic=%u hardware=%d\n",
        s->commands, s->active_commands, s->published_bytes, s->published_seconds,
        s->pose_changes, s->active_pose_changes, s->page_pose_changes,
        s->old_n, scene->page_n[0], scene->page_n[1], scene->automatic, hardware);
}

/* Compatibility aliases for plausible generated call sites. */
static void l3_memory_sample_stats_observe(L3MemorySampleStats *s, int before_cold,
    const L3MemoryScene *scene, double elapsed_reel, unsigned commands,
    size_t pubbytes, double pubseconds, unsigned frameid)
{
    l3_memory_sample_observe(s, before_cold, scene, elapsed_reel, commands,
                            pubbytes, pubseconds, frameid);
}
static int l3_memory_sample_stats_valid(const L3MemorySampleStats *s, const L3MemoryScene *scene)
{
    return l3_memory_sample_valid(s, scene);
}
static void l3_memory_sample_stats_report(const L3MemorySampleStats *s, const L3MemoryScene *scene, int hardware)
{
    l3_memory_sample_report(s, scene, hardware);
}

    const L3MemoryScene *scene, double elapsed_reel, unsigned commands,
    size_t pubbytes, double pubseconds, unsigned frameid)
{

                            pubbytes, pubseconds, frameid);
}
static int l3_memory_phase_valid(const L3MemorySample *s, const L3MemoryScene *scene)
{
    return l3_memory_sample_valid(s, scene);
}
static void l3_memory_phase_report(const L3MemorySample *s, const L3MemoryScene *scene, int hardware)
{
    l3_memory_sample_report(s, scene, hardware);
}
