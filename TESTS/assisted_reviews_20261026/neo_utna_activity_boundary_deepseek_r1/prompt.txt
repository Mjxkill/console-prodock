L'observateur exécute les commandes et transfère des données qui changent.

au remplissage suivant, qui n'est pas observée. Il est donc insuffisant sous

persistante de case/adresse/enable peut toujours produire le motif.

désignerait pas la cause et ne justifierait pas une promotion.

// Passive IN/NAK observer. No output feeds the USB datapath.
// Commands and coherent snapshots use two dual-clock FIFOs, independent of EP0.
// AHB window: +00 ID, +04 status (bit0 busy, bit1 rejected command),
// +08 completed snapshot count, +0c command, +40..bc snapshot (32 words).
// Command: bit0 clear+arm, bit1 snapshot, bit2 disarm, bits14:8 USB address.
// Use one command at a time; clear+snapshot returns the PRE-clear state.
// USB bus reset is observed, not used to erase this diagnostic.
`default_nettype none
module usb_idle_trace (
 input wire clk, reset_n, bus_reset, sof,
 input wire rx_active, rx_valid, rx_error,
 input wire [7:0] rx_data,
 input wire tx_valid, tx_ready,
 input wire [7:0] tx_data,
 input wire [3:0] endpoint,
 input wire cork,
 input wire hclk, hsel, hwrite,
 input wire [1:0] htrans,
 input wire [31:0] haddr, hwdata,
 output reg [31:0] hrdata,
 output wire hready
);
 // Hold both FIFO ends in reset until each clock has actually restarted.
 (* ASYNC_REG="TRUE" *) reg [2:0] hr=0, ur=0;
 (* ASYNC_REG="TRUE" *) reg [1:0] hu=0, uh=0;
 always @(posedge hclk or negedge reset_n)
  if(!reset_n)begin hr<=0;uh<=0;end
  else begin hr<={hr[1:0],1'b1};uh<={uh[0],ur[2]};end
 always @(posedge clk or negedge reset_n)
  if(!reset_n)begin ur<=0;hu<=0;end
  else begin ur<={ur[1:0],1'b1};hu<={hu[0],hr[2]};end

 wire cmd_ready, cmd_valid, cmd_pop, reply_ready, reply_valid;
 wire [31:0] command, reply;
 reg write_pending, read_pending;
 reg [7:0] address;
 reg busy, rejected;
 reg [31:0] snapshot_count;
 reg [4:0] receive_index;
 reg [31:0] bank[0:31];
 wire [4:0] bank_index=address[6:2]-5'd16;
 wire push_command=write_pending && address==8'h0c && !busy && cmd_ready && !hreset;
 assign hready=!read_pending; // one local wait; never wait on the USB clock
 ddr_dcfifo #(.WIDTH(32),.DEPTH_LOG2(2)) u_commands (
  .wr_clk(hclk),.wr_rst(hreset),.wr_valid(push_command),.wr_data(hwdata),
  .wr_ready(cmd_ready),.wr_full(),.wr_lost(),.wr_count(),
  .rd_clk(clk),.rd_rst(ureset),.rd_valid(cmd_valid),.rd_data(command),
  .rd_ready(cmd_pop),.rd_count());
 ddr_dcfifo #(.WIDTH(32),.DEPTH_LOG2(5)) u_replies (
  .wr_clk(clk),.wr_rst(ureset),.wr_valid(sending && reply_ready && !ureset),
  .wr_data(snapshot[send_index*32+:32]),.wr_ready(reply_ready),
  .wr_full(),.wr_lost(),.wr_count(),
  .rd_clk(hclk),.rd_rst(hreset),.rd_valid(reply_valid),.rd_data(reply),
  .rd_ready(!hreset),.rd_count());
 integer b;
 always @(posedge hclk)begin
  if(hreset)begin
   hrdata<=0;write_pending<=0;read_pending<=0;address<=0;
   busy<=0;rejected<=0;snapshot_count<=0;receive_index<=0;
   for(b=0;b<32;b=b+1)bank[b]<=0;
  end else begin
   write_pending<=hsel && htrans[1] && hwrite && hready;
   read_pending<=hsel && htrans[1] && !hwrite && hready;
   if(hready)address<=haddr[7:0];
   if(write_pending && address==8'h0c)begin
    if(push_command)begin busy<=1;rejected<=0;end
    else rejected<=1;
   end
   if(reply_valid)begin
    bank[receive_index]<=reply;receive_index<=receive_index+1'b1;
    if(receive_index==31)begin snapshot_count<=snapshot_count+1'b1;busy<=0;end
   end
   if(read_pending)begin
    case(address)
     8'h00:hrdata<=32'h55544e41; // UTNA
     8'h04:hrdata<={30'd0,rejected,busy};
     8'h08:hrdata<=snapshot_count;
     default:hrdata<=address>=8'h40 && address<8'hc0 ? bank[bank_index] : 32'd0;
    endcase
   end
  end
 end

 function [4:0] crc5;
  input [10:0] token;
  integer i;reg [4:0] c;
  begin
   c=5'h1f;
   for(i=0;i<11;i=i+1)
    c=(c[0]^token[i]) ? ((c>>1)^5'h14) : (c>>1);
   crc5=~c;
  end
 endfunction
 reg armed;
 reg [6:0] target;
 reg [31:0] ticks,in2,in3,nak2,nak3,data2,data3,miss2,miss3;
 reg [31:0] bad_response,rx_errors,bad_token,resets,sofs,stalls,last_token,last_response;
 reg [7:0] max_offer,max_accept;
 reg bus_reset_q;
 reg [2:0] rx_count;
 reg [7:0] rx_pid,token0,token1;
 reg rx_bad;
 wire token_end=!rx_active_sample && rx_count!=0;
 wire token_in=token_end && rx_pid==8'h69;
 wire token_good=rx_count==3 && !rx_bad && !rx_error_sample &&
                 token1[7:3]==crc5({token1[2:0],token0});
 wire [3:0] token_ep={token1[2:0],token0[7]};
 wire selected_in=token_in && token_good && token0[6:0]==target &&
                  (token_ep==2 || token_ep==3);
 // Any subsequent valid token ends the old transaction context, including
 // SOF, OUT/SETUP/IN for EP0 or another address. Never attribute its reply
 // to an unanswered EP2/EP3 IN.
 wire next_token=token_end && token_good &&
                  (rx_pid==8'h69 || rx_pid==8'he1 || rx_pid==8'h2d ||
                   rx_pid==8'ha5 || rx_pid==8'hb4);
 reg pending;
 reg [3:0] pending_ep;
 reg [31:0] pending_time;
 reg [9:0] age;
 wire [7:0] short_age=|age[9:8] ? 8'hff : age[7:0];
 reg tx_busy,tx_cork;
 reg [3:0] tx_endpoint;
 reg [7:0] tx_pid,tx_count,offer_age,accept_age;
 wire finish=tx_busy && !tx_valid_sample && pending && tx_endpoint==pending_ep;
 wire expired=pending && ((age==127 && (!tx_busy || tx_count==0)) || age==1023);
 wire displaced=next_token && pending && !finish;
 wire missing=expired || displaced;
 wire event_valid=armed && (finish || missing);
 wire malformed=finish && !((tx_pid==8'h5a && tx_count==1) ||
                   (pending_ep==2 && (tx_pid==8'hc3 || tx_pid==8'h4b) && tx_count>=3));
 wire [63:0] event_record={pending_time,1'b1,missing,malformed,tx_cork,
                           pending_ep,tx_pid,accept_age,tx_count};
 reg [255:0] history;
 reg [63:0] first_fault;
 reg [1023:0] snapshot;
 reg sending;
 reg [4:0] send_index;
 assign cmd_pop=cmd_valid && !sending && !ureset;
 wire clear=cmd_pop && command[0];
 wire [1023:0] live={
  {21'd0,armed,pending,2'd0,target},first_fault,history,
  {16'd0,max_accept,max_offer},last_response,last_token,stalls,sofs,resets,
  bad_token,rx_errors,bad_response,miss3,miss2,data3,data2,nak3,nak2,in3,in2,ticks,
  command,32'd1,32'h55544e41};
 always @(posedge clk)begin
  if(ureset)begin sending<=0;send_index<=0;snapshot<=0;end
  else begin
   // Every command returns a complete bank, including clear/disarm, so the
   // CPU can acknowledge execution without sampling an asynchronous flag.
   if(cmd_pop)begin snapshot<=live;sending<=1;send_index<=0;end
   else if(sending && reply_ready)begin
    send_index<=send_index+1'b1;
    if(send_index==31)sending<=0;
   end
  end
 end
 always @(posedge clk)begin
  if(ureset || clear)begin
   armed<=!ureset;target<=ureset ? 7'd0 : command[14:8];
   ticks<=0;in2<=0;in3<=0;nak2<=0;nak3<=0;data2<=0;data3<=0;miss2<=0;miss3<=0;
   bad_response<=0;rx_errors<=0;bad_token<=0;resets<=0;sofs<=0;stalls<=0;
   last_token<=0;last_response<=0;max_offer<=0;max_accept<=0;
   history<=0;first_fault<=0;bus_reset_q<=bus_reset_sample;
   rx_count<=0;rx_pid<=0;token0<=0;token1<=0;rx_bad<=0;
   pending<=0;pending_ep<=0;pending_time<=0;age<=0;
   tx_busy<=0;tx_cork<=0;tx_endpoint<=0;tx_pid<=0;tx_count<=0;offer_age<=0;accept_age<=0;

    if(rx_error_sample)rx_errors<=rx_errors+1'b1;
    if(tx_valid_sample && !tx_ready_sample)stalls<=stalls+1'b1;
    if(token_in && !token_good)bad_token<=bad_token+1'b1;
   end
   if(rx_active_sample)begin
    if(rx_error_sample)rx_bad<=1;
    if(rx_valid_sample)begin
     case(rx_count)
      0:rx_pid<=rx_data_sample;
      1:token0<=rx_data_sample;
      2:token1<=rx_data_sample;
      default:;
     endcase
     if(rx_count!=4)rx_count<=rx_count+1'b1;
    end
   end else begin rx_count<=0;rx_bad<=0;end
   if(pending && age!=1023)age<=age+1'b1;

     if(pending_ep==2)miss2<=miss2+1'b1;else miss3<=miss3+1'b1;

     if(tx_pid==8'h5a && tx_count==1)begin
      if(pending_ep==2)nak2<=nak2+1'b1;else nak3<=nak3+1'b1;

     end

   end

  end

// ddr_dcfifo.v — FIFO double horloge (pointeurs Gray) entre le domaine d'acquisition (clk 100 MHz, sortie de
// raw_fifo / event_enc) et le domaine de l'IP DDR3 (clk_out 75 MHz, ddr_writer) — ARCHI_V3 § 9.2.
//
// Écrite à la main plutôt que générée par l'IP FIFO Gowin : une netlist chiffrée ne se simule pas, celle-ci
// passe au banc cocotb (rapports 100/75 et 100/50, vide/plein, 10 000 mots aléatoires, pertes comptées).
// Aucune perte silencieuse : un mot présenté quand la FIFO est pleine est refusé (wr_ready = 0) ET compté
// dans `wr_lost` (domaine écriture) — l'appelant décide quoi en faire (ddr_writer publie le compteur).
// Neo-only isolation experiment: exactly the AHB/DDR command (100-bit) and
// response (64-bit) four-slot specializations use explicit flip-flops. All
// pointers, flags, resets and the registered head latency are unchanged.
// Larger FIFOs and the USB debug command FIFO retain inferred memory.
// The shared Pro Dock source is NOT modified by this variant.
`default_nettype none
module ddr_dcfifo #(
    parameter WIDTH      = 64,
    parameter DEPTH_LOG2 = 9          // 512 mots
) (
    // écriture
    input  wire             wr_clk,
    input  wire             wr_rst,       // synchrone au domaine écriture
    input  wire             wr_valid,
    input  wire [WIDTH-1:0] wr_data,
    output wire             wr_ready,     // = pas pleine
    output wire             wr_full,      // registered fullness, independent of reset
    output reg  [31:0]      wr_lost,      // mots refusés (pleine), jamais remis à zéro sauf reset
    output wire [DEPTH_LOG2:0] wr_count,  // remplissage vu du côté écriture (pessimiste)
    // lecture
    input  wire             rd_clk,
    input  wire             rd_rst,       // synchrone au domaine lecture
    output wire             rd_valid,     // = pas vide ; rd_data valide
    output wire [WIDTH-1:0] rd_data,
    input  wire             rd_ready,     // consomme le mot présenté
    output wire [DEPTH_LOG2:0] rd_count   // remplissage vu du côté lecture (pessimiste)
);
    localparam AW = DEPTH_LOG2;

    wire [WIDTH-1:0] storage_read;

    // --- pointeurs binaires et Gray (AW+1 bits : le bit de poids fort distingue plein de vide) ---
    reg  [AW:0] wr_bin, wr_gray, rd_bin, rd_gray;
    reg  [AW:0] wr_gray_s1, wr_gray_s2;      // wr_gray synchronisé dans rd_clk
    reg  [AW:0] rd_gray_s1, rd_gray_s2;      // rd_gray synchronisé dans wr_clk

    function [AW:0] bin2gray; input [AW:0] b; begin bin2gray = b ^ (b >> 1); end endfunction
    function [AW:0] gray2bin; input [AW:0] g; integer i; reg [AW:0] b;
        begin b = 0; for (i = AW; i >= 0; i = i - 1) b[i] = (i == AW) ? g[i] : b[i+1] ^ g[i]; gray2bin = b; end
    endfunction

    // --- côté écriture ----------------------------------------------------------------------
    // `full` et `wr_count` sont REGISTRÉS (E2-e, 2026-09-20) : calculés à partir du pointeur d'écriture du
    // cycle suivant et du pointeur de lecture synchronisé courant. Sans cela, la chaîne Gray → gray2bin →
    // comparaison → wr_ready → rd_ready de raw_fifo (clk100) et → m1_urgent → arbitre → IP (75 MHz) ne fermait
    // plus le timing (−0,32 ns sur clk100, −0,27 ns sur ddr3_sys_clk, build e2e_con2). Pessimisme d'un cycle
    // sur la libération d'une place : sans effet sur les comptes (le lecteur libère bien avant le plein).
    wire [AW:0] rd_bin_w = gray2bin(rd_gray_s2);
    reg         full;
    reg  [AW:0] wr_count_r;
    wire        accept   = wr_valid && !full;
    wire [AW:0] wr_bin_n = wr_bin + {{AW{1'b0}}, accept};
    // Refus pendant le vidage (E2-h, 2026-09-20) : avec wr_ready = !full seul, une écriture arrivant pendant wr_rst était
    // acceptée puis effacée par la remise à zéro du pointeur — la première entrée de chaque capture V3 (état initial,
    // t = 0) disparaissait ; l'amont (out_slice / raw_fifo) retient tant que wr_ready = 0, rien n'est perdu.
    assign wr_ready = !full && !wr_rst;
    assign wr_full = full;
    assign wr_count = wr_count_r;

    always @(posedge wr_clk) begin
        if (wr_rst) begin
            wr_bin <= 0; wr_gray <= 0; wr_lost <= 0; rd_gray_s1 <= 0; rd_gray_s2 <= 0; full <= 1'b0; wr_count_r <= 0;
        end else begin
            rd_gray_s1 <= rd_gray; rd_gray_s2 <= rd_gray_s1;
            full       <= (wr_bin_n[AW] != rd_bin_w[AW]) && (wr_bin_n[AW-1:0] == rd_bin_w[AW-1:0]);
            wr_count_r <= wr_bin_n - rd_bin_w;
            if (accept) begin
                wr_bin  <= wr_bin + 1'b1;
                wr_gray <= bin2gray(wr_bin + 1'b1);
            end else if (wr_valid && full) begin
                wr_lost <= wr_lost + 1'b1;
            end
        end
    end

    // Data is published only by the synchronized write pointer. Reading
    // into head retires the RAM slot; the synchronized read pointer delays
    // its reuse. No additional register or pointer stage is introduced.
    generate if (AW == 2 && (WIDTH == 100 || WIDTH == 64)) begin : g_register_storage
        wire [(1<<AW)*WIDTH-1:0] words;
        for (genvar slot = 0; slot < (1<<AW); slot = slot + 1) begin : g_slot
            (* syn_preserve = 1 *) reg [WIDTH-1:0] value;
            always @(posedge wr_clk)
                if (!wr_rst && accept && wr_bin[AW-1:0] == AW'(slot))
                    value <= wr_data;
            assign words[slot*WIDTH +: WIDTH] = value;
        end
        assign storage_read = words[rd_bin[AW-1:0]*WIDTH +: WIDTH];
    end else begin : g_inferred_storage
        reg [WIDTH-1:0] mem [0:(1<<AW)-1];
        always @(posedge wr_clk)
            if (!wr_rst && accept) mem[wr_bin[AW-1:0]] <= wr_data;
        assign storage_read = mem[rd_bin[AW-1:0]];
    end endgenerate

    // --- côté lecture : mot de tête présenté en sortie (registre de tête rechargé à chaque pop) ----
    // Pointeur d'écriture REGISTRÉ après le décodage Gray (build USB 24, 2026-09-21) : la chaîne gray2bin (XOR en série)
    // + comparaison + validation du pop tenait en un cycle par chance de placement (design fusionné : −0,09 ns / 17 chemins
    // clk100 sur le port m2). Une image en retard d'un cycle est prudente (empty un cycle de plus, jamais un faux non-vide).
    wire [AW:0] wr_bin_w = gray2bin(wr_gray_s2);
    reg  [AW:0] wr_bin_r;
    wire        empty    = (rd_bin == wr_bin_r);
    reg  [WIDTH-1:0] head;
    reg              head_valid;
    reg  [AW:0]      rd_count_r;               // registré (même raison que wr_count) : alimente m1_urgent
    assign rd_valid = head_valid;
    assign rd_data  = head;
    assign rd_count = rd_count_r;

    // La mémoire est lue de façon synchrone : quand la tête est libre (ou consommée ce cycle) et que la
    // FIFO n'est pas vide, on charge mem[rd_bin] dans head au front suivant et on avance rd_bin.
    wire pop = !empty && (!head_valid || rd_ready);
    always @(posedge rd_clk) begin
        if (rd_rst) begin
            rd_bin <= 0; rd_gray <= 0; wr_gray_s1 <= 0; wr_gray_s2 <= 0; wr_bin_r <= 0; head_valid <= 1'b0; head <= 0; rd_count_r <= 0;
        end else begin
            wr_gray_s1 <= wr_gray; wr_gray_s2 <= wr_gray_s1; wr_bin_r <= wr_bin_w;
            rd_count_r <= wr_bin_r - rd_bin;
            if (pop) begin
                head       <= storage_read;
                head_valid <= 1'b1;
                rd_bin     <= rd_bin + 1'b1;
                rd_gray    <= bin2gray(rd_bin + 1'b1);
            end else if (rd_ready && head_valid) begin
                head_valid <= 1'b0;
            end
        end
    end
endmodule
`default_nettype wire
