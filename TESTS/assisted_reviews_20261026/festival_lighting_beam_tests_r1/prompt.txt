typedef struct { uint16_t r,g,b,alpha; } FestivalLightingLamp;
static inline float festival_lighting_unit(float x){
    return isfinite(x)?fminf(1,fmaxf(0,x)):0;
}
static inline float festival_lighting_step(float current,unsigned night,float dt){
    current=festival_lighting_unit(current);
    if(!isfinite(dt)||dt<=0)return current;
    float weight=1-expf(-1.5f*fminf(dt,.1f));
    return festival_lighting_unit(current+((night?1.0f:0)-current)*weight);
}
