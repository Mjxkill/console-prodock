/* festival_lighting_test.c
   C99 conformance/regression tests for FestivalLightingLamp pure coefficients. */
#include <assert.h>
#include <math.h>
#include <stdio.h>

#include "festival_lighting.h"

static float run_seconds(float start, int night, float seconds, int hz)
{
    float x = start;
    float dt = 1.0f / (float)hz;
    int n = (int)lroundf(seconds * (float)hz);

    for (int i = 0; i < n; ++i)
        x = festival_lighting_step(x, night, dt);
    return x;
}

static int channel_delta(uint16_t a, uint16_t b)
{
    return (a > b) ? (int)(a - b) : (int)(b - a);
}

static void test_unit_finite_bounds(void)
{
    assert(festival_lighting_unit(NAN) == 0.0f);
    assert(festival_lighting_unit(INFINITY) == 0.0f);
    assert(festival_lighting_unit(-INFINITY) == 0.0f);
    assert(festival_lighting_unit(-1.0f) == 0.0f);
    assert(festival_lighting_unit(2.0f) == 1.0f);
    assert(festival_lighting_unit(0.25f) == 0.25f);

    assert(festival_lighting_step(0.3f, 1, NAN) == 0.3f);
    assert(festival_lighting_step(0.3f, 1, -0.05f) == 0.3f);
    assert(festival_lighting_step(NAN, 0, 0.1f) == 0.0f);
    assert(isfinite(festival_lighting_step(0.3f, 0, 1000.0f)));
}

static void test_day_night_monotonic(void)
{
    float x = 0.0f;
    float prev = x;

    for (int i = 0; i < 80; ++i) {
        x = festival_lighting_step(x, 1, 0.1f);
        assert(x >= prev - 1.0e-7f);
        prev = x;
    }
    assert(x > 0.95f);

    x = 1.0f;
    prev = x;
    for (int i = 0; i < 80; ++i) {
        x = festival_lighting_step(x, 0, 0.1f);
        assert(x <= prev + 1.0e-7f);
        prev = x;
    }
    assert(x < 0.05f);
}

static void test_cadence_invariance(void)
{
    float at_10 = run_seconds(0.0f, 1, 1.0f, 10);
    float at_30 = run_seconds(0.0f, 1, 1.0f, 30);
    float at_60 = run_seconds(0.0f, 1, 1.0f, 60);

    assert(fabsf(at_10 - at_30) <= 1.0e-5f);
    assert(fabsf(at_10 - at_60) <= 1.0e-5f);
    assert(fabsf(at_30 - at_60) <= 1.0e-5f);

    at_10 = run_seconds(0.2f, 0, 2.0f, 10);
    at_30 = run_seconds(0.2f, 0, 2.0f, 30);
    at_60 = run_seconds(0.2f, 0, 2.0f, 60);
    assert(fabsf(at_10 - at_30) <= 1.0e-5f);
    assert(fabsf(at_10 - at_60) <= 1.0e-5f);
}

static void test_no_freeze_after_1000s(void)
{
    float x = 0.2f;

    for (int i = 0; i < 10000; ++i)
        x = festival_lighting_step(x, 1, 0.1f);

    assert(fabsf(x - 1.0f) <= 1.0e-4f);
    assert(x > 0.99f);
}

static void test_lamp_bounds_long_times(void)
{
    const float moods[] = { 0.0f, 0.25f, 0.5f, 0.75f, 1.0f, -1.0f, NAN };
    const float times[] = { 0.0f, -1.0f, 1.0f, 37.0f, 1000.0f };

    for (unsigned lamp = 0; lamp < 16; ++lamp) {
        for (size_t mi = 0; mi < sizeof(moods) / sizeof(moods[0]); ++mi) {
            for (size_t ti = 0; ti < sizeof(times) / sizeof(times[0]); ++ti) {
                FestivalLightingLamp p = festival_lighting_lamp(lamp, moods[mi], times[ti]);
                assert(p.r <= 256);
                assert(p.g <= 256);
                assert(p.b <= 256);
                assert(p.alpha <= 256);
            }
        }
    }
}

static void test_uint_quantization_continuity(void)
{
    for (unsigned lamp = 0; lamp < 4; ++lamp) {
        const float base_time = 7.25f;
        const float base_mood = 0.37f;

        FestivalLightingLamp a = festival_lighting_lamp(lamp, base_mood, base_time);
        FestivalLightingLamp b = festival_lighting_lamp(lamp, base_mood, base_time + 0.001f);

        assert(channel_delta(a.r, b.r) <= 2);
        assert(channel_delta(a.g, b.g) <= 2);
        assert(channel_delta(a.b, b.b) <= 2);
        assert(channel_delta(a.alpha, b.alpha) <= 2);

        a = festival_lighting_lamp(lamp, base_mood, base_time);
        b = festival_lighting_lamp(lamp, base_mood + 0.001f, base_time);
        assert(channel_delta(a.r, b.r) <= 2);
        assert(channel_delta(a.g, b.g) <= 2);
        assert(channel_delta(a.b, b.b) <= 2);
        assert(channel_delta(a.alpha, b.alpha) <= 2);
    }
}

int main(void)
{
    test_unit_finite_bounds();
    test_day_night_monotonic();
    test_cadence_invariance();
    test_no_freeze_after_1000s();
    test_lamp_bounds_long_times();
    test_uint_quantization_continuity();

    puts("festival_lighting tests: PASS");
    return 0;
}
