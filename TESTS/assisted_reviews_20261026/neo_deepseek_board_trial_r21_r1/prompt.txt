25:     unsigned(value, 64, 'capture')
26:     unsigned(epoch, 1, 'epoch')
27:     if value >> 60 != 0xd or (value >> 55) & 1:
28:         raise ValueError('Not the D* capture format, or reserved bit set')

30:         raise ValueError('Capture belongs to another acknowledged ARM epoch')

37:     armed = bool((value >> 51) & 1)
