1400:     parameter P_ASIZE    = 9,
1401:     // 1 : l'application écrit avec l'horloge du contrôleur USB ; la FIFO de changement
1402:     // d'horloge (64 octets en registres) est alors supprimée. 0 : comportement d'origine.
1403:     parameter P_SYNC     = 0
1404: )
1405: (
1406:     input                      i_clk         ,//clock
1407:     input                      i_reset       ,//reset
1408:     input      [3:0]           i_usb_endpt   ,//
1409:     input                      i_usb_txact   ,//
1410:     input                      i_usb_txpop   ,//
1411:     input                      i_usb_txpktfin,//

1413:     output                     o_usb_txcork  ,//
1414:     output     [P_ASIZE:0]     o_usb_txlen   ,//
1415:     input                      i_ep_clk      ,//
1416:     input                      i_ep_tx_dval  ,//
1417:     input      [P_DSIZE-1:0]   i_ep_tx_data   //

1421:     wire               pkt_fifo_wr;

1426:     reg                pkt_fifo_rd_dval;
1427:     wire [P_DSIZE-1:0] pkt_fifo_rd_data;

1434:     reg                c_fifo_rd_dval;
1435:     wire [P_DSIZE-1:0] c_fifo_rd_data;

1437:     assign c_fifo_wr_data = i_ep_tx_data;

1440:     clk_cross_fifo #(
1441:        .DSIZE (8  )
1442:       ,.ASIZE (6  )

1450:         ,.Data       (c_fifo_wr_data)
1451:         ,.AlmostFull (              )
1452:         ,.Full       ()
1453:         ,.RdClock    (i_clk         )
1454:         ,.RPReset    (i_reset       )
1455:         ,.RdEn       (c_fifo_rd     )
1456:         ,.Q          (c_fifo_rd_data)
1457:         ,.AlmostEmpty()

1466:     end endgenerate
1467:     always@(posedge i_clk, posedge i_reset) begin
1468:         if (i_reset) begin
1469:             c_fifo_rd <= 1'b0;
1470:         end
1471:         else begin
1472:             if (c_fifo_empty) begin
1473:                 c_fifo_rd <= 1'b0;
1474:             end
1475:             else begin
1476:                 c_fifo_rd <= 1'b1;
1477:             end
1478:         end
1479:     end

1481:         if (i_reset) begin
1482:             c_fifo_rd_dval <= 1'b0;
1483:         end
1484:         else begin
1485:             c_fifo_rd_dval <= c_fifo_rd & (!c_fifo_empty);
1486:         end
1487:     end
