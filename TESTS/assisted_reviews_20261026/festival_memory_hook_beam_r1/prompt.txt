    unsigned completed=renderer.frame-measured_frame;
    if(benchmark&&(!measured_frame||completed!=frames))ok=0;

    printf("LUMA3D_AMBER region=%u phase=%u point=%u intake=%u sluice=%u\n",game.story.region,game.story.amber_phase,game.story.amber_point,game.story.amber_intake,game.story.amber_sluice);
    printf("LUMA3D_WORKSHOP phase=%u mode=%u point=%u copper=%u blue=%u\n",game.story.workshop_phase,game.story.workshop_mode,game.story.workshop_point,game.story.wind_a,game.story.wind_b);
    printf("LUMA3D_RESULT frames=%u elapsed=%.3f fps=%.2f commands=%u batches=%u feed_seconds=%.3f present_seconds=%.3f draws_last=%u triangles_last=%u benchmark=%d hardware=%d warmup=%u\n",completed,elapsed,elapsed>0?completed/elapsed:0,renderer.total_commands-measured_commands,renderer.total_flushes-measured_batches,renderer.feed_seconds-measured_feed,renderer.present_seconds-measured_present,renderer.draws,renderer.triangles,benchmark,!host,benchmark?warmup:0);fflush(stdout);
    if(terminal) {tcsetattr(STDIN_FILENO,TCSANOW,&saved);if(stdin_flags>=0)fcntl(STDIN_FILENO,F_SETFL,stdin_flags);}
    if(save_path) {
        if(l3_progress_write(save_path,&game,&progress_schema)!=L3_PROGRESS_OK) {fputs("Final save failed\n",stderr);ok=0;}
        else {puts("LUMA3D_SAVE saved");fflush(stdout);}
    }
    if(sample_actions){l3_sample_report(&sample,festival,!host);if(!l3_sample_valid(&sample,festival,festival_game))ok=0;}
    if(festival_start)printf("LUMA3D_FESTIVAL active=%u mode=%u visited=%x won=%x time=%.3f hardware=%d\n",festival->active,festival->kind,festival->visited,festival->won,festival->time,!host);
    if(festival_save&&!l3_festival_save(festival,festival_save))ok=0;
    if(memory_save&&!l3_festival_memory_save(festival,memory_save))ok=0;
    l3_festival_destroy(festival);renderer.festival=NULL;
    if(!l3_render_close(&renderer))ok=0;return !ok;
}

FILE memory_progress.c
#define _POSIX_C_SOURCE 200809L
#include "memory_progress.h"
#include <errno.h>
#include <fcntl.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>
#include <sys/stat.h>
#include <unistd.h>
static uint32_t get(const uint8_t *p){return p[0]|(uint32_t)p[1]<<8|(uint32_t)p[2]<<16|(uint32_t)p[3]<<24;}
static void put(uint8_t *p,uint32_t x){for(unsigned i=0;i<4;i++)p[i]=x>>(8*i);}
static uint32_t crc(const uint8_t *p,unsigned n){uint32_t c=~0u;while(n--){c^=*p++;for(unsigned i=0;i<8;i++)c=(c>>1)^(0xedb88320u&-(c&1));}return ~c;}
static int valid(const L3MemoryProgress *p){return p&&p->visited<=1&&l3_memory_validate(&p->game)&&(p->visited||(!p->game.turns&&!p->game.wins[0]&&p->game.phase==L3_MEMORY_READY));}
static int decode(L3MemoryProgress *p,const uint8_t b[152]){
