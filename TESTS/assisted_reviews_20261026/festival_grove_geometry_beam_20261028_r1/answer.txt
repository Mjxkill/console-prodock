## Faits extraits du paquet

- Budget cible total : 504 triangles, dont 240 pour les couronnes.
- Fonction concernée uniquement : couronne par index `i`, paramètres `TreeParams` existants.
- `i % 3 == 0` : 12 couronnes sapin, 4 triangles chacune.
- `i % 3 != 0` : 24 couronnes larges, 8 triangles chacune.
- 12 × 4 + 24 × 8 = 48 + 192 = 240 triangles de couronne.
- Contraintes de placement : hors carré accessible, `max(|x|,|z|) >= 19`, `width <= 2.7`, `height` dans 4.2..6.4, inclinaison `<= 0.3 m`.
- Base de canopée `>= height * 0.34`.
- Callback : `int tri(void*, Vec3, Vec3, Vec3, unsigned, unsigned, unsigned)`.

## Hypothèses

- `TreeParams` contient `x,z,height,width,lean,yaw`.
- `yaw` est en radians dans le repère horizontal.
- La callback est appelée avec un contexte utilisateur `ctx`; si le harness utilise seulement 3 arguments, `ctx` peut être passé à `NULL`.
- La géométrie finale est validée par le parent; le code ici vise la structure, les comptages et l’absence de triangles très dégénérés.

## Code candidat

```c
#include <math.h>

typedef struct { float x, y, z; } Vec3;
typedef struct { float x, z, height, width, lean, yaw; } TreeParams;
typedef int (*crown_emit_fn)(void *, Vec3, Vec3, Vec3, unsigned, unsigned, unsigned);

int crown(int i, TreeParams p, crown_emit_fn emit, void *ctx) {
    int n = 0;
    if (i % 3 != 0) {
        float r = p.width * 0.92f;
        float mid = p.height * 0.62f;
        float hh = p.width * 0.40f;
        float bot = mid - hh;
        float minbot = p.height * 0.34f;
        if (bot < minbot) { mid = minbot + hh; bot = minbot; }
        float top = mid + hh;

        float lean = p.lean + 0.08f * sinf((float)i * 0.8f);
        Vec3 t = { p.x + lean * sinf(p.yaw) + 0.08f * cosf((float)i * 1.9f),
                   top + 0.08f * sinf((float)i * 0.9f),
                   p.z + lean * cosf(p.yaw) + 0.08f * sinf((float)i * 2.3f) };
        Vec3 b = { p.x + 0.06f * sinf((float)i * 1.3f),
                   bot,
                   p.z + 0.06f * cosf((float)i * 2.7f) };

        Vec3 e[4];
        for (int k = 0; k < 4; ++k) {
            float a = p.yaw + (float)k * 1.5707963f
                      + 0.25f * sinf((float)i * 0.7f + (float)k);
            float rr = r * (1.0f + 0.07f * sinf((float)i * 1.1f + (float)k * 2.3f));
            e[k].x = p.x + rr * cosf(a);
            e[k].y = mid;
            e[k].z = p.z + rr * sinf(a);
        }

        unsigned tr = 90  + (unsigned)((i * 17) % 31);
        unsigned tg = 145 + (unsigned)((i * 23) % 41);
        unsigned tb = 65  + (unsigned)((i * 11) % 26);
        unsigned br = 55  + (unsigned)((i * 19) % 26);
        unsigned bg = 105 + (unsigned)((i * 29) % 31);
        unsigned bb = 45  + (unsigned)((i * 7)  % 21);

        for (int k = 0; k < 4; ++k) {
            int j = (k + 1) & 3;
            emit(ctx, t, e[k], e[j], tr, tg, tb); n++;
            emit(ctx, b, e[j], e[k], br, bg, bb); n++;
        }
    } else {
        float baseY = p.height * 0.48f;
        float apexY = p.height * 0.88f + 0.18f * sinf((float)i * 0.7f);
        float baseR = p.width * 0.46f;

        Vec3 a = { p.x + 0.06f * sinf((float)i * 1.2f),
                   apexY,
                   p.z + 0.06f * cosf((float)i * 1.8f) };
        Vec3 bs[4];
        for (int k = 0; k < 4; ++k) {
            float ang = p.yaw + (float)k * 1.5707963f;
            float rr = baseR * (1.0f + 0.05f * sinf((float)i * 1.4f + (float)k));
            bs[k].x = p.x + rr * cosf(ang);
            bs[k].y = baseY;
            bs[k].z = p.z + rr * sinf(ang);
        }

        unsigned fr = 48 + (unsigned)((i * 13) % 28);
        unsigned fg = 88 + (unsigned)((i * 19) % 32);
        unsigned fb = 42 + (unsigned)((i * 9)  % 24);

        for (int k = 0; k < 4; ++k) {
            int j = (k + 1) & 3;
            emit(ctx, a, bs[k], bs[j], fr, fg, fb); n++;
        }
    }
    return n;
}
```

## Preuves attendues

```c
#include <assert.h>

static int sink(void *ctx, Vec3 a, Vec3 b, Vec3 c,
                unsigned r, unsigned g, unsigned bl) {
    (void)a; (void)b; (void)c; (void)r; (void)g; (void)bl;
    ++*(int *)ctx;
    return 1;
}

static TreeParams P(float x, float z, float h, float w, float lean, float yaw) {
    TreeParams p = { x, z, h, w, lean, yaw };
    return p;
}

/* Assertion 1 */
{
    int c = 0;
    TreeParams p = P(21.0f, -19.0f, 5.0f, 2.2f, 0.2f, 0.7f);
    assert(crown(0, p, sink, &c) == 4);
    assert(c == 4);
}

/* Assertion 2 */
{
    int c = 0;
    TreeParams p = P(-20.0f, 22.0f, 5.8f, 2.6f, 0.15f, 1.2f);
    assert(crown(1, p, sink, &c) == 8);
    assert(c == 8);
}

/* Assertion 3 */
{
    int total = 0;
    TreeParams p = P(21.0f, -19.0f, 5.0f, 2.4f, 0.2f, 0.7f);
    for (int i = 0; i < 36; ++i) {
        int c = 0;
        crown(i, p, sink, &c);
        total += c;
    }
    assert(total == 240);
}
```

## Limites / évidence manquante

- Types `TreeParams` et convention de `yaw` supposés selon le paquet; la compilation finale dépend des définitions réelles du projet.
- Le contexte de la callback est passé en `void *ctx`; si le harness attend strictement 3 arguments, il faut adapter l’appel à `crown(i, p, sink, NULL)` ou supprimer ce paramètre.
- Le code ne valide pas visuellement la géométrie; le parent doit vérifier les images GPU, les collisions visuelles et la contrainte de couronne hors du carré accessible.
